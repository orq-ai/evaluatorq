1"""Anthropic prompt-cache breakpoints for multi-turn conversations. 

2 

3Anthropic caching is **opt-in**: without a ``cache_control`` breakpoint there is 

4no caching at all, however byte-stable the prefix is. An append-only transcript 

5buys nothing on its own — it only makes a breakpoint worth placing. 

6 

7OpenAI, Gemini 2.0+, DeepSeek and xAI cache the prefix automatically, so they 

8need no request change and OpenAI's ``prompt_cache_key`` is not set anywhere — 

9it is a routing hint, not an enabler. 

10 

11**Callers gate on** `caching_applies`, which requires *both* the Orq router and 

12an Anthropic model. The marker is only known to be tolerated by the router on an 

13Anthropic backend; ``cache_control`` inside a content part is not in the direct 

14OpenAI schema (the installed SDK's ``ChatCompletionContentPartTextParam`` defines 

15``type`` and ``text`` only), and no one has measured what a routed non-Anthropic 

16model or a self-hosted OpenAI-compatible server does with it. 

17 

18Only apply these to a conversation that is replayed with a growing prefix (an 

19agent-simulation turn loop, a red-team attack thread). A cache **write** costs 

201.25x the input tokens, so marking a one-shot prompt is a straight loss — which 

21is why both helpers refuse to mark an input below `CACHE_MIN_PROMPT_TOKENS`. 

22 

23Placement rules (Orq docs, ``/docs/ai-gateway/features/prompt-caching``): the 

24breakpoint marks the *end* of the cacheable prefix and there are at most 4 per 

25request. 

26 

27Both APIs take a **positioned, per-item** marker — `apply_cache_breakpoints` for 

28Chat Completions, `mark_responses_input` for Responses. Responses also accepts a 

29*top-level* ``cache_control`` body field, which marks the end of the whole input 

30and therefore cannot be kept off a rebuilt trailing item; it is deliberately not 

31used here. 

32 

33Never set ``ttl``: 5m is the default, 1h costs more, and only Anthropic honours 

34it. 

35"""

50CACHE_MIN_PROMPT_TOKENS = 1024 

51"""Below this, no Anthropic model caches and a breakpoint is a pure 1.25x loss. 

52 

53A **necessary, not sufficient** condition: Orq's table gives per-model floors of 

54512 (Opus 5, Fable 5), 1024 (Sonnet 4.6/5, Opus 4.8), 2048 (Opus 4.7, Haiku 3.5) 

55and 4096 (Haiku 4.5, Opus 4.5/4.6). Gating on the lowest common floor that 

56applies to the models we actually run avoids a per-model table that would rot 

57with every release; a prompt between 1024 and its model's real floor still pays a 

58write nothing reads, which is the same behaviour as before this guard and strictly 

59better than paying it on a two-line prompt. 

60""" 

61 

62_CHARS_PER_TOKEN = 4 

63"""Crude estimator, deliberately not a tokenizer. 

64 

65The check only has to separate "a two-line one-shot prompt" from "a replayed 

66transcript", and importing a tokenizer for that would add a dependency and a 

67per-call cost to save nothing. It over-estimates tokens for text with long 

68words, which errs toward marking — the same behaviour as before this guard 

69existed. 

70""" 

71 

72# Only these carry a plain-text body we can safely re-render as a block list. 

73# `tool` and tool-calling `assistant` turns are left alone: their content shape 

74# is load-bearing for the provider and a breakpoint there buys nothing extra. 

75_MARKABLE_ROLES = frozenset({'system', 'user'})

78def caching_applies(client: AsyncOpenAI | None, model: str) -> bool: 

79 """True when a breakpoint on this ``client``/``model`` can pay off. 

80 

81 The **router** is required: ``cache_control`` inside a content part is outside 

82 the direct OpenAI schema, and a self-hosted OpenAI-compatible server may 

83 reject the whole request rather than ignore the key. Orq's own docs are 

84 explicit that on the router it is safe either way — *"``cache_control`` on a 

85 non-Anthropic model is ignored, not rejected"* 

86 (``/docs/ai-gateway/features/anthropic-messages-api``) — so this second check 

87 is not a safety gate but a scope one: every other provider caches 

88 automatically, so marking them changes the request shape for no gain. 

89 

90 ``agent/<key>`` is included because an Orq agent deployment resolves its model 

91 server-side: we cannot see whether it is Anthropic, and excluding it would 

92 leave the default red-team and simulation target — the largest replayed 

93 transcript in either surface — uncached. The documented ignore-don't-reject 

94 behaviour is what makes that safe. 

95 """ 

96 if not client_routes_through_orq(client): 

97 return False 

98 name = model.lower() 

99 return name.startswith(('anthropic/', 'agent/')) or 'claude' in name

102def _text_length(value: Any) -> int: 

103 """Total length of the text a message/item content carries, any shape.""" 

104 if isinstance(value, str): 

105 return len(value) 

106 if isinstance(value, list): 

107 return sum(len(part.get('text', '')) for part in value if isinstance(part, dict)) 

108 return 0

111def _clears_minimum(items: list[dict[str, Any]]) -> bool: 

112 total = sum(_text_length(item.get('content')) for item in items) 

113 return total >= CACHE_MIN_PROMPT_TOKENS * _CHARS_PER_TOKEN

116def _prefix_index( 

117 items: list[dict[str, Any]], 

118 volatile_tail: int, 

119 is_markable: Callable[[dict[str, Any]], bool], 

120 *, 

121 unit: str, 

122) -> int: 

123 """Index of the last markable item at or before the end of the prefix. 

124 

125 Walks backwards past trailing items that cannot carry a breakpoint (an 

126 ``assistant`` reply, a ``tool`` result, a ``function_call``). Returns ``-1`` 

127 when there is none. 

128 

129 Raises: 

130 ValueError: if ``volatile_tail`` is negative. 

131 """ 

132 if volatile_tail < 0: 

133 raise ValueError(f'volatile_tail must be >= 0, got {volatile_tail}') 

134 index = len(items) - 1 - volatile_tail 

135 while index >= 0 and not is_markable(items[index]): 

136 index -= 1 

137 if index < 0 and items: 

138 logger.warning( 

139 'No cacheable {} in the prompt-cache prefix ({} {}s, volatile_tail={}): the transcript ' 

140 'is re-encoded on every turn.', 

141 unit, 

142 len(items), 

143 unit, 

144 volatile_tail, 

145 ) 

146 return index

149def _is_markable(message: dict[str, Any]) -> bool: 

150 content = message.get('content') 

151 return message.get('role') in _MARKABLE_ROLES and isinstance(content, str) and bool(content)

154def apply_cache_breakpoints(messages: list[dict[str, Any]], *, volatile_tail: int) -> list[dict[str, Any]]: 

155 """Return a copy of ``messages`` with breakpoints on the system + prefix end. 

156 

157 Two of the four allowed breakpoints, deduped to one when the system message 

158 *is* the prefix end: 

159 

160 - the leading ``system`` message — static for the whole conversation, so it 

161 is a cache read on every turn after the first; 

162 - the end of the **persisted** prefix, so the *next* turn (which appends past 

163 it) reads the whole transcript back. 

164 

165 ``volatile_tail`` is the number of trailing messages the caller rebuilds on 

166 every turn rather than appending to a transcript — a per-call instruction, a 

167 re-rendered scratchpad. They are excluded from the prefix. Marking one is 

168 worse than marking nothing: the next turn puts persisted content at that 

169 position, the prefix diverges immediately after the system message, and the 

170 whole transcript pays a 1.25x write it can never read back. 

171 

172 It is **required, with no default**, on purpose: a caller that rebuilds its 

173 last message and does not say so gets a silent per-turn write and no read, 

174 which is the expensive failure and shows up as a bill rather than a bug. Pass 

175 ``0`` when the whole list persists into the next turn. 

176 

177 The leading ``system`` breakpoint ignores ``volatile_tail`` deliberately: the 

178 system message is built by the framework from a stable prompt, not by the 

179 caller per turn, so it is a valid read even when every other message is 

180 volatile. 

181 

182 Returns ``messages`` unchanged when the total text is below 

183 `CACHE_MIN_PROMPT_TOKENS` — the provider would cache nothing and the write 

184 premium would be pure loss. 

185 

186 A message is skipped when its role is not ``system``/``user`` or its content 

187 is not a non-empty string — a caller that already built content blocks owns 

188 its own breakpoints, and re-wrapping would clobber them. 

189 

190 Raises: 

191 ValueError: if ``volatile_tail`` is negative. 

192 """ 

193 out = list(messages) 

194 prefix_end = _prefix_index(out, volatile_tail, _is_markable, unit='message') 

195 if not _clears_minimum(out): 

196 logger.debug( 

197 'Prompt below the {}-token cache minimum ({} messages) — no breakpoint placed.', 

198 CACHE_MIN_PROMPT_TOKENS, 

199 len(out), 

200 ) 

201 return out 

202 leading = 0 if out and out[0].get('role') == 'system' else -1 

203 # A set, not a pair: dedupes when the prefix end *is* the system message, and 

204 # empties out on an empty list. 

205 for index in {i for i in (leading, prefix_end) if 0 <= i < len(out) and _is_markable(out[i])}: 

206 out[index] = { 

207 **out[index], 

208 'content': [{'type': 'text', 'text': out[index]['content'], 'cache_control': {'type': 'ephemeral'}}], 

209 } 

210 return out

213def _has_markable_parts(item: dict[str, Any]) -> bool: 

214 """A message item carrying text we can attach a breakpoint to. 

215 

216 Excludes `function_call` / `function_call_output` items, which have no 

217 ``content`` and whose shape is load-bearing for the API. 

218 """ 

219 content = item.get('content') 

220 if 'role' not in item: 

221 return False 

222 if isinstance(content, str): 

223 return bool(content) 

224 return isinstance(content, list) and bool(content) and all(isinstance(part, dict) for part in content)

227def mark_responses_input(input_items: list[dict[str, Any]], *, volatile_items: int) -> list[dict[str, Any]]: 

228 """Return a copy of a Responses ``input`` list with a positioned breakpoint. 

229 

230 The Responses `input` is a list of items carrying content parts, exactly like 

231 a chat message list, and the router honours a **per-item** ``cache_control`` 

232 on the last part of an item. Marking the end of the persisted prefix keeps 

233 the breakpoint off a trailing item the caller rebuilds. 

234 

235 The count is ``volatile_items``, **not** ``volatile_tail``: one `Message` 

236 can render to several Responses items (an assistant turn with tool calls 

237 becomes a content item plus one ``function_call`` per call), so a count of 

238 messages would land the breakpoint inside the rebuilt region. Callers holding 

239 messages compute the item count with `responses_volatile_items`. 

240 

241 Preferred over the top-level ``cache_control`` body field, which marks the 

242 end of the whole input: the two compose, but the top-level one then adds a 

243 write at the end that nothing reads back. 

244 

245 Returns ``input_items`` unchanged when the total text is below 

246 `CACHE_MIN_PROMPT_TOKENS`. 

247 

248 Raises: 

249 ValueError: if ``volatile_items`` is negative. 

250 """ 

251 out = list(input_items) 

252 index = _prefix_index(out, volatile_items, _has_markable_parts, unit='item') 

253 if index < 0: 

254 return out 

255 if not _clears_minimum(out): 

256 logger.debug( 

257 'Responses input below the {}-token cache minimum ({} items) — no breakpoint placed.', 

258 CACHE_MIN_PROMPT_TOKENS, 

259 len(out), 

260 ) 

261 return out 

262 content = out[index]['content'] 

263 # `messages_to_responses_input` renders a plain user turn as a bare string and 

264 # only an assistant turn as parts, so both shapes arrive here. A string is 

265 # promoted to a single `input_text` part, which the API accepts for every 

266 # non-assistant role (an assistant turn is always already a part list). 

267 parts: list[dict[str, Any]] = ( 

268 [{'type': 'input_text', 'text': content}] if isinstance(content, str) else list(content) 

269 ) 

270 parts[-1] = {**parts[-1], 'cache_control': {'type': 'ephemeral'}} 

271 out[index] = {**out[index], 'content': parts} 

272 return out

275def responses_volatile_items(messages: list[Any], *, volatile_tail: int) -> int: 

276 """How many Responses ``input`` items the last ``volatile_tail`` messages render to. 

277 

278 `messages_to_responses_input` is a per-message flat-map, so the item count of 

279 a suffix is exactly the item count that suffix contributes to the whole. This 

280 is the only supported way to turn a message count into the ``volatile_items`` 

281 that `mark_responses_input` takes. 

282 

283 Raises: 

284 ValueError: if ``volatile_tail`` is negative. 

285 """ 

286 if volatile_tail < 0: 

287 raise ValueError(f'volatile_tail must be >= 0, got {volatile_tail}') 

288 if volatile_tail == 0: 

289 return 0 

290 from evaluatorq.openresponses.input_items import messages_to_responses_input 

291 

292 return len(messages_to_responses_input(messages[-volatile_tail:]))
