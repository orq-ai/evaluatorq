1"""Provider content-filter detection on the OpenAI-compatible wire protocol. 

2 

3Domain-neutral: any caller making chat-completion calls through the Orq router (or 

4OpenAI directly) can use these to tell a *structural* content-filter block apart from 

5a model's natural-language self-refusal. Two shapes exist, both normalized to the 

6``content_filter`` code by the router: 

7 

81. 200 response with ``finish_reason='content_filter'`` (canonical OpenAI signal). 

92. HTTP error with structured ``code='content_filter'`` — verified empirically: Azure 

10 via the Orq router raises ``openai.BadRequestError`` (HTTP 400) this way. 

11 

12A model that self-censors in prose is deliberately NOT detected: at the protocol level 

13it is indistinguishable from a real response (``finish_reason='stop'``, no ``refusal`` 

14field — verified for Anthropic and Gemini via the Orq router), so it is left for the 

15caller to forward downstream rather than treated as a block. 

16""" 

17 

18from __future__ import annotations 

19 

20from typing import TYPE_CHECKING, Literal, TypeVar 

21 

22from loguru import logger 

23from openai import APIConnectionError, APIStatusError 

24 

25if TYPE_CHECKING: 

26 from collections.abc import Awaitable, Callable 

27 

28_ResponseT = TypeVar('_ResponseT') 

29 

30 

31def classify_finish_reason(finish_reason: str | None) -> Literal['content_filter'] | None: 

32 """Return ``'content_filter'`` when the provider blocked generation, else ``None``.""" 

33 return 'content_filter' if finish_reason == 'content_filter' else None 

34 

35 

36def is_content_filter_error(exc: BaseException) -> bool: 

37 """Return True if ``exc`` is a provider content-filter block surfaced as an error. 

38 

39 Keys on the structured error ``code`` (exposed by the OpenAI SDK as ``exc.code`` and 

40 mirrored in the JSON body), NOT on message prose, so it stays provider-agnostic for 

41 any block the router normalizes to that code. 

42 """ 

43 if getattr(exc, 'code', None) == 'content_filter': 

44 return True 

45 body = getattr(exc, 'body', None) 

46 return isinstance(body, dict) and body.get('code') == 'content_filter' 

47 

48 

49async def regenerate_on_content_filter( 

50 call: Callable[[int], Awaitable[_ResponseT]], 

51 *, 

52 max_attempts: int, 

53 label: str = 'attacker generation', 

54) -> _ResponseT: 

55 """Invoke ``call`` repeatedly, regenerating while the provider content-filters it. 

56 

57 ``call(attempt)`` makes one chat-completion request (and owns its own span / token 

58 accounting — the attempt index is passed so it can tag retries). A block is detected 

59 in either shape: a 200 response whose ``finish_reason='content_filter'``, or a raised 

60 error carrying ``code='content_filter'``. Each is retried up to ``max_attempts`` total. 

61 

62 Returns the first usable response. When every attempt is content-filtered, returns the 

63 final filtered *response* (200-form) so the caller can inspect it and stop cleanly — but 

64 a content-filter *error* on the final attempt (and any non-filter error) propagates, so 

65 the caller's own error handling classifies it. ``max_attempts`` is clamped to >= 1. 

66 """ 

67 attempts = max(1, max_attempts) 

68 for attempt in range(attempts): 

69 try: 

70 response = await call(attempt) 

71 except (APIConnectionError, APIStatusError) as e: 

72 if is_content_filter_error(e) and attempt < attempts - 1: 

73 logger.warning(f'{label} content-filtered (error form); regenerating ({attempt + 1}/{attempts - 1})') 

74 continue 

75 raise 

76 choices = getattr(response, 'choices', None) or [] 

77 finish_reason = getattr(choices[0], 'finish_reason', None) if choices else None 

78 if classify_finish_reason(finish_reason) is None or attempt >= attempts - 1: 

79 return response 

80 logger.warning(f'{label} content-filtered (finish_reason); regenerating ({attempt + 1}/{attempts - 1})') 

81 raise RuntimeError('regenerate_on_content_filter: loop exhausted without returning') # unreachable 

82 

83 

84__all__ = ['classify_finish_reason', 'is_content_filter_error', 'regenerate_on_content_filter']
