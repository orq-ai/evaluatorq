22# Accepted callable signatures — receive the conversation as a list of typed 

23# :class:`~evaluatorq.contracts.Message` objects. The list holds one message for 

24# the opening turn and grows with every turn, so the same callable handles 

25# single- and multi-turn red teaming. May return ``str`` or :class:`AgentResponse`. 

26AgentCallable = ( 

27 Callable[[list[Message]], Awaitable[AgentResponse]]

34# token accounting stays correct across multi-turn conversations.

38def _is_async_callable(fn: object) -> bool: 

39 """True if ``fn`` is a coroutine function or a callable object with an async ``__call__``.""" 

40 if asyncio.iscoroutinefunction(fn): 

41 return True 

42 call = getattr(fn, '__call__', None) # noqa: B004 — intentional: detect async callable objects 

43 return call is not None and asyncio.iscoroutinefunction(call)

46class CallableTarget(AgentTarget): 

47 """Wraps any sync or async function as an AgentTarget. 

48 

49 Use this as an escape hatch for frameworks that don't have a dedicated 

50 integration. You provide a function that takes the conversation (a list of 

51 typed :class:`~evaluatorq.contracts.Message` objects) and returns a response 

52 — the wrapper handles the rest. The list contains one message on the opening 

53 turn and every prior turn on later turns, so a stateless callable still sees 

54 full context, matching the stateless OpenAI / Vercel / OpenAI-Agents targets 

55 (which likewise consume the typed ``Message`` list at the boundary). 

56 

57 Usage:: 

58 

59 from evaluatorq.contracts import Message 

60 from evaluatorq.integrations.callable_integration import CallableTarget 

61 

62 # Async function — receives the whole conversation as Message objects 

63 async def my_agent(messages: list[Message]) -> str: 

64 result = await some_framework.run(messages[-1].content) 

65 return result.text 

66 

67 target = CallableTarget(my_agent) 

68 

69 # Need OpenAI chat-completion dicts? Convert at the boundary yourself: 

70 async def openai_agent(messages: list[Message]) -> str: 

71 chat = [m.to_chat_completion() for m in messages] 

72 return (await client.chat.completions.create(model="gpt-4o", messages=chat)).choices[0].message.content 

73 

74 target = CallableTarget(openai_agent) 

75 

76 # Plumb token counts via usage_fn — it sees the full transcript 

77 def get_usage(messages: list[Message], response: str) -> TokenUsage: 

78 return TokenUsage(prompt_tokens=10, completion_tokens=5, total_tokens=15, calls=1) 

79 

80 target = CallableTarget(my_agent, usage_fn=get_usage) 

81 

82 # Pass to simulation or red teaming 

83 config = DynamicRunConfig(targets=[target]) 

84 """

86 def __init__( 

87 self, 

88 fn: AgentCallable, 

89 *, 

90 reset_fn: Callable[[], None] | None = None, 

91 usage_fn: UsageFn | None = None, 

92 agent_context: AgentContext | None = None, 

93 ) -> None: 

94 """Create a callable agent target. 

95 

96 Args: 

97 fn: A sync or async function taking the conversation as a 

98 ``list[Message]`` and returning a ``str`` or an 

99 :class:`AgentResponse`. The list grows by one turn each round, so 

100 the same callable serves single- and multi-turn runs. Callables 

101 that want OpenAI chat-completion dicts can call 

102 :meth:`Message.to_chat_completion` on each element themselves. 

103 reset_fn: Optional callback invoked on ``new()`` to clear shared callable state between attacks. 

104 usage_fn: Optional callable taking ``(messages, response) -> TokenUsage | None``, 

105 where ``messages`` is the full transcript forwarded to ``fn`` and 

106 ``response`` is the response text. Use this to plumb token counts 

107 from your underlying framework when the callable itself only 

108 returns a string. The function must be synchronous; async usage 

109 extraction is not supported. Exceptions raised by ``usage_fn`` are 

110 logged as warnings and result in ``usage=None``. 

111 agent_context: Optional :class:`AgentContext` describing the wrapped 

112 callable's tools, memory, system prompt, etc. The red teaming 

113 pipeline uses this for capability-aware strategy filtering — 

114 without it, all strategies (including nonsensical ones) will be 

115 applied. If not provided, a minimal context is returned. 

116 Callables are opaque — the wrapper cannot manage memory isolation. 

117 If the wrapped callable holds state, use ``reset_fn`` to clear it. 

118 """ 

119 super().__init__(memory_entity_id=None) 

120 self._fn = fn 

121 self._is_async = _is_async_callable(fn) 

122 self._reset_fn = reset_fn 

123 self._usage_fn = usage_fn 

124 self._agent_context = agent_context 

125 

126 async def respond(self, messages: list[Message]) -> AgentResponse: 

127 """Send the full conversation to the wrapped callable; return a structured response. 

128 

129 Forwards the entire transcript as typed :class:`Message` objects — tool 

130 turns included — so a stateless callable sees prior context. The list 

131 holds a single message on the opening turn and grows each round. Like the 

132 stateless OpenAI / Vercel targets, no constraint is placed on the last 

133 turn's role. Callables that return a plain ``str`` are wrapped in an 

134 :class:`AgentResponse`; those returning :class:`AgentResponse` pass 

135 through. Token usage from ``usage_fn`` (if provided) is attached to the 

136 returned ``AgentResponse``. 

137 """ 

138 fn: Callable[[list[Message]], Any] = self._fn 

139 try: 

140 if self._is_async: 

141 result = await fn(messages) 

142 else: 

143 result = await asyncio.to_thread(fn, messages) 

144 if inspect.isawaitable(result): 

145 result = await result 

146 except (asyncio.CancelledError, asyncio.TimeoutError): 

147 raise 

148 except Exception as exc: 

149 raise RuntimeError(f'CallableTarget: callable raised {exc!r}') from exc 

150 

151 if isinstance(result, AgentResponse): 

152 return result 

153 text = str(result) if result is not None else '' 

154 usage: TokenUsage | None = None 

155 if self._usage_fn is not None: 

156 try: 

157 usage = self._usage_fn(messages, text) 

158 except Exception: 

159 logger.exception('usage_fn raised an exception; using usage=None') 

160 text_item: OutputMessage = TextOutputItem(text=text, annotations=[]) 

161 return AgentResponse(output=[text_item], usage=usage) 

162 

163 async def get_agent_context(self) -> AgentContext: 

164 """Return the user-provided agent context, or a minimal placeholder.""" 

165 if self._agent_context is not None: 

166 return self._agent_context 

167 name = getattr(self._fn, '__name__', '') 

168 # Lambdas share the useless name "<lambda>"; fall back to a stable key. 

169 key = name if name and name != '<lambda>' else 'callable_target' 

170 return AgentContext(key=key, description='opaque callable target') 

171 

172 def new(self) -> CallableTarget: 

173 """Return a fresh copy sharing the same callable, with state reset via reset_fn.""" 

174 if self._reset_fn is not None: 

175 self._reset_fn() 

176 return CallableTarget( 

177 self._fn, 

178 reset_fn=self._reset_fn, 

179 usage_fn=self._usage_fn, 

180 agent_context=self._agent_context, 

181 )
