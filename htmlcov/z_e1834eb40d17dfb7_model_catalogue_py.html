1"""Orq's model catalogue: per-model prices, provider ids, and endpoint support. 

2 

3Fetched once per process (per host) from ``GET /v2/models`` and used for three things: 

4 

5* **Pricing calls the router does not price.** ``/v3/router/responses`` returns 

6 ``usage.input_cost``/``output_cost``/``total_cost``; ``/v3/router/chat/completions`` 

7 returns token counts only. Judges now default to Responses, but any caller 

8 still on Chat Completions gets its cost filled in here rather than recording a 

9 billed-to-nobody call (RES-1295). 

10* **Qualifying a bare model id.** The router's Responses endpoint requires 

11 ``provider/model`` (``openai/gpt-5-mini``); Chat Completions accepts the bare 

12 ``gpt-5-mini`` that configs are written with. The catalogue maps one to the other. 

13* **Knowing which models serve Responses at all.** Entries carry 

14 ``metadata.supports_responses_api``, so a caller can ask before it calls 

15 instead of discovering it from a 400. 

16 

17A model absent from the catalogue stays unpriced — ``total_cost`` ``None``, 

18``priced_calls`` 0 — which the dashboard already renders as "no cost data" 

19rather than "$0.00", and unqualified, which sends the caller back to Chat 

20Completions rather than guessing a provider. 

21 

22Payload shape confirmed against live ``my.orq.ai/v2/models`` (2026-08-13): a bare 

23JSON list of 109 entries, no pagination envelope. Prices are **per 1k tokens** — 

24``gpt-5-mini`` returns ``input_cost: 0.00025`` / ``output_cost: 0.002``, matching 

25OpenAI's published $0.25 / $2.00 per 1M. ``input_currency``/``output_currency`` 

26are ``''`` on most entries and ``'usd'`` on the rest; a non-empty non-USD entry 

27is skipped rather than silently mixed into a USD total. 

28"""

59class ModelInfo(_ModelInfoFields): 

60 """One catalogue entry. 

61 

62 Costs are USD **per 1000 tokens** — the unit ``/v2/models`` publishes, and the 

63 reason `price_usage` divides by 1000. The names carry the unit so the 

64 arithmetic stays self-checking: ``Usage.input_cost`` is absolute USD for a 

65 call, 1000x away from this field, and the two meet in one expression. 

66 

67 Split over a ``_ModelInfoFields`` base purely so ``__new__`` can normalize: 

68 ``typing.NamedTuple`` refuses to let a class body override it. 

69 """ 

70 

71 __slots__ = ()

73 def __new__( # noqa: PYI034 — a NamedTuple subclass really does construct the subclass 

74 cls, 

75 input_cost_per_1k: float, 

76 output_cost_per_1k: float, 

77 provider: str, 

78 supports_responses: bool, # noqa: FBT001 — positional to match the tuple field order 

79 reasoning_efforts: frozenset[str] | None = None,

81 """Normalize an empty ``reasoning_efforts`` to ``None``. 

82 

83 The two are not the same thing and the field's contract allows only one of 

84 them: ``None`` is "unknown, cannot validate"; an empty set would have to 

85 mean "this model accepts no effort at all". `validate_reasoning_effort` 

86 branches on ``is not None``, so an empty set makes it reject *every* value — 

87 including the defaults — with an empty accepted-values list, killing a run at 

88 pre-flight over a state that means nothing. `_parse_reasoning_efforts` already 

89 collapses the two; doing it in the type closes the same hole for 

90 `register_model`, which takes whatever a caller hands it. 

91 """ 

92 return super().__new__( 

93 cls, 

94 input_cost_per_1k, 

95 output_cost_per_1k, 

96 provider, 

97 supports_responses, 

98 reasoning_efforts or None, 

99 )

102# host -> (model_id -> ModelInfo). A host is absent until its first fetch, and 

103# maps to {} when the fetch failed or the account has no models. Keyed on host 

104# because a long-lived process (jobs runner, dashboard) can serve runs against 

105# prod and staging in turn, and prices attributed to the wrong deployment are 

106# worse than no prices. 

107_catalogues: dict[str, dict[str, ModelInfo]] = {} 

108# Built lazily: a module-level asyncio.Lock() binds to the loop running at import 

109# time, and the CLI and test suite both drive several asyncio.run() loops, where 

110# a lock bound to a dead loop raises "attached to a different loop". 

111_lock: asyncio.Lock | None = None 

112# Caller-registered entries, consulted before the fetched catalogue — the only way 

113# to price or correct a model /v2/models does not list. Keyed on the BARE id: 

114# `register_model` strips any `provider/` prefix so both spellings collapse to one. 

115_overrides: dict[str, ModelInfo] = {} 

116# Consecutive failed fetches per host: retry a few times, then give up for the 

117# process rather than hammering a host that is genuinely down. 

118_fetch_failures: dict[str, int] = {} 

119_MAX_FETCH_FAILURES = 3 

120# Env var rather than a parameter: `_load_catalogue` is called from pricing paths 

121# with no config object to thread. 

122_CATALOGUE_TIMEOUT_S = float(os.environ.get('EVALUATORQ_CATALOGUE_TIMEOUT_S', '30'))

144def register_model(model_id: str, info: ModelInfo) -> None: 

145 """Register or override one catalogue entry for the rest of the process. 

146 

147 Takes priority over the fetched catalogue, so this both adds a model Orq 

148 does not list and corrects one it lists wrongly. Without it an unknown model 

149 is silently unpriced (`price_usage`), silently forced off the Responses 

150 endpoint (`qualified_model`), and unvalidatable (`validate_reasoning_effort`) 

151 — three degradations a caller could previously neither see nor fix. 

152 

153 ``model_id`` is stored **unprefixed**: ``'openai/gpt-x'`` and ``'gpt-x'`` name 

154 the same model and register the same entry. Normalizing on write rather than 

155 on read is what makes the two spellings interchangeable — the internal lookups 

156 ask for the bare id (that is what `qualified_model` produces), so a qualified 

157 registration stored verbatim would never be found and the override would be a 

158 silent no-op. A second registration under the other spelling replaces the 

159 first, which is correct: there is one model. 

160 

161 ```python 

162 from evaluatorq.common.model_catalogue import ModelInfo, register_model 

163 

164 register_model( 

165 'my-self-hosted-llama', 

166 ModelInfo( 

167 input_cost_per_1k=0.0002, 

168 output_cost_per_1k=0.0008, 

169 provider='self', 

170 supports_responses=False, 

171 reasoning_efforts=None, 

172 ), 

173 ) 

174 ``` 

175 """ 

176 _overrides[model_id.split('/', 1)[-1]] = info

181 _overrides.clear()

204def _parse_reasoning_efforts(entry: dict[str, object]) -> frozenset[str] | None: 

205 """Accepted ``reasoning.effort`` values for one catalogue entry, or ``None``. 

206 

207 Read from ``parameters[parameter == 'reasoningEffort'].config.options[].value`` 

208 rather than the ``metadata.supports_reasoning_effort_*`` booleans: the two 

209 disagree on 50 of the 148 live entries (``kimi-k3`` advertises low/high in the 

210 flags and low/high/max in the options), and only the options list grows a new 

211 level without a schema change. ``None`` when the entry has no such parameter, 

212 which the caller must treat as "cannot validate", not "nothing allowed". 

213 """ 

214 parameters = entry.get('parameters') 

215 if not isinstance(parameters, list): 

216 return None 

217 for parameter in parameters: 

218 if not isinstance(parameter, dict) or parameter.get('parameter') != 'reasoningEffort': 

219 continue 

220 config = parameter.get('config') 

221 options = config.get('options') if isinstance(config, dict) else None 

222 if not isinstance(options, list): 

223 return None 

224 values = {o['value'] for o in options if isinstance(o, dict) and isinstance(o.get('value'), str)} 

225 return frozenset(values) or None 

226 return None

229def _parse_catalogue(payload: object) -> dict[str, ModelInfo]: 

230 """Build the model table from a ``/v2/models`` payload. 

231 

232 Keyed on the bare ``model_id`` (``gpt-5-mini``), which is how callers name 

233 models; a provider-prefixed id is normalised at lookup time. The same 

234 model_id appears once per hosting provider (openai, azure, …) at the same 

235 price, so the entry whose provider matches ``model_developer`` wins — that is 

236 the one the router resolves ``openai/gpt-5-mini`` to — and otherwise the 

237 first seen is kept. Verified against the live catalogue: of 9 duplicated 

238 model_ids, none disagreed on price. A future disagreement is logged rather 

239 than silently resolved. 

240 """ 

241 if not isinstance(payload, list): 

242 logger.warning( 

243 'Model catalogue had unexpected shape {} (expected a JSON list); costs and Responses routing unavailable', 

244 type(payload).__name__, 

245 ) 

246 return {} 

247 models: dict[str, ModelInfo] = {} 

248 for entry in payload: 

249 if not isinstance(entry, dict): 

250 continue 

251 model_id, provider = entry.get('model_id'), entry.get('provider') 

252 inp, out = entry.get('input_cost'), entry.get('output_cost') 

253 if not isinstance(model_id, str) or not isinstance(provider, str): 

254 continue 

255 # bool is an int subclass: `input_cost: true` would otherwise price at $1.00/1k. 

256 if isinstance(inp, bool) or isinstance(out, bool): 

257 continue 

258 if not isinstance(inp, (int, float)) or not isinstance(out, (int, float)): 

259 continue 

260 if inp < 0 or out < 0: 

261 continue 

262 # Currency is '' on most entries and 'usd' on the rest; only a stated 

263 # non-USD price is skipped, so the common empty case still prices. 

264 currencies = {str(entry.get('input_currency') or ''), str(entry.get('output_currency') or '')} 

265 if currencies - {'', 'usd'}: 

266 continue 

267 info = ModelInfo( 

268 input_cost_per_1k=float(inp), 

269 output_cost_per_1k=float(out), 

270 provider=provider, 

271 # `metadata` is provider-shaped and has arrived as a list; `.get` on a 

272 # non-mapping took the whole catalogue down rather than one model. 

273 supports_responses=bool(_entry_metadata(entry).get('supports_responses_api')), 

274 reasoning_efforts=_parse_reasoning_efforts(entry), 

275 ) 

276 existing = models.get(model_id) 

277 if existing is not None: 

278 if (existing.input_cost_per_1k, existing.output_cost_per_1k) != ( 

279 info.input_cost_per_1k, 

280 info.output_cost_per_1k, 

281 ): 

282 logger.debug( 

283 'Catalogue prices disagree for {} ({} vs {}); keeping the model_developer entry if there is one', 

284 model_id, 

285 existing.provider, 

286 provider, 

287 ) 

288 if provider != entry.get('model_developer'): 

289 continue 

290 models[model_id] = info 

291 if isinstance(payload, list) and payload and not models: 

292 logger.warning( 

293 'Model catalogue returned {} entries but none parsed (payload shape change?); ' 

294 'judges will run unpriced on chat completions', 

295 len(payload), 

296 ) 

297 return models

300async def _load_catalogue(client: AsyncOpenAI | None = None) -> dict[str, ModelInfo]: 

301 """Fetch the Orq model catalogue once per process, per host. 

302 

303 The host is taken from ``client`` when it routes through the Orq router, so an 

304 injected staging/on-prem client is not priced against prod; otherwise 

305 ``ORQ_BASE_URL``. 

306 

307 Failures are cached as an empty table: a run makes dozens of judge calls and 

308 must not re-pay a failing HTTP round-trip on each one. Cost simply stays 

309 unknown for the run, which is the honest reading — but note this also 

310 disables Responses routing for the run, so it is logged at WARNING with that 

311 consequence spelled out. 

312 """ 

313 host = resolve_results_base_url(client) if client is not None else orq_base_url() 

314 cached = _catalogues.get(host) 

315 if cached is not None: 

316 return cached 

317 async with _catalogue_lock(): 

318 cached = _catalogues.get(host) 

319 if cached is not None: 

320 return cached 

321 # An injected client's own key matches its host; an ambient ORQ_API_KEY for a 

322 # different workspace would 401 (caching {}) or price against the wrong one. 

323 api_key = ( 

324 (getattr(client, 'api_key', None) or os.environ.get('ORQ_API_KEY')) 

325 if client is not None 

326 else os.environ.get('ORQ_API_KEY') 

327 ) 

328 if not api_key: 

329 logger.debug('No ORQ_API_KEY and no client credential; model catalogue unavailable') 

330 _catalogues[host] = {} 

331 return _catalogues[host] 

332 payload: object = None 

333 try: 

334 async with httpx.AsyncClient(timeout=_CATALOGUE_TIMEOUT_S) as http_client: 

335 response = await http_client.get( 

336 f'{host}/v2/models', 

337 headers={'Authorization': f'Bearer {api_key}'}, 

338 ) 

339 response.raise_for_status() 

340 payload = response.json() 

341 except (httpx.HTTPError, ValueError) as exc: 

342 # Narrow on purpose: a bug inside _parse_catalogue must not be cached 

343 # away as "the network was down", so parsing happens outside the try. 

344 failures = _fetch_failures[host] = _fetch_failures.get(host, 0) + 1 

345 give_up = failures >= _MAX_FETCH_FAILURES 

346 logger.warning( 

347 'Orq model catalogue unavailable ({}: {}) at {}/v2/models (attempt {} of {}) — ' 

348 'judges will run on chat completions and report no cost {}', 

349 type(exc).__name__, 

350 exc, 

351 host, 

352 failures, 

353 _MAX_FETCH_FAILURES, 

354 'for the rest of this process' if give_up else 'until a later call succeeds', 

355 ) 

356 if give_up: 

357 _catalogues[host] = {} 

358 return _catalogues[host] 

359 # Not cached: a transient hiccup must not degrade the whole run. 

360 return {} 

361 _fetch_failures.pop(host, None) 

362 _catalogues[host] = _parse_catalogue(payload) 

363 logger.debug('Loaded catalogue for {} models from {}', len(_catalogues[host]), host) 

364 return _catalogues[host]

367async def _lookup(model: str, client: AsyncOpenAI | None = None) -> ModelInfo | None: 

368 """Catalogue entry for ``model``, trying the id as given then unprefixed. 

369 

370 Caller `register_model` overrides win over the fetched catalogue. 

371 """ 

372 bare = model.split('/', 1)[-1] 

373 # One probe, not two: `register_model` normalizes to the bare id on write, so 

374 # the override key space has a single canonical spelling. 

375 override = _overrides.get(bare) 

376 if override is not None: 

377 return override 

378 catalogue = await _load_catalogue(client) 

379 return catalogue.get(model) or catalogue.get(bare)

382async def qualified_model(model: str, client: AsyncOpenAI | None = None) -> str | None: 

383 """The ``provider/model`` id to send to the router's Responses endpoint. 

384 

385 A bare id is qualified from the catalogue and yields ``None`` when the model 

386 is absent or its entry says it does not serve Responses — the caller must 

387 then stay on Chat Completions. An already-qualified id (one containing 

388 ``/``) passes through unchecked, since the caller has named a provider 

389 explicitly and the catalogue has nothing to add. 

390 """ 

391 if '/' in model: 

392 return model 

393 info = await _lookup(model, client) 

394 if info is None or not info.supports_responses: 

395 return None 

396 return f'{info.provider}/{model}'

399async def validate_reasoning_effort(effort: str, model: str, client: AsyncOpenAI | None = None) -> None: 

400 """Fail a run up front when ``model`` cannot serve ``effort``. 

401 

402 Raises ``ValueError`` only when the catalogue actually lists the accepted 

403 values for ``model`` and ``effort`` is not among them. Every other case — 

404 model absent from the catalogue (an ``agent/<key>`` id always is), catalogue 

405 fetch failed, entry carries no ``reasoningEffort`` parameter — logs and 

406 returns, leaving the provider as the authority. The point is to fail before 

407 strategy generation and attack generation are paid for, not to become a 

408 second source of truth about what a model accepts. 

409 """ 

410 info = await _lookup(model, client) 

411 if info is None or info.reasoning_efforts is None: 

412 logger.warning( 

413 'Cannot pre-validate reasoning effort {!r}: model {} is {} — the provider will ' 

414 'reject an unsupported value at call time instead.', 

415 effort, 

416 model, 

417 'absent from the Orq catalogue' if info is None else 'listed without a reasoningEffort parameter', 

418 ) 

419 return 

420 if effort not in info.reasoning_efforts: 

421 raise ValueError( 

422 f'Reasoning effort {effort!r} is not accepted by {model}. ' 

423 f'Accepted values: {", ".join(sorted(info.reasoning_efforts))}.' 

424 )

427async def price_usage(usage: Usage | None, model: str, client: AsyncOpenAI | None = None) -> Usage | None: 

428 """Fill in cost on a **single call's** ``usage`` when the provider reported none. 

429 

430 Returns ``usage`` unchanged when it is ``None``, already priced, spans more 

431 than one call, or the model is absent from the catalogue. The multi-call 

432 guard matters: pricing an aggregate at one model's rate would also set 

433 ``priced_calls == calls``, asserting full pricing for calls this function 

434 never saw and defeating `Usage.cost_is_partial`. 

435 """ 

436 if usage is None or usage.total_cost is not None or usage.calls > 1: 

437 return usage 

438 info = await _lookup(model, client) 

439 if info is None: 

440 logger.debug('Model {} is not in the Orq catalogue; call stays unpriced', model) 

441 return usage 

442 # ponytail: flat input rate — cached-read and cache-write tokens are billed at a 

443 # discount/premium the catalogue does not expose, so a cache-heavy call reads 

444 # slightly high. Split the rate here if /v2/models ever publishes the tiers. 

445 input_cost = usage.input_tokens / 1000 * info.input_cost_per_1k 

446 output_cost = usage.output_tokens / 1000 * info.output_cost_per_1k 

447 return usage.model_copy( 

448 update={ 

449 'input_cost': input_cost, 

450 'output_cost': output_cost, 

451 'total_cost': input_cost + output_cost, 

452 'priced_calls': usage.calls, 

453 } 

454 )
