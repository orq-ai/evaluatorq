1"""LLM-as-a-jury preset definitions (RES-1171, derived in RES-996/RES-1346). 

2 

3The single source for what a named jury preset means: which judges, which 

4aggregation mode, and which reserve judges replace them when one is retired, 

5deprecated or repriced. `llm_jury(preset=...)` builds a panel from here, and a 

6test recomputes every published figure from `common/data/jury_judge_rates.json`, 

7so the table in the docs cannot drift away from what the code seats. 

8 

9Seats are derived rather than hand-listed. The derivation itself is not here: 

10it reads a capture of the whole model garden, ranks each lineage on an 

11intelligence index and a blended price, and is rerun and reviewed in the 

12research repo that owns the capture. What ships to callers is its output, the 

13panels and the rates they were costed at. Seats are compared in-family, because 

14the lineage diversity is the point: judged against the whole garden most seats 

15look dominated, and acting on that would collapse every panel onto whichever 

16vendor is cheapest this month and recreate the correlated errors a panel exists 

17to cancel. 

18 

19A preset that is not in `PRESETS` still owes an explanation, so `WITHHELD_PRESETS` 

20and `DROPPED_PRESETS` carry one and `get_preset` hands it back by name instead of 

21a bare "unknown preset". The registers behind the seating itself, which record a 

22successor deliberately passed over or a seat knowingly held past its age limit, 

23live with the derivation in the research repo where they can be checked against 

24the garden. 

25 

26Reserves are a reviewed change to this file, not a grade-time substitution. A 

27panel that contains the generator warns and proceeds (`_panel_composition_messages` 

28in `common.jury`, or `strict_panel=True` to make it fatal); it never silently 

29swaps a judge, because a user who picks a preset should get the panel they picked. 

30 

31Router IDs are the literal strings the orq model garden returns from 

32`GET /v2/models`; they are what the router and `common.model_catalogue` expect. 

33"""

61@lru_cache(maxsize=1) 

62def _load_rates() -> MappingProxyType[str, JudgeRates]: 

63 """The committed rate table, read once and handed out read-only. 

64 

65 Frozen on purpose. This is cached process-wide, so a caller who mutated the 

66 returned mapping would silently reprice every published preset for the life 

67 of the process. 

68 """ 

69 raw = cast('dict[str, Any]', json.loads(_RATES_PATH.read_text())) 

70 return MappingProxyType({rid: JudgeRates(**fields) for rid, fields in raw['judges'].items()})

83# Host, region and serving-variant noise that makes one model look like several. 

84# `openai/gpt-5.6-terra`, `azure/global.gpt-5.6-terra` and `openai/eu.gpt-5.6-terra` 

85# are one model; a successor check that counts them separately reports work that 

86# does not exist. 

87_REGION_PREFIXES: tuple[str, ...] = ('eu.', 'us.', 'apac.', 'global.') 

88# Vendor tokens `model_identity` strips when one leads a dotted model name, which 

89# is how Bedrock spells an id. A vendor whose name can open a legitimate model 

90# name (the way 'gpt-5.6-terra' opens with a version, not a vendor) would make 

91# `model_identity` truncate it, so this list is not the family list. 

92_VENDOR_TOKENS: frozenset[str] = frozenset({ 

93 'openai', 

94 'anthropic', 

95 'google', 

96 'zhipu', 

97 'moonshot', 

98 'minimax', 

99 'deepseek', 

100 'alibaba', 

101 'meta', 

102 'mistral', 

103 'xai', 

104})

107def model_identity(router_id: str) -> str: 

108 """The model behind a router ID, with host, region, vendor and version stripped. 

109 

110 Two IDs sharing an identity are the same weights served twice, so seating one 

111 means the other needs no separate review. Bedrock is the awkward case: 

112 `aws/eu.anthropic.claude-haiku-4-5-20251001-v1:0` and 

113 `anthropic/claude-haiku-4-5-20251001` are one model, and EU Region seats the 

114 first while Balanced Trio used to seat the second. 

115 """ 

116 name = router_id.split('/')[-1].lower() 

117 for prefix in _REGION_PREFIXES: 

118 name = name.removeprefix(prefix) 

119 head, _, tail = name.partition('.') 

120 if tail and head in _VENDOR_TOKENS: 

121 name = tail 

122 return re.sub(r'-v\d+:\d+$', '', name)

125def judge_family(router_id: str) -> str: 

126 """Training lineage of a router ID, the axis self-preference runs on. 

127 

128 Delegates to `common.jury.provider_family`, which is what the runtime panel 

129 check uses: two lineage implementations would let a preset be composed on one 

130 reading and warned about on another. Raises rather than returning 'unknown', 

131 because an unclassified judge would silently never match a generator and the 

132 composition check would pass by accident. 

133 

134 Lineage is who trained the model, not who serves it: `groq/openai/gpt-oss-120b` 

135 is OpenAI despite being open-weight on Groq, and `baseten/kimi-k3` is Moonshot. 

136 """ 

137 family = provider_family(router_id) 

138 if family == 'unknown': 

139 raise ValueError(f'unknown lineage for {router_id!r}; add a marker to common.jury._FAMILY_MARKERS') 

140 return family

143class JuryPreset(BaseModel): 

144 """A named panel: fixed judges, fixed aggregation, and reserves for retirement.""" 

145 

146 name: str 

147 judges: tuple[str, ...] 

148 reserve_judges: tuple[str, ...] 

149 numeric_aggregation: AggregatorName = 'mean_std' 

150 use_when: str 

151 estimated_cost_per_1k: float = Field( 

152 description='USD per 1,000 pointwise items at 1,500 input / 150 output tokens, uncached.' 

153 ) 

154 

155 @model_validator(mode='after') 

156 def _panel_is_well_formed(self) -> 'JuryPreset': 

157 """Even panels split, and a repeated judge is one voter casting two votes.""" 

158 if len(self.judges) % 2 == 0: 

159 raise ValueError(f'{self.name}: panel size {len(self.judges)} is even') 

160 if len(set(self.judges)) != len(self.judges): 

161 raise ValueError(f'{self.name}: panel has a duplicate judge') 

162 if set(self.reserve_judges) & set(self.judges): 

163 raise ValueError(f'{self.name}: a reserve judge is already on the panel') 

164 if len(set(judge_family(r) for r in self.reserve_judges)) != len(self.reserve_judges): 

165 raise ValueError(f'{self.name}: reserves share a lineage with each other') 

166 return self

168 def seated_efforts(self) -> dict[str, str | None]: 

169 """The reasoning effort each judge is ranked at, and so must be called at. 

170 

171 A preset names judges and a caller sends the call, so until the seated 

172 effort is stated the two can disagree silently: the panel is costed and 

173 ranked at one operating point and run at another. None means the card 

174 publishes no ladder and the provider default stands. 

175 

176 `llm_jury` takes one `reasoning_effort` for the whole panel, so a preset 

177 whose seats disagree cannot express itself through it yet; per-judge call 

178 settings are a schema change and its own ticket (RES-1347). 

179 

180 These are the rungs the cards were scored at, not values to send: a card 

181 that only distinguishes thinking from not thinking is scored at 

182 `reasoning` or `none`, which the EU Region haiku seat reads back as 

183 `reasoning` and no provider accepts as a `reasoning_effort`. Sending one 

184 is a 400 on the value, which the retry path used to read as the model 

185 refusing the parameter. 

186 """ 

187 return {judge: (r.seated_effort if (r := judge_rates(judge)) else None) for judge in self.judges} 

188 

189 def priced_below_seated_effort(self) -> tuple[str, ...]: 

190 """Judges whose captured price is the blend at a cheaper rung than the seated one. 

191 

192 The published $/1k understates these until a probe measures them at the 

193 effort they are seated at. Disclosed rather than corrected: correcting it 

194 is a re-probe of every panel, not an arithmetic fix. 

195 """ 

196 return tuple(j for j in self.judges if (r := judge_rates(j)) and not r.priced_at_ceiling) 

197 

198 def duplicated_lineages(self) -> dict[str, tuple[str, ...]]: 

199 """Lineages seated more than once, whose errors correlate. 

200 

201 Not an error. A panel may repeat a lineage deliberately, as 

202 Single-Provider Trio does with three OpenAI judges, but the diversity 

203 claim weakens and the reviewer should see it rather than count IDs by 

204 hand. 

205 """ 

206 seated: dict[str, list[str]] = {} 

207 for judge in self.judges: 

208 seated.setdefault(judge_family(judge), []).append(judge) 

209 return {family: tuple(js) for family, js in seated.items() if len(js) > 1} 

210 

211 def cost_per_1k(self) -> float: 

212 """The panel's $/1k recomputed from the captured billing rates. 

213 

214 `estimated_cost_per_1k` is the figure the docs publish; this is where it 

215 has to come from. A test asserts the two agree, so a repricing in the 

216 snapshot fails CI rather than quietly making a published table wrong. 

217 """ 

218 total = Decimal(0) 

219 for judge in self.judges: 

220 rates = judge_rates(judge) 

221 if rates is None: 

222 raise KeyError(f'{self.name}: no captured pricing for {judge!r}; re-run the rate capture') 

223 total += Decimal(str(rates.input_rate)) * Decimal(ESTIMATED_PROMPT_TOKENS) 

224 total += Decimal(str(rates.output_rate)) * Decimal(ESTIMATED_COMPLETION_TOKENS) 

225 # Decimal, and one rounding at the end. Two published figures land exactly 

226 # on a half-cent (Strong Jury 23.625, EU Region 6.555), so binary floats 

227 # and banker's rounding would decide them by representation rather than by 

228 # a price, and a recapture that moved nothing could flip the table. 

229 return float((total / Decimal(1_000)).quantize(Decimal('0.01'), rounding=ROUND_HALF_UP))

424def get_preset(name: str) -> JuryPreset: 

425 """Look a preset up by name, listing the alternatives when there is no match.""" 

426 name = name.strip() 

427 try: 

428 return PRESETS[name] 

429 except KeyError: 

430 dropped = DROPPED_PRESETS.get(name) 

431 if dropped: 

432 raise ValueError(f'jury preset {name!r} was retired: {dropped}') from None 

433 withheld = WITHHELD_PRESETS.get(name) 

434 if withheld: 

435 raise ValueError(f'jury preset {name!r} is not shipped yet: {withheld}') from None 

436 raise ValueError(f'unknown jury preset {name!r}; available: {sorted(PRESETS)}') from None
