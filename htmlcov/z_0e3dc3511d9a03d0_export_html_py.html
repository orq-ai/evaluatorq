41# Two brand hues far enough apart that the side bars and the vote chips read as

57def _side_bar(label: str, rate: float | None, color: str) -> str: 

58 """One labelled, full-width bar for a side of the comparison.""" 

59 width = 0.0 if rate is None else max(0.0, min(1.0, rate)) * 100 

60 return ( 

61 '<div class="pw-side">' 

62 f'<div class="pw-side__label">{_esc(label)}</div>' 

63 '<div class="pw-side__track">' 

64 f'<div class="pw-side__fill" style="width:{width:.1f}%;background:{color}"></div>' 

65 '</div>' 

66 f'<div class="pw-side__value">{_rate(rate)}</div>' 

67 '</div>'

71def _render_consensus_html(section: ReportSection) -> str: 

72 d = section.data 

73 total = d.get('comparisons', 0) 

74 decided = d.get('decided', 0) 

75 bars = _side_bar(d['label_a'], d.get('a_win_rate'), _A_COLOR) + _side_bar( 

76 d['label_b'], d.get('b_win_rate'), _B_COLOR 

77 ) 

78 # Stated out loud because both win rates share a denominator that excludes 

79 # ties and inconclusive comparisons. 

80 caption = f'<p class="pw-caption">Win rates over the {decided} of {total} comparisons the panel decided.</p>' 

81 stats = _kpi_cards([ 

82 {'label': 'Ties', 'value': _rate(d.get('tie_rate')), 'status': 'neutral'}, 

83 { 

84 'label': 'Inconclusive', 

85 'value': _rate(d.get('inconclusive_rate')), 

86 'status': 'warn' if (d.get('inconclusive_rate') or 0) > INCONCLUSIVE_WARN else 'neutral', 

87 }, 

88 {'label': 'Mean agreement', 'value': _num(d.get('mean_agreement')), 'status': 'neutral'}, 

89 ]) 

90 weighted = d.get('bt_sigma') 

91 weighted_html = '' 

92 if weighted is not None: 

93 weighted_bars = _side_bar(d['label_a'], weighted.get('a_win_rate'), _A_COLOR) + _side_bar( 

94 d['label_b'], weighted.get('b_win_rate'), _B_COLOR 

95 ) 

96 warnings = ''.join(f'<li>{_esc(warning)}</li>' for warning in weighted.get('fit_warnings', [])) 

97 warning_html = f'<ul class="pw-caption">{warnings}</ul>' if warnings else '' 

98 weighted_html = ( 

99 '<h3>Reliability-weighted consensus (BT-sigma)</h3>' 

100 f'<div class="pw-sides">{weighted_bars}</div>{warning_html}' 

101 ) 

102 return ( 

103 f'<section id="{section.kind}"><h2>{_esc(section.title)}</h2>' 

104 f'<div class="pw-sides">{bars}</div>{caption}{stats}{weighted_html}</section>' 

105 )

108def _bias_cell(row: dict[str, Any], *, observed_swap: bool) -> str: 

109 bias = row.get('position_bias') 

110 if bias is None: 

111 # Blaming the wrong party is the failure to avoid here. When no judge in 

112 # the run completed a pair, that is a run-level fact and saying "this 

113 # judge" would pin a run's shape on each judge in turn. 

114 reason = ( 

115 'This judge never completed both orderings of a pair, so no flip could be observed' 

116 if observed_swap 

117 else 'No pair completed both orderings, so no flip was observable in this run' 

118 ) 

119 return f'<span class="pw-muted" title="{_esc(reason)}">n/a</span>' 

120 width = max(0.0, min(1.0, bias)) * 100 

121 warn = ' pw-bias__fill--warn' if row.get('biased') else '' 

122 marker = ' ⚠' if row.get('biased') else '' 

123 return ( 

124 '<span class="pw-bias">' 

125 f'<span class="pw-bias__track"><span class="pw-bias__fill{warn}" style="width:{width:.1f}%"></span></span>' 

126 f'{bias:.2f}{marker}</span>' 

127 )

130def _render_judges_html(section: ReportSection) -> str: 

131 d = section.data 

132 observed = bool(d.get('observed_swap')) 

133 headers = ['Judge', f'{d["label_a"]} rate', f'{d["label_b"]} rate', 'Tie rate', 'Position bias'] 

134 if d.get('bt_sigma'): 

135 headers.append('Sigma') 

136 rows: list[list[str]] = [] 

137 for r in d.get('rows', []): 

138 row = [ 

139 _esc(r['model']) + (' <span class="pw-tag">stand-in</span>' if r.get('replacement') else ''), 

140 _rate(r.get('a_rate')), 

141 _rate(r.get('b_rate')), 

142 _rate(r.get('tie_rate')), 

143 _bias_cell(r, observed_swap=observed), 

144 ] 

145 if d.get('bt_sigma'): 

146 row.append(_num(r.get('sigma'))) 

147 rows.append(row) 

148 if not rows: 

149 return '' 

150 # The caption keys off what ran, not what was configured: a run saved with 

151 # the default swap=True but executed single-ordering would otherwise show a 

152 # clean table with nothing saying the column is empty for a reason. 

153 if observed: 

154 note = '' 

155 elif d.get('swap'): 

156 note = ( 

157 '<p class="pw-caption">No pair completed both orderings, so position bias could not be ' 

158 'measured for this run despite swapping being enabled.</p>' 

159 ) 

160 else: 

161 note = '<p class="pw-caption">This run used a single ordering, so position bias was not measured.</p>' 

162 return f'<section id="{section.kind}"><h2>{_esc(section.title)}</h2>{_html_table(headers, rows)}{note}</section>'

168def _vote_chip(vote: dict[str, Any]) -> str: 

169 value = vote.get('vote') 

170 if not vote.get('present') or value is None: 

171 # An abstention is a recorded outcome, not missing data. 

172 return '<span class="pw-chip pw-chip--abstain" title="abstained">&middot;</span>' 

173 tone = 'a' if value == 'A' else ('b' if value == 'B' else 'tie') 

174 flip = ' pw-chip--flip' if vote.get('flipped') else '' 

175 return f'<span class="pw-chip pw-chip--{tone}{flip}" title="{_esc(vote["model"])}">{_esc(vote["label"])}</span>'

178def _detail_html(row: dict[str, Any], labels: tuple[str, str]) -> str: 

179 label_a, label_b = labels 

180 responses = ( 

181 '<div class="pw-responses">' 

182 f'<div class="pw-response"><h4>{_esc(label_a)}</h4><pre>{_esc(row["response_a"])}</pre></div>' 

183 f'<div class="pw-response"><h4>{_esc(label_b)}</h4><pre>{_esc(row["response_b"])}</pre></div>' 

184 '</div>' 

185 ) 

186 votes = ''.join(_vote_row(v) for v in row.get('votes', [])) 

187 return f'{responses}<ul class="pw-votes">{votes}</ul>'

204def _render_comparisons_html(section: ReportSection) -> str: 

205 d = section.data 

206 labels = (d['label_a'], d['label_b']) 

207 rows = d.get('rows', []) 

208 if not rows: 

209 return ( 

210 f'<section id="{section.kind}"><h2>{_esc(section.title)}</h2><p>No comparisons in this run.</p></section>' 

211 ) 

212 

213 parts = [f'<section id="{section.kind}"><h2>{_esc(section.title)}</h2>'] 

214 for row in rows: 

215 chips = ''.join(_vote_chip(v) for v in row.get('votes', [])) 

216 status = _WINNER_STATUS.get(row['winner'], 'pass') 

217 mark = '<span class="pw-split" title="judges split or panel undecided">⚠</span>' if row.get('split') else '' 

218 parts.append( 

219 '<details class="pw-cmp">' 

220 '<summary>' 

221 f'<span class="pw-cmp__idx">{row["index"]}</span>' 

222 f'<span class="pw-cmp__q">{_esc(row["question"])}</span>' 

223 f'{_status_badge(row["winner_label"], status)}' 

224 f'<span class="pw-cmp__chips">{chips}</span>{mark}' 

225 '</summary>' 

226 f'{_detail_html(row, labels)}' 

227 '</details>' 

228 ) 

229 parts.append('</section>') 

230 return ''.join(parts)

282def _swap_summary(run: PairwiseRun) -> str: 

283 """Report swapping as configured, qualified by whether it was ever observed. 

284 

285 A bare "on" taken from the flag is the claim a reader relies on when 

286 deciding whether the position-bias column means anything, so it says so when 

287 the votes never bore it out. 

288 """ 

289 if not run.swap: 

290 return 'off' 

291 if _observed_swap(run): 

292 return 'on' 

293 return 'on <span class="pw-muted">(never observed)</span>'

296def _header_html(run: PairwiseRun) -> str: 

297 report = run.rollup() 

298 meta = ( 

299 f'<p><strong>Comparing:</strong> {_esc(run.label_a)} vs {_esc(run.label_b)} &nbsp;|&nbsp; ' 

300 f'<strong>Judges:</strong> {len(run.judges) or len(report.per_judge)} &nbsp;|&nbsp; ' 

301 f'<strong>Swap:</strong> {_swap_summary(run)} &nbsp;|&nbsp; ' 

302 f'<strong>Date:</strong> {_format_date(run.created_at)}</p>' 

303 ) 

304 criteria = f'<p class="pw-caption"><strong>Criteria:</strong> {_esc(run.criteria)}</p>' if run.criteria else '' 

305 return f'<header class="hero"><h1>{_esc(run.run_name)}</h1>{meta}{criteria}</header>'

318def export_html(run: PairwiseRun) -> str: 

319 """Render a pairwise run as a self-contained HTML document.""" 

320 head = ( 

321 '<meta charset="utf-8">\n' 

322 '<meta name="viewport" content="width=device-width, initial-scale=1">\n' 

323 f'<title>{_esc(run.run_name)} - Pairwise Report</title>\n' 

324 f'<style>\n{_load_css()}\n</style>' 

325 ) 

326 return ( 

327 '<!DOCTYPE html>\n' 

328 '<html lang="en">\n' 

329 f'<head>\n{head}\n</head>\n' 

330 '<body>\n<div class="container">\n' 

331 f'{render_report_body(run)}\n' 

332 '</div>\n</body>\n' 

333 '</html>\n' 

334 )
