1"""Run a local coding-agent CLI (Claude Code, Codex CLI, OpenCode) as an ``AgentTarget``. 

2 

3Each ``respond()`` renders the whole transcript into one prompt, runs a fresh agent process in a 

4per-clone temporary working directory, parses the agent's JSONL stdout into an ``AgentResponse`` 

5and maps failures to ``cli.*`` error codes via ``map_error``. There is no session resume. 

6 

7Two launchers: ``direct`` runs the binary with the caller's environment and credentials; ``orq`` 

8runs ``orq launch <agent>`` so every model call goes through the Orq gateway with the workspace's 

9skills and MCP server attached. 

10 

11Retry lives in ``common.target_call.call_target_with_retry`` only. ``respond()`` never retries. 

12 

13With container mode, one long-lived container belongs to each target clone and each turn runs through 

14``docker exec``. ``close()``, cancellation, and early turn termination remove that container. 

15"""

212def build_argv( 

213 *, 

214 agent: AgentName, 

215 launcher: Launcher, 

216 model: str | None, 

217 permission_mode: str | None, 

218 system_prompt: str | None, 

219 extra_args: list[str] | None, 

220 orq: OrqLaunchOptions | None, 

221 prompt: str, 

222) -> tuple[list[str], str | None]: 

223 """Return ``(argv, stdin_text)``. ``stdin_text`` is ``None`` when the child's stdin must be closed. 

224 

225 Under ``direct`` the prompt travels on stdin (claude and opencode read it there; codex needs the 

226 ``-`` positional), so a long transcript never hits ``ARG_MAX``. Under ``orq`` the codex and 

227 opencode prompts go through ``orq launch -p`` and stdin is closed, because codex appends any 

228 open stdin to the prompt. Claude under ``orq`` still reads stdin. 

229 """ 

230 spec = AGENTS[agent] 

231 if permission_mode is not None and spec.permission_flag is None: 

232 raise ValueError(f'{agent} has no permission-mode flag; pass its own flags through extra_args instead') 

233 

234 agent_args: list[str] = list(spec.output_args) 

235 if launcher == 'direct' and model: 

236 agent_args += [spec.model_flag, model] 

237 if permission_mode is not None and spec.permission_flag is not None: 

238 agent_args += [spec.permission_flag, permission_mode] 

239 if system_prompt and spec.system_prompt_flag: 

240 agent_args += [spec.system_prompt_flag, system_prompt] 

241 agent_args += list(extra_args or []) 

242 

243 if launcher == 'direct': 

244 return [spec.binary, *agent_args, *spec.stdin_marker], prompt 

245 

246 orq_flags = ['--model', model] if model else [] 

247 orq_flags += (orq or OrqLaunchOptions()).to_flags() 

248 if agent == 'claude': 

249 return ['orq', 'launch', 'claude', *orq_flags, '--', *agent_args], prompt 

250 # orq launch injects the subcommand (``exec`` / ``run``) itself, so the leading entry is dropped. 

251 return ['orq', 'launch', agent, *orq_flags, '-p', prompt, '--', *agent_args[1:]], None

261def message_to_dict(message: Message) -> dict[str, Any]: 

262 if message.role == 'tool': 

263 return { 

264 'role': 'tool', 

265 'tool_call_id': message.tool_call_id or '', 

266 'name': message.name, 

267 'content': content_to_text(message.content), 

268 } 

269 out: dict[str, Any] = { 

270 'role': message.role, 

271 'content': None if message.content is None else content_to_text(message.content), 

272 } 

273 if message.tool_calls: 

274 out['tool_calls'] = [ 

275 {'id': tc.id, 'name': tc.function.name, 'arguments': tc.function.arguments} for tc in message.tool_calls 

276 ] 

277 return out

280def render_prompt(messages: list[Message], *, system_prompt: str | None, inline_system: bool) -> str: 

281 """Render the transcript as a delimited JSON conversation followed by the reply instruction. 

282 

283 ``inline_system`` prepends ``system_prompt`` as a ``system`` entry for agents without a 

284 system-prompt flag (codex, opencode). Claude receives it via ``--append-system-prompt`` instead. 

285 """ 

286 entries: list[dict[str, Any]] = [] 

287 if inline_system and system_prompt: 

288 entries.append({'role': 'system', 'content': system_prompt}) 

289 entries += [message_to_dict(m) for m in messages] 

290 block = delimit(json.dumps(entries, ensure_ascii=False, indent=None), tag='conversation') 

291 return f'{PROMPT_INSTRUCTION}\n{block}'

324def usage_or_none(agent: str, usage_block: Any, required: tuple[str, str]) -> Usage | None: 

325 """``Usage.extract`` for one agent usage record; ``None`` plus a warning when a required count is absent.""" 

326 if usage_block is None:

335def opencode_usage(tokens: Any) -> dict[str, Any] | None: 

336 """Spell OpenCode's ``step_finish.tokens`` block in the canonical names ``Usage.extract`` reads. 

337 

338 OpenCode uses bare ``input``/``output``/``reasoning`` and a nested ``cache`` object; those words are 

339 too generic to live in the shared alias table, so the rename stays here. 

340 """ 

341 if not isinstance(tokens, dict): 

342 return None 

343 cache: dict[str, Any] = tokens['cache'] if isinstance(tokens.get('cache'), dict) else {} 

344 renamed: dict[str, Any] = { 

345 'input_tokens': tokens.get('input'), 

346 'output_tokens': tokens.get('output'), 

347 'cache_read_input_tokens': cache.get('read'), 

348 'cache_creation_input_tokens': cache.get('write'), 

349 } 

350 if tokens.get('reasoning') is not None: 

351 renamed['output_tokens_details'] = {'reasoning_tokens': tokens['reasoning']} 

352 # `total` is left out: OpenCode adds cache reads on top of input there, which breaks the 

353 # `total == input + output` invariant Usage keeps, so the extractor's own sum is used instead. 

354 return {k: v for k, v in renamed.items() if v is not None}

357def heaviest_model(model_usage: dict[str, dict[str, Any]]) -> str | None: 

358 """The ``modelUsage`` entry with the most tokens; the first key is often an auxiliary model.""" 

359 if not model_usage: 

360 return None 

361 return max(model_usage, key=lambda m: model_usage[m].get('inputTokens', 0) + model_usage[m].get('outputTokens', 0))

369def parse_claude(events: list[dict[str, Any]]) -> ParsedTurn: 

370 turn = ParsedTurn() 

371 calls: dict[str, ToolCallOutputItem] = {} 

372 order: list[str] = [] 

373 last_text: list[str] = [] 

374 content_events = [e for e in events if e.get('type') not in CLAUDE_HOUSEKEEPING_EVENTS] 

375 for event in content_events: 

376 kind = event.get('type') 

377 if kind == 'assistant': 

378 turn.model = event.get('message', {}).get('model') or turn.model 

379 last_text = [] 

380 for block in event.get('message', {}).get('content', []) or []: 

381 if block.get('type') == 'tool_use': 

382 call_id = str(block.get('id')) 

383 calls[call_id] = tool_call( 

384 call_id=call_id, 

385 name=str(block.get('name')), 

386 arguments=block.get('input', {}), 

387 result=None, 

388 status='in_progress', 

389 ) 

390 order.append(call_id) 

391 elif block.get('type') == 'text': 

392 last_text.append(str(block.get('text', ''))) 

393 elif kind == 'user': 

394 for block in event.get('message', {}).get('content', []) or []: 

395 if isinstance(block, dict) and block.get('type') == 'tool_result': 

396 call_id = str(block.get('tool_use_id')) 

397 if call_id in calls: 

398 status: Literal['completed', 'incomplete'] = ( 

399 'incomplete' if block.get('is_error') else 'completed' 

400 ) 

401 calls[call_id] = calls[call_id].model_copy( 

402 update={ 

403 'result': tool_result_to_text(block.get('content')), 

404 'status': FunctionCallStatus(status), 

405 } 

406 ) 

407 elif kind == 'result': 

408 turn.session_id = event.get('session_id') 

409 turn.cost_usd = event.get('total_cost_usd') 

410 result_text = event.get('result') 

411 turn.text = result_text if isinstance(result_text, str) else (''.join(last_text) or None) 

412 if event.get('is_error'): 

413 turn.agent_error = result_text if isinstance(result_text, str) else str(event.get('subtype')) 

414 turn.usage = usage_or_none('claude', event.get('usage'), ('input_tokens', 'output_tokens')) 

415 turn.model = turn.model or heaviest_model(event.get('modelUsage') or {}) 

416 for denial in event.get('permission_denials') or []: 

417 call_id = str(denial.get('tool_use_id') or f'denied-{len(order)}') 

418 denied = tool_call( 

419 call_id=call_id, 

420 name=str(denial.get('tool_name')), 

421 arguments=denial.get('tool_input', {}), 

422 result='[denied by claude]', 

423 status='incomplete', 

424 ) 

425 if call_id not in calls: 

426 order.append(call_id) 

427 calls[call_id] = denied 

428 else: 

429 logger.warning(f'claude skipped unknown event type: {kind}') 

430 if turn.text is None and last_text: 

431 turn.text = ''.join(last_text) 

432 turn.tool_calls = [calls[c] for c in order] 

433 return turn

436def codex_status(item: dict[str, Any]) -> Literal['in_progress', 'completed', 'incomplete']: 

437 status = item.get('status') 

438 if status == 'completed' and (item.get('exit_code') in (None, 0)): 

439 return 'completed' 

440 if status in ('in_progress', None): 

441 return 'in_progress' 

442 return 'incomplete'

445def parse_codex(events: list[dict[str, Any]]) -> ParsedTurn: 

446 turn = ParsedTurn() 

447 calls: dict[str, ToolCallOutputItem] = {} 

448 order: list[str] = [] 

449 for event in events: 

450 kind = event.get('type') 

451 if kind == 'thread.started': 

452 turn.session_id = event.get('thread_id') 

453 elif kind == 'turn.failed': 

454 turn.agent_error = str((event.get('error') or {}).get('message') or 'turn.failed') 

455 elif kind == 'turn.completed': 

456 turn.usage = usage_or_none('codex', event.get('usage'), ('input_tokens', 'output_tokens')) 

457 elif kind in ('item.started', 'item.completed'): 

458 item = event.get('item') or {} 

459 item_id = str(item.get('id')) 

460 item_type = item.get('type') 

461 if item_type == 'agent_message': 

462 turn.text = str(item.get('text', '')) 

463 elif item_type == 'error': 

464 logger.warning(f'codex reported: {item.get("message")}') 

465 elif item_type == 'command_execution': 

466 calls[item_id] = tool_call( 

467 call_id=item_id, 

468 name='shell', 

469 arguments={'command': item.get('command')}, 

470 result=item.get('aggregated_output') if kind == 'item.completed' else None, 

471 status=codex_status(item), 

472 ) 

473 elif item_type == 'file_change': 

474 calls[item_id] = tool_call( 

475 call_id=item_id, 

476 name='apply_patch', 

477 arguments={'changes': item.get('changes', [])}, 

478 result='applied' if item.get('status') == 'completed' else None, 

479 status=codex_status(item), 

480 ) 

481 elif item_type == 'mcp_tool_call': 

482 error = item.get('error') or {} 

483 calls[item_id] = tool_call( 

484 call_id=item_id, 

485 name=f'{item.get("server")}.{item.get("tool")}', 

486 arguments=item.get('arguments', {}), 

487 result=error.get('message') if error else tool_result_to_text(item.get('result')), 

488 status='incomplete' if error else codex_status(item), 

489 ) 

490 else: 

491 logger.warning(f'codex skipped unknown item type: {item_type}') 

492 continue 

493 if item_id in calls and item_id not in order: 

494 order.append(item_id) 

495 else: 

496 logger.warning(f'codex skipped unknown event type: {kind}') 

497 turn.tool_calls = [calls[c] for c in order] 

498 return turn

501def parse_opencode(events: list[dict[str, Any]]) -> ParsedTurn: 

502 turn = ParsedTurn() 

503 last_text: str | None = None 

504 cost = 0.0 

505 saw_step_finish = False 

506 saw_cost = False 

507 for event in events: 

508 turn.session_id = turn.session_id or event.get('sessionID') 

509 part = event.get('part') or {} 

510 kind = event.get('type') 

511 if kind == 'text': 

512 last_text = str(part.get('text', '')) 

513 elif kind == 'tool_use': 

514 state = part.get('state') or {} 

515 status_raw = state.get('status') 

516 status: Literal['in_progress', 'completed', 'incomplete'] = ( 

517 'completed' if status_raw == 'completed' else 'incomplete' if status_raw == 'error' else 'in_progress' 

518 ) 

519 call_id = str(part.get('callID') or part.get('id')) 

520 turn.tool_calls.append( 

521 tool_call( 

522 call_id=call_id, 

523 name=str(part.get('tool')), 

524 arguments=state.get('input', {}), 

525 result=tool_result_to_text(state.get('output')) 

526 if state.get('output') is not None 

527 else tool_result_to_text(state.get('error')) 

528 if 'error' in state 

529 else None, 

530 status=status, 

531 ) 

532 ) 

533 elif kind == 'step_finish': 

534 saw_step_finish = True 

535 step_usage = usage_or_none( 

536 'opencode', opencode_usage(part.get('tokens')), ('input_tokens', 'output_tokens') 

537 ) 

538 if step_usage is not None: 

539 turn.usage = step_usage if turn.usage is None else turn.usage + step_usage 

540 if 'cost' in part: 

541 saw_cost = True 

542 cost += float(part['cost'] or 0) 

543 elif kind == 'error': 

544 turn.agent_error = str((part.get('error') or event.get('error') or {}).get('message') or 'error') 

545 else: 

546 logger.warning(f'opencode skipped unknown event type: {kind}') 

547 if last_text is not None: 

548 turn.text = last_text 

549 if saw_step_finish and turn.usage is None: 

550 logger.warning('opencode step_finish carried no usable token counts; usage is unknown') 

551 if saw_cost: 

552 turn.cost_usd = cost 

553 return turn

584async def finish_task_uninterruptibly(task: asyncio.Future[Any]) -> None: 

585 """Wait for a shielded operation despite repeated caller cancellation, consuming its result.""" 

586 while not task.done(): 

587 try: 

588 await asyncio.shield(task) 

589 except asyncio.CancelledError: # noqa: PERF203 — the caller may cancel this wait more than once. 

590 continue 

591 except (OSError, subprocess.TimeoutExpired): 

592 break 

593 with contextlib.suppress(BaseException): 

594 task.result()

607class CodingAgentTarget(AgentTarget): 

608 """Run a local coding-agent CLI as the system under test. 

609 

610 Each ``respond()`` spawns a fresh process (no session resume), in a working directory that is 

611 private to this instance: an empty temp dir, or a copy of ``workdir``. ``new()`` clones carry 

612 the same configuration and get their own copy, so concurrent jobs never share a tree. ``close()`` 

613 kills a live process group and removes the temp dir unless ``keep_workdir`` is set; the runner 

614 calls it through ``common.target_call.close_target``. A ``weakref.finalize`` is the leak backstop. 

615 

616 Privilege is expressed in each agent's own vocabulary and defaults to the agent's own default: 

617 

618 - ``permission_mode``: claude ``--permission-mode`` (``default``, ``acceptEdits``, ``plan``, 

619 ``bypassPermissions``, ``dontAsk``); codex ``--sandbox`` (``read-only``, ``workspace-write``, 

620 ``danger-full-access``); opencode has no such flag and raises ``ValueError``. 

621 - ``extra_args``: appended to the agent argv untouched. OpenCode blocks on its first permission 

622 prompt when run headless; pass ``extra_args=['--auto']`` to auto-approve. 

623 

624 ``system_prompt`` is claude's ``--append-system-prompt``; codex and opencode have no equivalent, so 

625 it is prepended to the rendered conversation as a ``system`` entry and a warning says so. 

626 

627 ``launcher='orq'`` runs ``orq launch <agent>`` so model calls route through the Orq gateway with 

628 the workspace skills and MCP server attached; ``orq`` tunes its flags and is ignored (with a warning) 

629 under ``direct``. ``model`` is the agent's own flag under ``direct`` and ``orq launch --model 

630 provider/id`` under ``orq``. 

631 

632 ``env`` overlays ``os.environ``; caller-supplied values win. ``timeout_ms`` is the idle limit: the 

633 maximum time without agent output. ``max_turn_ms`` is the wall-clock hard cap. Both raise the 

634 non-retryable ``cli.timeout`` error. Coding agents enforce these limits themselves, so the runner's 

635 ``target_agent_timeout_ms`` does not apply. 

636 """

639 manages_own_timeout = True

750 async def _docker(self, argv: list[str], *, timeout_s: float = 60) -> subprocess.CompletedProcess[str]: 

751 """Run one bounded Docker control call with the host environment.""" 

752 opts = self._container 

753 if opts is None: 

754 raise RuntimeError('container options are required for Docker calls') 

755 operation = asyncio.create_task( 

756 asyncio.to_thread(subprocess.run, argv, capture_output=True, text=True, timeout=timeout_s, check=False) 

757 ) 

758 try: 

759 return await asyncio.shield(operation) 

760 except asyncio.CancelledError: 

761 # `to_thread` cannot stop subprocess.run. Wait for its bounded result before allowing cleanup 

762 # to issue `rm`, so a slow `docker run` cannot create a container after cleanup has passed. 

763 await finish_task_uninterruptibly(operation) 

764 raise 

765 except FileNotFoundError as exc: 

766 raise CodingAgentUnavailableError('cli.not_found', f'{opts.binary!r} not found on PATH') from exc 

767 except (subprocess.TimeoutExpired, OSError) as exc: 

768 operation_name = next( 

769 (arg for arg in argv if arg in {'image', 'run', 'inspect', 'exec', 'rm', 'ps'}), 'control' 

770 ) 

771 detail = str(exc)[:STDERR_EXCERPT_CHARS].strip() or type(exc).__name__ 

772 raise CodingAgentUnavailableError( 

773 'cli.container_start', f'{opts.binary} {operation_name} control call failed: {detail}' 

774 ) from exc

840 state = await self._docker([*opts.cli(), 'inspect', '-f', '{{.State.Running}}', name])

884 if self._workdir is not None:

1014 raise CodingAgentUnavailableError( 

1015 'cli.agent_not_found',

1017 f'{stderr_excerpt}', 

1018 )

1020 log_kill(self._agent, 'container_lost', f'container {name} was removed while the agent ran') 

1021 await self._drop_container_shielded() 

1022 set_span_attrs(span, {'evaluatorq.coding_agent.kill_reason': 'container_lost'}) 

1023 raise CodingAgentUnavailableError( 

1024 'cli.timeout', 

1025 'the container was removed while the agent was running', 

1026 kill_reason='container_lost', 

1027 )

1034 )

1045 else: 

1046 record_token_usage(span, usage=turn.usage, total_cost=turn.cost_usd)

1048 span, {'evaluatorq.coding_agent.cost_usd': turn.cost_usd, 'gen_ai.response.id': turn.session_id} 

1049 )

1068 async def _run( 

1069 self, 

1070 argv: list[str], 

1071 stdin_text: str | None, 

1072 *, 

1073 cwd: Path, 

1074 env: dict[str, str], 

1075 ) -> tuple[int, str, str]: 

1076 """Run one agent process; end it on the idle limit, the hard cap, an error or cancellation. 

1077 

1078 Stdout and stderr are drained concurrently, and chunks from either stream reset the idle deadline. 

1079 Stdout is read in chunks, so neither a long JSONL line nor a partial one can stall or crash the read. 

1080 The caller owns container cleanup after this method has killed the process group on early exit. 

1081 """ 

1082 try: 

1083 proc = await asyncio.create_subprocess_exec( 

1084 *argv, 

1085 cwd=cwd, 

1086 env=env, 

1087 stdin=asyncio.subprocess.PIPE if stdin_text is not None else asyncio.subprocess.DEVNULL, 

1088 stdout=asyncio.subprocess.PIPE, 

1089 stderr=asyncio.subprocess.PIPE, 

1090 start_new_session=True, 

1091 )

1159) -> CodingAgentUnavailableError | None: 

1160 if now >= started + hard_s:

1162 detail = f'turn exceeded {hard_s:.0f}s ({event_count} events)' 

1163 log_kill(agent, 'hard_cap', detail)
