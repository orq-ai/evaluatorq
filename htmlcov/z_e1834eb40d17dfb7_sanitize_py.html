1"""Shared sanitization utilities for prompt injection prevention. 

2 

3Canonical for every surface: untrusted text reaching a prompt goes through 

4``delimit`` — do not hand-roll boundary markers. 

5 

6Two complementary functions are provided: 

7 

8* ``delimit(text, tag="data")`` — wraps user-controlled text in XML-like 

9 boundary tags (``<tag>…</tag>``) *and* escapes any occurrences of the 

10 chosen tag inside the text so the boundary cannot be broken. Use this 

11 whenever you embed untrusted strings into an LLM prompt. 

12 

13* ``xml_escape(text)`` — character-level XML escaping (``&``, ``<``, ``>``). 

14 Use this when you need to embed text inside XML tags that 

15 are already controlled by the caller (e.g. building an XML payload where 

16 the tag structure is fixed but the content is untrusted). 

17""" 

18 

19from __future__ import annotations 

20 

21import re 

22import xml.sax.saxutils 

23 

24 

25def delimit(text: str, *, tag: str = 'data') -> str: 

26 """Wrap user-controlled text in delimiters to prevent prompt injection. 

27 

28 Uses XML-like boundary tags to clearly separate user content from 

29 system instructions in LLM prompts. Both the opening and closing 

30 tag inside the input are escaped (case-insensitive) to prevent 

31 breakout. 

32 

33 Args: 

34 text: The untrusted text to wrap. 

35 tag: Tag name to use for the boundary (default ``"data"``). 

36 

37 Returns: 

38 The text wrapped in ``<tag>…</tag>`` with internal occurrences 

39 of the tag escaped. 

40 """ 

41 if not re.fullmatch(r'[a-zA-Z][a-zA-Z0-9_-]*', tag): 

42 raise ValueError(f'Invalid tag name: {tag!r}') 

43 sanitized = text.replace('&', '&amp;') 

44 sanitized = re.sub(rf'<{re.escape(tag)}>', f'&lt;{tag}&gt;', sanitized, flags=re.IGNORECASE) 

45 sanitized = re.sub(rf'</{re.escape(tag)}>', f'&lt;/{tag}&gt;', sanitized, flags=re.IGNORECASE) 

46 return f'<{tag}>{sanitized}</{tag}>' 

47 

48 

49def xml_escape(text: str) -> str: 

50 """Escape text for safe embedding inside XML tags. 

51 

52 Wraps `xml.sax.saxutils.escape` for convenience and 

53 discoverability. 

54 """ 

55 return xml.sax.saxutils.escape(text)
