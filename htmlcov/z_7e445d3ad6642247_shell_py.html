131def _sidebar_html(active_nav: str) -> str: 

132 # The v1 brand lockup is the orq mark (not the full wordmark) + the 

133 # "evaluatorq" wordmark with an orange "q". 

134 mark = _load_mark() 

135 logo_html = f'<span class="nav-mark">{mark}</span>' if mark else '' 

136 items: list[str] = [] 

137 for key, label, href, icon in _NAV: 

138 active = ' active' if key == active_nav else '' 

139 items.append(f'<a class="nav-item{active}" href="{href}">{_icon(icon)}<span>{esc(label)}</span></a>') 

140 return ( 

141 '<aside class="app-sidebar">' 

142 f'<a class="app-brand" href="/">{logo_html}' 

143 '<span class="brand-name">evaluator<span class="brand-q">q</span></span></a>' 

144 f'<nav class="app-nav" aria-label="Primary">{"".join(items)}</nav>' 

145 '<button class="sidebar-toggle" type="button" aria-label="Toggle sidebar"' 

146 ' onclick="eqToggleSidebar()">' 

147 '<svg class="nav-icon" width="16" height="16" viewBox="0 0 24 24" fill="none"' 

148 ' stroke="currentColor" stroke-width="2" stroke-linecap="round"' 

149 ' stroke-linejoin="round"><polyline points="15 18 9 12 15 6"/></svg>' 

150 '<span>Collapse</span>' 

151 '<kbd class="sidebar-toggle-kbd">⌘ B</kbd></button>' 

152 '</aside>' 

153 )

164def page( 

165 title: str, 

166 body_html: str, 

167 *, 

168 active_surface: str | None = None, 

169 active_nav: str | None = None, 

170 actions_html: str = '', 

171 back_html: str = '', 

172) -> str: 

173 """Render a complete HTML page in the dashboard sidebar shell. 

174 

175 Args: 

176 title: ``<title>`` text and the topbar heading. 

177 body_html: Pre-rendered HTML fragment for the ``<main>`` body. 

178 active_surface: Surface key (``'redteam'`` | ``'sim'`` | ``'pairwise'``) 

179 for the report view, used to highlight the matching nav item. 

180 active_nav: Explicit nav key (``'dashboard'`` | ``'redteam'`` | ``'sim'`` 

181 | ``'pairwise'`` | ``'settings'``) overriding the surface-derived 

182 default. 

183 actions_html: Optional pre-rendered HTML for the topbar action area 

184 (e.g. export buttons on a report view). 

185 

186 Returns: 

187 A complete HTML document string starting with ``<!DOCTYPE html>``. 

188 """ 

189 css = load_css() 

190 nav_key = _resolve_nav(active_surface, active_nav) 

191 sidebar = _sidebar_html(nav_key) 

192 scripts = ''.join(str(a) for a in head_assets()) 

193 # On report pages the run name is the hero H1, so the topbar carries the 

194 # back link instead of repeating the title. 

195 topbar_lead = back_html or f'<h1 class="app-title">{esc(title)}</h1>' 

196 

197 return ( 

198 '<!DOCTYPE html>\n' 

199 '<html lang="en">\n' 

200 '<head>\n' 

201 '<meta charset="utf-8">\n' 

202 '<meta name="viewport" content="width=device-width, initial-scale=1">\n' 

203 f'{_favicon_link()}' 

204 f'<title>{esc(title)} — evaluatorq</title>\n' 

205 f'<style>\n{css}\n</style>\n' 

206 f'<style>\n{EDITORIAL_CSS}\n</style>\n' 

207 f'<style>\n{DASHBOARD_CSS}\n</style>\n' 

208 f'{scripts}\n' 

209 '</head>\n' 

210 '<body class="eq-dashboard">\n' 

211 f'{_SIDEBAR_TOGGLE_SCRIPT}' 

212 '<div class="app-shell">\n' 

213 f'{sidebar}\n' 

214 '<div class="app-main">\n' 

215 '<header class="app-topbar">\n' 

216 f'{topbar_lead}\n' 

217 f'<div class="app-actions">{actions_html}</div>\n' 

218 '</header>\n' 

219 '<main class="app-content">\n' 

220 f'{body_html}\n' 

221 '</main>\n' 

222 '</div>\n' 

223 '</div>\n' 

224 '</body>\n' 

225 '</html>\n' 

226 )
