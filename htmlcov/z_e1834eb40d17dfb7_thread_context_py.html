1"""Per-conversation Orq thread id, carried via a ContextVar. 

2 

3Multi-turn calls on the Orq router / Responses API are grouped in Orq 

4observability by sending a ``thread: {id: ...}`` body param on every turn 

5(see https://docs.orq.ai/docs/ai-gateway/thread-management). The id must be 

6stable for one conversation but distinct across conversations. 

7 

8Targets are stateless and shared across concurrently-running conversations 

9(one ``SimulationRunner`` / target instance serves every datapoint), so the 

10thread id can't live on the target instance. A ContextVar carries it instead: 

11each asyncio Task gets an isolated copy of the context, so concurrent 

12conversations don't leak thread ids into each other. 

13 

14Callers that own a conversation (the sim runner, the redteam orchestrator) 

15wrap it in ``conversation_thread()``; the target reads ``current_thread_id()`` 

16and attaches the thread param. When unset, no thread param is sent. 

17""" 

18

23from contextvars import ContextVar

31_thread_id: ContextVar[str | None] = ContextVar('orq_thread_id', default=None) 

32 

33# The evaluatorq surface driving the current run ('agent_simulation' / 

34# 'red_teaming'), sent as request metadata so Orq traces are attributable to the 

35# pipeline that produced them. Carried via a ContextVar for the same reason as 

36# the thread id: targets are stateless and shared across concurrent runs. 

37_pipeline: ContextVar[str | None] = ContextVar('evaluatorq_pipeline', default=None) 

38 

39# Per-run identifier ('run_id') for the active red-team / agent-simulation run. 

40# Carried on the same ContextVar rail as the pipeline label so it reaches the 

41# stateless, shared target/judge/generator instances that read pipeline_metadata(). 

42_run_id: ContextVar[str | None] = ContextVar('evaluatorq_run_id', default=None) 

43 

44# Separator joining the run id and the per-conversation discriminators into a 

45# thread id. Kept out of the OQL query grammar's way: the run id is uuid hex and 

46# the parts are ints / sanitized keys, so ':' never appears inside a segment. 

47_THREAD_ID_SEP = ':' 

48 

49 

50def build_thread_id(run_id: str | None, *parts: object) -> str | None: 

51 """Compose a run-scoped thread id, or None when there is no run to link. 

52 

53 Single source of truth for the ``{run_id}:{...}`` format that the dashboard's 

54 run-level deep link (``thread_id:contains:{run_id}``) depends on: sim passes 

55 ``build_thread_id(run_id, index)`` → ``{run_id}:{index}``; red teaming passes 

56 ``build_thread_id(run_id, agent_key, index)`` → ``{run_id}:{agent_key}:{index}``. 

57 The ``run_id`` prefix is what makes the run-level ``contains`` query match 

58 every conversation in the run. 

59 """ 

60 if not run_id: 

61 return None 

62 return _THREAD_ID_SEP.join([run_id, *(str(p) for p in parts)])

65def build_static_thread_id(run_id: str | None, target_id: str, row: int) -> str: 

66 """Build a deterministic thread id for one static red-team datapoint. 

67 

68 Normal runs use the caller's run id, matching dynamic red-team targets. The 

69 fallback keeps direct job invocations (including isolated tests) correlated 

70 to a stable target/row pair instead of minting a random context id. 

71 """ 

72 stable_run_id = run_id or f'static-{target_id}' 

73 thread_id = build_thread_id(stable_run_id, target_id, row) 

74 if thread_id is None: # Defensive: ``stable_run_id`` is always non-empty. 

75 raise RuntimeError('Static thread id requires a non-empty stable run id') 

76 return thread_id

79def current_thread_id() -> str | None: 

80 """Return the thread id for the active conversation, or None if unset.""" 

81 return _thread_id.get()

84def thread_body_param() -> dict[str, dict[str, str]]: 

85 """Return ``{'thread': {'id': ...}}`` for the active conversation, or ``{}``. 

86 

87 Ready to splat into an Orq router request body / SDK ``extra_body``. 

88 """ 

89 tid = _thread_id.get() 

90 return {'thread': {'id': tid}} if tid else {}

93def pipeline_metadata() -> dict[str, str]: 

94 """Return run-identifying invocation metadata for the active run, or ``{}``. 

95 

96 Emits ``evaluatorq_pipeline`` (run surface) and ``evaluatorq_run_id`` (this run's 

97 id) when each is bound. Pass straight to the native ``metadata=`` kwarg on 

98 ``chat.completions.create`` / ``responses.create`` so the invocation's Orq trace 

99 is filterable by run type and by run. 

100 """ 

101 md: dict[str, str] = {} 

102 label = _pipeline.get() 

103 if label: 

104 md['evaluatorq_pipeline'] = label 

105 run_id = _run_id.get() 

106 if run_id: 

107 md['evaluatorq_run_id'] = run_id 

108 return md

111def pipeline_metadata_param() -> dict[str, dict[str, str]]: 

112 """``extra_body``-ready ``{'metadata': {...}}`` form of `pipeline_metadata`. 

113 

114 Wraps `pipeline_metadata` (single source of truth) so the two forms stay 

115 in sync. The SDK merges ``extra_body`` into the request body, yielding the same 

116 top-level ``metadata`` property. 

117 """ 

118 md = pipeline_metadata() 

119 return {'metadata': md} if md else {}

122@contextmanager 

123def evaluatorq_pipeline(label: str) -> Iterator[str]: 

124 """Bind the pipeline label ('agent_simulation' / 'red_teaming') for a run. 

125 

126 Restores the previous value on exit so nested/sequential runs don't bleed. 

127 

128 Yields: 

129 The bound pipeline label. 

130 """ 

131 token = _pipeline.set(label) 

132 try: 

133 yield label 

134 finally: 

135 _pipeline.reset(token)

156def _evaluatorq_run_scope(run_id: str, span: Span | None = None) -> Iterator[str]: 

157 """Stamp a root span and bind its run id for nested work.

169@contextmanager 

170def conversation_thread(thread_id: str | None = None) -> Iterator[str]: 

171 """Bind a thread id for the duration of one conversation. 

172 

173 Generates a uuid when none is given. Restores the previous value on exit so 

174 nested/sequential conversations don't bleed into each other. 

175 

176 Yields: 

177 The bound thread id. 

178 """ 

179 tid = thread_id or str(uuid.uuid4()) 

180 token = _thread_id.set(tid) 

181 try: 

182 yield tid 

183 finally: 

184 _thread_id.reset(token)
