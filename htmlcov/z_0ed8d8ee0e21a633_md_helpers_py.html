16# both are re-exported from evaluatorq.common.reports under the name ``pct``, so a 

17# second definition here silently shadows one of them depending on import order —

22def fmt_cost(cost: float | None) -> str: 

23 """Format a USD cost value, e.g. ``0.0032`` -> ``'$0.0032'``. 

24 

25 The single cost formatter for every surface — reports and dashboard alike — 

26 so the same value never renders as ``$0.0032`` in one place and ``$0.00`` in 

27 another. Fixed at 4dp because per-call costs are routinely sub-cent, and 

28 rounding those to cents rounds them to nothing. 

29 

30 ``None`` means the provider did not report a cost — distinct from a real 

31 ``0.0`` — and renders as an em dash. Callers that would rather omit the row 

32 entirely should check for ``None`` themselves rather than render the dash. 

33 """ 

34 if cost is None: 

35 return '—' 

36 return f'${cost:,.4f}'

39def cost_coverage(priced_calls: int, calls: int) -> str: 

40 """Label a cost total that only some calls contributed to, e.g. ``' (3 of 10 calls)'``. 

41 

42 Returns ``''`` when every call was priced, or when ``priced_calls`` is 0 — 

43 reports written before this was tracked have no coverage data, and claiming 

44 "0 of N" for them would be a lie in the other direction. 

45 """ 

46 if 0 < priced_calls < calls: 

47 return f' ({priced_calls:,} of {calls:,} calls)' 

48 return ''

51def bar(rate: float | None, width: int = 10) -> str: 

52 """Render a Unicode block-character progress bar with a numeric percentage. 

53 

54 Uses U+2588 (full block) for filled segments and U+2591 (light shade) for 

55 empty segments. Always ``width`` characters wide, followed by the numeric 

56 percentage. Example: ``'████░░░░░░ 40%'``. An unknown rate renders as an empty 

57 bar labelled ``n/a`` — there is no bar length that honestly means "not measured". 

58 """ 

59 if rate is None: 

60 return '░' * width + f' {pct(rate)}' 

61 filled = round(rate * width) 

62 return '█' * filled + '░' * (width - filled) + f' {pct(rate)}'

65def bold_bar(rate: float | None, threshold: float = 0.5) -> str: 

66 """Return a Unicode bar, bolded when rate exceeds ``threshold``.""" 

67 cell = bar(rate) 

68 return f'**{cell}**' if rate is not None and rate > threshold else cell

71def md_table( 

72 headers: list[str], 

73 rows: list[list[str]], 

74 right_align: set[int] | None = None, 

75) -> str: 

76 """Render a Markdown table from headers and string rows. 

77 

78 Args: 

79 headers: Column header labels. 

80 rows: Table data rows; each element is a list of cell values. 

81 right_align: Optional set of zero-based column indices that should be 

82 right-aligned (rendered with ``---:`` separator). 

83 """ 

84 right_align = right_align or set() 

85 lines: list[str] = [] 

86 lines.append('| ' + ' | '.join(headers) + ' |') 

87 separators = ['---:' if i in right_align else '---' for i in range(len(headers))] 

88 lines.append('| ' + ' | '.join(separators) + ' |') 

89 for row in rows: 

90 sanitized = [str(cell).replace('|', '\\|').replace('\n', ' ') for cell in row] 

91 lines.append('| ' + ' | '.join(sanitized) + ' |') 

92 return '\n'.join(lines)

95def center_table(headers: list[str], rows: list[list[str]]) -> str: 

96 """Render a Markdown table with all columns center-aligned.""" 

97 sep = ' | '.join(':---:' for _ in headers) 

98 lines = [ 

99 '| ' + ' | '.join(headers) + ' |', 

100 '| ' + sep + ' |', 

101 ] 

102 for row in rows: 

103 sanitized = [str(c).replace('|', '\\|').replace('\n', ' ') for c in row] 

104 lines.append('| ' + ' | '.join(sanitized) + ' |') 

105 return '\n'.join(lines)

108def truncate(text: str, max_chars: int = 800) -> str: 

109 """Truncate long text with an ellipsis indicator.""" 

110 if len(text) <= max_chars: 

111 return text 

112 return text[:max_chars] + '\n\n*[truncated — full text in report JSON]*'

115def details_block(summary: str, body: str) -> str: 

116 """Wrap content in a collapsible ``<details>`` block. 

117 

118 The ``summary`` argument is HTML-escaped so titles containing ``&``, 

119 ``<``, ``>`` render correctly in GitHub-Flavored Markdown. 

120 """ 

121 inner = textwrap.indent(body.strip(), ' ') 

122 escaped_summary = html.escape(summary) 

123 return f'<details>\n<summary>{escaped_summary}</summary>\n\n{inner}\n\n</details>'
