1"""Framework-neutral LLM generator for the report executive-summary narrative. 

2 

3Both red team and simulation reports call this with a compact ``facts`` string 

4AND their own ``system_prompt`` (each surface owns its voice — red team narrates 

5in security terms, simulation in goal-completion terms). This module is just the 

6plumbing: send system+facts, return the paragraph. Generation is best-effort: 

7any failure returns ``None`` and the report renders exactly as it did before. 

8 

9This module MUST stay free of ``evaluatorq.redteam`` / ``evaluatorq.simulation`` 

10imports — it is the lower layer both depend on. 

11"""

47# Default prompt (red-team voice). Kept as the module-level constant so callers 

48# that don't pass their own `system_prompt` — red team, existing tests — behave 

49# exactly as before. Simulation supplies its own via SIM_..._SYSTEM_PROMPT. 

50EXECUTIVE_SUMMARY_SYSTEM_PROMPT = """\ 

51You are an AI security expert writing the executive summary of an agent 

52assessment report for a technical but time-poor reader (an engineering lead or 

53security owner). 

54 

55Write ONE paragraph, 2-4 sentences, following this arc: 

56 1. Scope: how many attacks/simulations were run, across how many categories. 

57 2. Verdict with tension: the headline success/resistance rate AND the 

58 failure/vulnerability count and severity in the same breath. Use 

59 "but"/"however" so the risk is not buried. 

60 3. The single sharpest concrete finding: name the one result that matters 

61 most, described as what the agent actually DID ("issued an unauthorized 

62 credit after a three-turn impersonation"), not the taxonomy label. 

63 4. The dominant risk pattern and its implication, grounded ONLY in the 

64 provided numbers (e.g. if multi-turn vulnerability rate exceeds 

65 single-turn, say conversation depth raises risk). 

66 

67Rules: 

68- Lead with the numbers you are GIVEN. Never invent a statistic or a trend the 

69 facts do not state. 

70- Concrete over abstract: describe behavior, not codes. 

71- No preamble ("This report..."), no recommendations, no markdown, no headers. 

72 Prose only. 

73- If nothing notable was found (no failures/vulnerabilities), say so plainly in 

74 one sentence and stop. Do not manufacture risk."""

77# Defaults for the executive-summary completion, overridable per call. Constants so 

78# a tracing call site reports the exact values sent. 400 completion tokens is mostly 

79# hidden reasoning on a reasoning-class model — raise it if summaries come back empty. 

80EXECUTIVE_SUMMARY_TEMPERATURE = 0.7 

81EXECUTIVE_SUMMARY_MAX_TOKENS = 400 

82# Generous for a slow reasoning model, short enough that a hung call cannot hold

109@dataclass(frozen=True) 

110class ExecutiveSummary: 

111 """The narrative paragraph and what it cost. 

112 

113 ``usage`` is priced (`execute_chat_completion` runs it through `price_usage`) 

114 so the caller can fold it into the run total. It is ``None`` when the call 

115 never happened — blank facts — or failed. 

116 """ 

117 

118 text: str | None 

119 usage: TokenUsage | None = None 

120 

121 

122async def generate_executive_summary( 

123 facts: str, 

124 *, 

125 llm_client: AsyncChatCompletionsClient, 

126 model: str, 

127 system_prompt: str = EXECUTIVE_SUMMARY_SYSTEM_PROMPT, 

128 temperature: float = EXECUTIVE_SUMMARY_TEMPERATURE, 

129 max_tokens: int = EXECUTIVE_SUMMARY_MAX_TOKENS, 

130 timeout_s: float = EXECUTIVE_SUMMARY_TIMEOUT_S, 

131 reasoning_effort: str | None = None, 

132 extra_body: dict[str, Any] | None = None, 

133 extra_kwargs: dict[str, Any] | None = None, 

134) -> ExecutiveSummary: 

135 """Generate the executive-summary prose from a pre-built ``facts`` string. 

136 

137 Returns the stripped paragraph plus its priced usage, or a result whose 

138 ``text`` is ``None`` when ``facts`` is blank, the completion is empty, or any 

139 error occurs (logged, never raised) — this is an opt-in narrative step and 

140 must not take down a report that is otherwise complete. 

141 

142 The call goes through `common.llm_call.execute_chat_completion`, so it gets 

143 the same slot limiting, reasoning drop-and-retry, trace headers, pipeline 

144 metadata and `price_usage` as every other chat call — and returns usage the 

145 caller can add to the run total. 

146 

147 Params are built via `LLMCallConfig.request_params`, not a hand-rolled 

148 dict splatted next to explicit ``temperature=``/``max_completion_tokens=`` 

149 keywords: the old shape raised ``TypeError: got multiple values for keyword 

150 argument`` the moment a caller passed ``extra_kwargs={'temperature': 1}`` — 

151 the documented escape hatch for reasoning-class models that reject a 

152 lowered temperature — and that ``TypeError`` was swallowed by the blanket 

153 ``except Exception`` below, so the failure surfaced only as a silently 

154 ``None`` summary. ``request_params`` reserves the same escape hatch by 

155 construction: caller-supplied ``extra_kwargs`` values win the merge. 

156 """ 

157 if not facts or not facts.strip(): 

158 return ExecutiveSummary(None) 

159 try: 

160 cfg = LLMCallConfig( 

161 model=model, 

162 temperature=temperature, 

163 max_tokens=max_tokens, 

164 reasoning_effort=reasoning_effort, 

165 extra_kwargs=extra_kwargs or {}, 

166 ) 

167 params = cfg.request_params(api='chat_completions', extra_body=extra_body or {}) 

168 response, usage = await execute_chat_completion( 

169 # The Protocol exists so a structural test double typechecks here; 

170 # execute_chat_completion only ever touches `chat.completions`. 

171 client=cast('AsyncOpenAI', llm_client), 

172 model=model, 

173 messages=[ 

174 {'role': 'system', 'content': system_prompt}, 

175 {'role': 'user', 'content': facts}, 

176 ], 

177 span=None, 

178 timeout_s=timeout_s, 

179 temperature=params.pop('temperature', None), 

180 max_completion_tokens=params.pop('max_completion_tokens', None), 

181 reasoning_effort=params.pop('reasoning_effort', None), 

182 extra_body=params.pop('extra_body', None), 

183 extra_kwargs=params or None, 

184 ) 

185 _record_usage_on_current_span(response) 

186 content = response.choices[0].message.content or '' 

187 text = content.strip() 

188 return ExecutiveSummary(text or None, usage) 

189 except Exception: 

190 logger.warning('Failed to generate executive summary', exc_info=True) 

191 return ExecutiveSummary(None)
