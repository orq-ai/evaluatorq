49def reconcile_pair(first: VerdictValue | None, second: VerdictValue | None) -> tuple[VerdictValue | None, bool]: 

50 """Reconcile a judge's verdicts from both orderings into one vote. 

51 

52 Both verdicts are in the canonical A/B frame (the caller un-swaps the 

53 second ordering). Returns ``(vote, flipped)``: a consistent judge votes for 

54 the value it gave both times; an inconsistent one abstains (``vote=None``) 

55 and is recorded as a flip. A missing verdict abstains without a flip. 

56 """ 

57 if first is None or second is None: 

58 return None, False 

59 if first == second: 

60 return first, False 

61 return None, True

64def pairwise_consensus(votes: Sequence[VerdictValue | None]) -> str: 

65 """Reduce reconciled per-judge votes to a consensus winner. 

66

206class BTSigmaAggregation(BaseModel): 

207 """Reliability-weighted rollup of a pairwise run (BT-sigma, arXiv:2602.16610). 

208 

209 Fitted unsupervised on the run's own reconciled votes: hard BT-sigma jointly 

210 infers the A-vs-B skill gap and a discriminator per judge, then re-derives 

211 each comparison's winner as a reliability-weighted vote instead of uniform 

212 plurality. Down-weights noisy judges; needs no labels. The weight depends on 

213 the path: on the pooled two-item fit it is ``1/sigma``; on a repetition run 

214 (``repetition_consistency`` non-empty and at quorum) it is instead 

215 ``max(shrunk_consistency, 0.05)``, since within-datapoint consistency is the 

216 better reliability signal there. 

217 

218 Two caveats of the two-item collapse, both handled here: 

219 

220 - With only two items a judge's sigma is pinned by its own vote 

221 distribution, so a judge whose decisive votes are unanimous gets an 

222 arbitrarily small sigma that measures one-sidedness (e.g. position or 

223 verbosity bias), not reliability. Such judges are excluded from the 

224 weighting — they vote with a neutral weight, their sigma is dropped from 

225 ``judge_sigmas``, and ``fit_warnings`` names them. 

226 - ``p_a_beats_b``/``skill_gap`` come from the pooled global fit, while 

227 ``winners`` come from the per-comparison weighted vote. These are two 

228 different estimators and need not agree; read ``p_a_beats_b`` as the 

229 run-level headline and ``winners`` as the per-row calls. 

230 """ 

231 

232 p_a_beats_b: float = Field(description='Fitted global probability that A beats B (logistic of the skill gap)') 

233 skill_gap: float = Field(description='Fitted skill difference s_A - s_B') 

234 judge_sigmas: dict[str, float] = Field( 

235 default_factory=dict, 

236 description='Per-judge discriminator; smaller = sharper/more reliable. Empty on single-judge fallback.', 

237 ) 

238 repetition_consistency: dict[str, float] = Field( 

239 default_factory=dict, 

240 description='Per-judge SHRUNK within-datapoint repetition consistency in [0, 1] (RES-1251); when non-empty ' 

241 'and at quorum, the winner weights came from THESE, not from the global-fit sigmas. Shrunk toward the ' 

242 'panel mean, so it is a reliability weight, not raw self-agreement - a self-consistent judge reads below ' 

243 '1.0 unless the panel is. Measures self-agreement on repeated passes - NOT task difficulty and NOT ' 

244 'overall judge quality.', 

245 ) 

246 repetition_consistency_raw: dict[str, float] = Field( 

247 default_factory=dict, 

248 description='Per-judge RAW within-datapoint self-agreement in [0, 1] (RES-1251): neither shrunk nor ' 

249 'failure-adjusted, published alongside repetition_consistency so the un-shrunk signal (1.0 = always ' 

250 'agreed with itself on every completed pass) stays visible next to the weight derived from it.', 

251 ) 

252 winners: list[str] = Field( 

253 default_factory=list, 

254 description='Reliability-weighted winner per comparison, aligned with the input order', 

255 ) 

256 a_win_rate: float | None = Field(description='Weighted A-wins over decisive comparisons; None if none decisive') 

257 b_win_rate: float | None = Field(description='Weighted B-wins over decisive comparisons; None if none decisive') 

258 tie_rate: float = Field(description='Weighted-tie comparisons over all comparisons') 

259 inconclusive_rate: float = Field(description='Comparisons no judge decided, over all comparisons') 

260 converged: bool = Field( 

261 default=True, 

262 description='False when the fit stopped at the iteration cap; treat sigmas and weighted winners ' 

263 'with suspicion then (the cap is also reported in fit_warnings)', 

264 ) 

265 fit_warnings: list[str] = Field(default_factory=list, description='Degradations applied during the BT fit') 

266 

267 

268_VOTE_TO_P = {'A': 1.0, 'B': 0.0, 'tie': 0.5} 

269# Below this many decisive votes a judge's sigma is mostly prior, not evidence; 

270# the fit still runs but fit_warnings flags the judge. 

271_MIN_VOTES_FOR_SIGMA = 5

274def _uniform_plurality_aggregation( 

275 comparisons: Sequence[PairwiseComparison], warnings: list[str], *, converged: bool = True 

276) -> BTSigmaAggregation: 

277 winners = [c.winner for c in comparisons] 

278 counts = Counter(winners) 

279 decisive = counts['A'] + counts['B'] 

280 total = len(comparisons) 

281 return BTSigmaAggregation( 

282 p_a_beats_b=0.5, 

283 skill_gap=0.0, 

284 winners=winners, 

285 a_win_rate=_rate(counts['A'], decisive), 

286 b_win_rate=_rate(counts['B'], decisive), 

287 tie_rate=counts['tie'] / total if total else 0.0, 

288 inconclusive_rate=counts['inconclusive'] / total if total else 0.0, 

289 converged=converged, 

290 # A diagnostic, not a weight, so publish it even when weighting is off (RES-1251 review, 

291 # item 19): a swap=False or all-abstained run still collected these observations. 

292 repetition_consistency_raw=repetition_consistency_raw(comparisons), 

293 fit_warnings=warnings, 

294 )

297def repetition_consistency(comparisons: Sequence[PairwiseComparison]) -> dict[str, float]: 

298 """Per-judge within-datapoint repetition consistency, in [0, 1] (RES-1251). 

299 

300 The unit of analysis is one (judge, comparison, ordering) group: repeated 

301 passes of the SAME prompt. A group needs >= 2 decisive repetitions; 

302 its consistency is the mean pairwise agreement among them, then discounted 

303 by the share of that vote's passes that errored or came back off-contract 

304 (``repetition_failures``) so a flaky judge scores below a clean one; a 

305 genuine abstention is not counted as a failure and does not penalise. A 

306 comparison contributes at most ONE observation per judge (its groups 

307 averaged), so repetition count never multiplies a judge's evidence, and 

308 different datapoints are never compared to each other - which is exactly why 

309 this is interpretable as judge reliability where the global two-item fit was not. 

310 

311 Empty dict when no judge has any qualifying group (e.g. repetitions=1 or a 

312 legacy run without observations). 

313 

314 This is the SHRUNK reliability weight, not raw self-agreement: at the recommended 

315 R=2 a group's agreement is only 0.0 or 1.0, so a judge with one lucky group would 

316 otherwise carry the same weight as a judge with many. Each judge's mean is shrunk 

317 toward the panel mean by its evidence count (empirical-Bayes, ``_SHRINKAGE_PSEUDO_OBS`` 

318 pseudo-observations), so thin evidence is pulled to neutral and cannot dominate a run. 

319 A perfectly self-consistent judge therefore reads below 1.0 unless the whole panel is; 

320 ``repetition_consistency_raw`` publishes the un-shrunk mean beside it (RES-1251). 

321 """ 

322 stats = _repetition_stats(comparisons) 

323 if not stats: 

324 return {} 

325 # Anchor the shrinkage on the EVIDENCE-COUNT-WEIGHTED panel mean of the discounted 

326 # means (the weight basis), not the plain mean of per-judge means: otherwise a 

327 # thin-evidence judge helps set the very anchor it is then shrunk toward 

328 # (self-reference), compressing the separation the weights are derived from 

329 # (RES-1251 review, item 13). 

330 total_n = sum(n for _, _, n in stats.values()) 

331 panel_mean = sum(disc * n for _, disc, n in stats.values()) / total_n 

332 k = _SHRINKAGE_PSEUDO_OBS 

333 return {judge: (n * disc + k * panel_mean) / (n + k) for judge, (_, disc, n) in stats.items()}

336def _repetition_stats(comparisons: Sequence[PairwiseComparison]) -> dict[str, tuple[float, float, int]]: 

337 """Per-judge repetition stats (RES-1251), computed once and shared. 

338 

339 Returns ``{judge: (raw_agreement_mean, discounted_mean, n)}`` sorted by judge, where ``n`` is the 

340 number of comparisons on which the judge had a qualifying group: 

341 

342 - ``raw_agreement_mean`` is PURE self-agreement (what ``repetition_consistency_raw`` publishes: 

343 1.0 = agreed with itself on every completed pass, regardless of how many passes failed). 

344 - ``discounted_mean`` additionally multiplies each comparison's agreement by its completion 

345 share, and is the basis for the shrunk ``repetition_consistency`` weight - a failed or 

346 off-contract pass lowers reliability while a clean abstention does not. The two differ only 

347 when a judge had failed/off-contract passes, which is exactly the case the split exists for 

348 (review item 17): the raw number stays honest self-agreement, the discount lives in the weight. 

349 """ 

350 raw_scores: dict[str, list[float]] = {} 

351 disc_scores: dict[str, list[float]] = {} 

352 for c in comparisons: 

353 for v in c.votes: 

354 groups: dict[str, list[str]] = {} 

355 for o in v.observations: 

356 if o.verdict is not None: 

357 groups.setdefault(o.ordering, []).append(o.verdict) 

358 group_scores = [] 

359 for vals in groups.values(): 

360 if len(vals) >= 2: 

361 pairs = [a == b for i, a in enumerate(vals) for b in vals[i + 1 :]] 

362 group_scores.append(sum(pairs) / len(pairs)) 

363 if not group_scores: 

364 continue 

365 agreement = sum(group_scores) / len(group_scores) 

366 # The failure discount: a judge that errors or returns off-contract on some passes is 

367 # less reliable, so scale by the share of THIS vote's passes that succeeded. Failures 

368 # are the Nones counted in ``repetition_failures``; a clean abstention is a None that is 

369 # NOT a failure, so it does not penalise. This shapes the WEIGHT only; raw stays pure. 

370 n_obs = len(v.observations) 

371 completion = max(0.0, (n_obs - v.repetition_failures) / n_obs) if n_obs else 1.0 

372 raw_scores.setdefault(v.model, []).append(agreement) 

373 disc_scores.setdefault(v.model, []).append(agreement * completion) 

374 return { 

375 judge: ( 

376 sum(raw_scores[judge]) / len(raw_scores[judge]), 

377 sum(disc_scores[judge]) / len(disc_scores[judge]), 

378 len(raw_scores[judge]), 

379 ) 

380 for judge in sorted(raw_scores) 

381 }

384def repetition_consistency_raw(comparisons: Sequence[PairwiseComparison]) -> dict[str, float]: 

385 """Per-judge RAW within-datapoint self-agreement mean (RES-1251). 

386 

387 1.0 = a judge that always agreed with itself on repeated passes of the same prompt. This is 

388 PURE agreement, not failure-adjusted and not shrunk: a judge that agreed with itself on every 

389 completed pass reads 1.0 even if some passes failed (the failure discount and the shrinkage 

390 both live in the ``repetition_consistency`` weight, not here). Published beside that weight so 

391 the raw signal stays visible. Empty dict when no judge has a qualifying group. 

392 """ 

393 return {judge: raw for judge, (raw, _disc, _n) in _repetition_stats(comparisons).items()}

396def has_repeated_ordering(comparisons: Sequence[PairwiseComparison]) -> bool: 

397 """Whether any judge ran an ordering more than once, i.e. whether repeats were attempted. 

398 

399 This is the "did the user pay for repeats" question, and it is deliberately NOT "are there 

400 observations": R=1 records one pass per ordering, so observations alone are true on the default 

401 path (RES-1251 review, item 15). Distinct from having consistency EVIDENCE - a repeated ordering 

402 whose extra passes all failed or abstained yields no measurable group - which is why both the 

403 ``fit_warnings`` here and the report's empty state need this predicate as well as the 

404 consistency dict: it separates "no repeats ran" from "repeats ran and measured nothing". 

405 """ 

406 return any( 

407 any(count >= 2 for count in Counter(o.ordering for o in v.observations).values()) 

408 for c in comparisons 

409 for v in c.votes 

410 ) 

411 

412 

413# Floor for consistency-derived weights: a judge that never agreed with itself 

414# still casts a (heavily discounted) vote rather than being erased outright. 

415_MIN_CONSISTENCY_WEIGHT = 0.05 

416 

417# Empirical-Bayes shrinkage strength for repetition_consistency: pseudo-observations at the 

418# panel mean, so a judge with thin evidence is pulled toward neutral (RES-1251). In 

419# (n * disc + k * panel_mean) / (n + k), n is the number of DATAPOINTS on which the judge had a 

420# qualifying group (one per comparison), NOT the group/repetition count and unrelated to R. So k=1 

421# is worth one datapoint of evidence: a judge measured on a single datapoint is pulled halfway to 

422# the panel mean, and the prior fades as more datapoints accumulate (2 datapoints -> 1/3 toward the 

423# anchor, 5 -> 1/6, and so on). Raising k shrinks harder / needs more datapoints to escape the 

424# prior; lowering it lets a single datapoint count for more. 1.0 is the chosen magnitude, not just 

425# "shrinkage in principle". 

426_SHRINKAGE_PSEUDO_OBS = 1.0 

427 

428# Reliability weighting from repetition consistency needs a quorum of judges with 

429# evidence. Below this the borrowed fallback is effectively one judge's own value, 

430# so we skip repetition weighting and keep the pooled fit instead (RES-1251 review). 

431# This is a FLOOR, not a sufficiency claim: at exactly two measured judges the median 

432# fallback is just those two averaged and is still borrowed by any number of unmeasured 

433# judges, so more measured judges is always better - two is only where it stops being 

434# one judge deciding the whole panel. 

435_MIN_MEASURED_JUDGES = 2

438def bt_sigma_aggregation(comparisons: Sequence[PairwiseComparison]) -> BTSigmaAggregation: 

439 """Fit hard BT-sigma over a run's reconciled votes and re-derive winners. 

440 

441 The two "items" are the run's A and B sides; every decisive reconciled vote 

442 is one comparison between them. Reconciliation has already symmetrised 

443 position bias (both orderings per judge), satisfying the model's 

444 commutativity requirement. Categorical votes make hard BT-sigma the natural 

445 variant, which is also the paper's most robust one under inconsistency. 

446 

447 Identical votes are collapsed into weighted records before the fit (a judge 

448 has at most three distinct judgements here: A, B, tie), so the fit cost 

449 stays flat no matter how many comparisons the run holds. 

450 

451 Judges whose decisive votes are unanimous are excluded from the reliability 

452 weighting: in the two-item collapse their sigma measures one-sidedness, not 

453 reliability (see `BTSigmaAggregation`). They vote with the median 

454 weight of the remaining judges (or uniformly when no judge remains), and 

455 ``fit_warnings`` names them. 

456 """ 

457 from evaluatorq.ranking import JudgedComparison, comparisons_per_judge, fit_bt 

458 

459 if any(v.vote is not None and not v.completed for c in comparisons for v in c.votes): 

460 return _uniform_plurality_aggregation( 

461 comparisons, 

462 ['single-ordering data: position-bias symmetrisation unavailable; used uniform plurality instead'], 

463 ) 

464 

465 # judge -> vote -> count. Collapsing keeps the optimum identical (the 

466 # likelihood is additive over identical records) while the per-iteration 

467 # cost drops from O(votes) to O(judges x 3). 

468 vote_counts: dict[str, Counter[str]] = {} 

469 for c in comparisons: 

470 for v in c.votes: 

471 if v.vote is not None: 

472 vote_counts.setdefault(v.model, Counter())[v.vote] += 1 

473 records = [ 

474 JudgedComparison(judge=judge, item_a='A', item_b='B', p_a=_VOTE_TO_P[vote], weight=float(n)) 

475 for judge, counts in sorted(vote_counts.items()) 

476 for vote, n in sorted(counts.items()) 

477 ] 

478 if not records: 

479 return BTSigmaAggregation( 

480 p_a_beats_b=0.5, 

481 skill_gap=0.0, 

482 winners=['inconclusive'] * len(comparisons), 

483 a_win_rate=None, 

484 b_win_rate=None, 

485 tie_rate=0.0, 

486 inconclusive_rate=1.0 if comparisons else 0.0, 

487 # Position-biased judges reconcile to no vote, but their repeated passes are still 

488 # collected; publish the raw diagnostic rather than dropping it (RES-1251 review, item 19). 

489 repetition_consistency_raw=repetition_consistency_raw(comparisons), 

490 fit_warnings=['no decisive votes to fit on'], 

491 ) 

492 

493 fit = fit_bt(records, judge_sigma=True, hard=True) 

494 fit_warnings = list(fit.warnings) 

495 # Consistency weights do not depend on the pooled fit, so a non-converged 

496 # fit only loses the run-level headline - repetition evidence still weights 

497 # the winners (RES-1251). Without repeats, fall back as before. 

498 rep_consistency = repetition_consistency(comparisons) 

499 rep_raw = repetition_consistency_raw(comparisons) 

500 # A run can carry a genuine repeat yet produce no consistency evidence: if no single ordering 

501 # held >= 2 DECISIVE passes (e.g. ab=[A, None], ba=[A, None] at R=2), rep_consistency is empty 

502 # and we silently fall back to the pooled fit. Warn there, since the user paid for repeats. 

503 if not rep_consistency and has_repeated_ordering(comparisons): 

504 fit_warnings.append( 

505 'repetition weighting skipped: repeated passes exist but no single ordering held >= 2 decisive ' 

506 'passes to measure consistency; using the pooled two-item fit' 

507 ) 

508 # Reliability weighting needs a quorum of judges with repetition evidence; below 

509 # that the borrowed fallback is one judge's own value, so keep the pooled fit and 

510 # say so rather than weight the whole panel off a single judge (RES-1251 review). 

511 rep_active = len(rep_consistency) >= _MIN_MEASURED_JUDGES 

512 if rep_consistency and not rep_active: 

513 fit_warnings.append( 

514 f'repetition weighting skipped: only {len(rep_consistency)} judge(s) had repeated decisive ' 

515 f'observations, need >= {_MIN_MEASURED_JUDGES}; using the pooled two-item fit' 

516 ) 

517 if not fit.converged: 

518 if not rep_active: 

519 fit_warnings.append('non-converged fit: used uniform plurality winners instead') 

520 return _uniform_plurality_aggregation(comparisons, fit_warnings, converged=False) 

521 fit_warnings.append( 

522 'non-converged fit: p_a_beats_b/skill_gap unavailable (reported neutral); winners still ' 

523 'weighted by repetition consistency' 

524 ) 

525 

526 # Two-item degeneracy guard: a unanimous judge's sigma is a closed-form 

527 # function of its own one-sidedness (its marginal pins the fit), so 1/sigma 

528 # would hand the most degenerate judge on the panel an unbounded weight. 

529 one_sided = {judge for judge, counts in vote_counts.items() if len(counts) == 1} 

530 # Single-judge fallback leaves sigmas empty; weight uniformly then. 

531 sigmas = {judge: sigma for judge, sigma in fit.sigmas.items() if fit.converged and judge not in one_sided} 

532 weights = {judge: fit.reliability(judge) for judge in sigmas} 

533 if fit.sigmas and one_sided: 

534 neutral = statistics.median(weights.values()) if weights else 1.0 

535 for judge in sorted(one_sided): 

536 weights[judge] = neutral 

537 direction = next(iter(vote_counts[judge])) 

538 if rep_active: 

539 # The repetition block below replaces every weight, so the 

540 # neutral assignment never survives; only the sigma exclusion 

541 # is real on this path. 

542 fit_warnings.append( 

543 f"judge '{judge}' voted '{direction}' on every decisive comparison; with two items its sigma " 

544 'measures one-sidedness, not reliability - excluded from judge_sigmas; its weight comes ' 

545 'from repetition consistency' 

546 ) 

547 else: 

548 fit_warnings.append( 

549 f"judge '{judge}' voted '{direction}' on every decisive comparison; with two items its sigma " 

550 'measures one-sidedness, not reliability - weighted neutrally and excluded from judge_sigmas' 

551 ) 

552 if fit.sigmas: 

553 for judge, n in sorted(comparisons_per_judge(records).items()): 

554 if n < _MIN_VOTES_FOR_SIGMA and judge not in one_sided: 

555 fit_warnings.append( 

556 f"judge '{judge}' has only {int(n)} decisive vote(s); its sigma is mostly prior, not evidence" 

557 ) 

558 

559 # RES-1251: when per-repetition observations exist, reliability weights come 

560 # from within-datapoint consistency instead of the global two-item fit, so 

561 # datapoint heterogeneity structurally cannot masquerade as judge noise. 

562 # The pooled fit still supplies the run-level headline (p_a_beats_b) - two 

563 # different questions, two different estimators. 

564 if rep_active: 

565 weights = {judge: max(c, _MIN_CONSISTENCY_WEIGHT) for judge, c in rep_consistency.items()} 

566 voted = {v.model for c in comparisons for v in c.votes if v.vote is not None} 

567 fallback = statistics.median(weights.values()) if weights else 1.0 

568 for judge in sorted(voted - set(weights)): 

569 weights[judge] = fallback 

570 # Not a fixed neutral: it is the median consistency of the MEASURED 

571 # judges, so name the number rather than call it neutral (review). A 

572 # lone measured judge makes this the same as that judge's weight. 

573 fit_warnings.append( 

574 f"judge '{judge}' has no repeated decisive observations; weighted at the median measured " 

575 f'consistency ({fallback:.2f})' 

576 ) 

577 # Mirrors the estimator's per-ordering-group rule: a vote only has 

578 # repeats if some single ordering holds >= 2 decisive passes. Two 

579 # decisive passes split across orderings carry no consistency evidence. 

580 n_dp = sum( 

581 1 

582 for c in comparisons 

583 if any( 

584 any(n >= 2 for n in Counter(o.ordering for o in v.observations if o.verdict is not None).values()) 

585 for v in c.votes 

586 ) 

587 ) 

588 fit_warnings.append( 

589 f'reliability weights from within-datapoint repetition consistency ' 

590 f'({len(rep_consistency)} judge(s), {n_dp} datapoint(s) with repeats)' 

591 ) 

592 

593 winners: list[str] = [] 

594 for c in comparisons: 

595 w: dict[str, float] = {'A': 0.0, 'B': 0.0, 'tie': 0.0} 

596 for v in c.votes: 

597 if v.vote is None: 

598 continue 

599 w[v.vote] += weights.get(v.model, 1.0) 

600 if not any(w.values()): 

601 winners.append('inconclusive') 

602 continue 

603 top = max(w.values()) 

604 # isclose, not ==: mathematically equal weights can differ in the last 

605 # bits (asymmetric comparison counts), and a genuine tie must stay a 

606 # tie rather than crown whichever side rounded up. 

607 leaders = [k for k, val in w.items() if math.isclose(val, top, rel_tol=1e-9)] 

608 # Mirrors plurality semantics: a unique leader wins, a split is inconclusive. 

609 winners.append(leaders[0] if len(leaders) == 1 else 'inconclusive') 

610 

611 counts = Counter(winners) 

612 decisive = counts['A'] + counts['B'] 

613 total = len(comparisons) 

614 return BTSigmaAggregation( 

615 p_a_beats_b=1.0 / (1.0 + math.exp(-(fit.skills['A'] - fit.skills['B']))) if fit.converged else 0.5, 

616 skill_gap=fit.skills['A'] - fit.skills['B'] if fit.converged else 0.0, 

617 judge_sigmas=sigmas, 

618 winners=winners, 

619 a_win_rate=_rate(counts['A'], decisive), 

620 b_win_rate=_rate(counts['B'], decisive), 

621 tie_rate=counts['tie'] / total if total else 0.0, 

622 inconclusive_rate=counts['inconclusive'] / total if total else 0.0, 

623 converged=fit.converged, 

624 repetition_consistency=rep_consistency, 

625 repetition_consistency_raw=rep_raw, 

626 fit_warnings=fit_warnings, 

627 )

630def build_report(comparisons: Sequence[PairwiseComparison], *, aggregation: str = 'plurality') -> PairwiseReport: 

631 """Roll a set of pairwise comparisons up into headline and per-judge metrics. 

632 

633 ``aggregation='plurality'`` (default) keeps the existing uniform plurality 

634 consensus. ``aggregation='bt-sigma'`` additionally fits hard BT-sigma over 

635 the run and attaches the reliability-weighted rollup (``report.bt_sigma``) 

636 plus each judge's discriminator (``JudgeStats.sigma``); the headline 

637 plurality rates are unchanged so the two aggregations stay comparable. 

638 

639 ``JudgeStats.consistency`` / ``consistency_raw`` are filled under BOTH aggregations: they are 

640 read straight off the recorded repetition observations and need no Bradley-Terry fit, so gating 

641 them on ``bt-sigma`` would report "no usable repeats" for a plurality run that has them.""" 

642 total = len(comparisons) 

643 winners = Counter(c.winner for c in comparisons) 

644 decisive = winners['A'] + winners['B'] 

645 

646 per_judge: dict[str, list[PairwiseVote]] = {} 

647 for comparison in comparisons: 

648 for vote in comparison.votes: 

649 per_judge.setdefault(vote.model, []).append(vote) 

650 

651 # Repetition consistency is read off the recorded observations, so it is available under every 

652 # aggregation. Computed here rather than taken from the BT block, which leaves it empty on its 

653 # degraded paths (uniform-plurality fallback, no decisive votes) even when the repeats are there. 

654 rep_consistency = repetition_consistency(comparisons) 

655 rep_consistency_raw = repetition_consistency_raw(comparisons) 

656 

657 judge_stats = [ 

658 JudgeStats( 

659 model=model, 

660 a_rate=_rate(sum(v.vote == 'A' for v in votes), sum(v.vote in ('A', 'B') for v in votes)), 

661 b_rate=_rate(sum(v.vote == 'B' for v in votes), sum(v.vote in ('A', 'B') for v in votes)), 

662 # Flips only over pairs where flipping was possible; a failed or missing 

663 # ordering never had the chance to flip and would dilute the rate. 

664 position_bias=_rate(sum(v.flipped for v in votes), sum(v.completed for v in votes)) or 0.0, 

665 tie_rate=sum(v.vote == 'tie' for v in votes) / len(votes), 

666 consistency=rep_consistency.get(model), 

667 consistency_raw=rep_consistency_raw.get(model), 

668 ) 

669 for model, votes in per_judge.items() 

670 ] 

671 

672 # A single decisive vote scores agreement 1.0 (highest exactly when the panel is 

673 # most degraded); only comparisons with >=2 decisive votes carry real agreement. 

674 agreements = [ 

675 rate 

676 for c in comparisons 

677 if len([v for v in c.votes if v.vote is not None]) >= 2 

678 and (rate := _agreement_rate([v.vote for v in c.votes if v.vote is not None])) is not None 

679 ] 

680 

681 bt_block: BTSigmaAggregation | None = None 

682 if aggregation == 'bt-sigma': 

683 bt_block = bt_sigma_aggregation(comparisons) 

684 for stats in judge_stats: 

685 # Only sigma is gated: it is a product of the fit and does not exist without one. 

686 stats.sigma = bt_block.judge_sigmas.get(stats.model) 

687 elif aggregation != 'plurality': 

688 msg = f"unknown aggregation {aggregation!r}; expected 'plurality' or 'bt-sigma'" 

689 raise ValueError(msg) 

690 

691 return PairwiseReport( 

692 comparisons=total, 

693 a_win_rate=_rate(winners['A'], decisive), 

694 b_win_rate=_rate(winners['B'], decisive), 

695 tie_rate=winners['tie'] / total if total else 0.0, 

696 inconclusive_rate=winners['inconclusive'] / total if total else 0.0, 

697 mean_agreement=sum(agreements) / len(agreements) if agreements else None, 

698 per_judge=judge_stats, 

699 bt_sigma=bt_block, 

700 )

706def _decisive_value(vote: JuryVote) -> VerdictValue | None: 

707 """The judge's verdict, or None if it failed or abstained. 

708 

709 Raises ``ValueError`` on a decisive verdict outside the pairwise contract 

710 (``'A'``/``'B'``/``'tie'``) — ``run_pairwise`` is a public export, so a 

711 caller-supplied ``judge_fn`` returning e.g. a bool must not leak into the 

712 winner as ``'True'``. 

713 """ 

714 if not vote.success or vote.abstained: 

715 return None 

716 if vote.value not in _PAIRWISE_VALUES: 

717 raise ValueError(f"pairwise judge returned {vote.value!r}; expected 'A', 'B', or 'tie'") 

718 return vote.value

721def _reconciled_explanation( 

722 value: VerdictValue | None, first_vote: JuryVote | None, second_vote: JuryVote | None 

723) -> str: 

724 """Explanation for a reconciled vote, from whichever ordering yielded ``value``.""" 

725 if value is None: 

726 return '' 

727 if first_vote is not None and _decisive_value(first_vote) == value and first_vote.explanation: 

728 return first_vote.explanation 

729 if second_vote is not None and _unswap(_decisive_value(second_vote)) == value and second_vote.explanation: 

730 return second_vote.explanation 

731 return (first_vote.explanation if first_vote else '') or (second_vote.explanation if second_vote else '')

734async def run_pairwise( 

735 *, 

736 judge_fn: PairwiseJudgeFn, 

737 panel: Sequence[str], 

738 response_a: Any, 

739 response_b: Any, 

740 swap: bool = True, 

741 repetitions: int = 1, 

742 replacement_judges: Sequence[str] | None = None, 

743 min_successful_judges: int = 1, 

744 propagate_errors: bool = False, 

745 max_concurrency: int | asyncio.Semaphore | None = None, 

746) -> PairwiseComparison: 

747 """Run a panel over one A-vs-B comparison and reconcile it into a verdict. 

748 

749 Each judge runs through the shared `run_jury`. When ``swap`` is on 

750 (default) every judge is also run with A and B exchanged; the two verdicts 

751 are un-swapped and passed through `reconcile_pair`, so a judge that 

752 just follows slot order abstains and is recorded as a flip. 

753 

754 Both orderings run concurrently. Replacement judges are promoted at the pair 

755 level — a stand-in is run in *both* orderings so it can cast a real 

756 reconciled vote, and a primary that fails in either ordering is what it 

757 stands in for. The winner is the plurality of the reconciled votes, or 

758 ``'inconclusive'`` when fewer than ``min_successful_judges`` cast a decisive 

759 reconciled vote. 

760 

761 ``max_concurrency`` caps how many judge calls run at once across the whole 

762 comparison (judges x orderings x repetitions, replacements included). Pass 

763 an existing ``asyncio.Semaphore`` to share one budget across several 

764 comparisons. ``None`` (default) keeps the fan-out unbounded. 

765 

766 Tracing: the whole comparison is ONE ``orq.pairwise_jury`` span (RES-985). Each 

767 ordering drives ``_run_jury_core`` rather than ``run_jury`` so it doesn't 

768 mint a second jury span whose aggregates describe half a comparison; every 

769 judge span hangs off this one, tagged ``judge.label_swapped``. 

770 """ 

771 # Normalized once so both orderings (and the replacement pass) draw from 

772 # the same budget rather than each minting their own. 

773 semaphore = as_semaphore(max_concurrency) 

774 resolved_panel = resolve_panel(panel) 

775 jury_ctx: object | None = None 

776 

777 async def _ordering( 

778 models: Sequence[str], *, swapped: bool, replacement: bool 

779 ) -> tuple[dict[str, JuryVote], TokenUsage | None]: 

780 async def _fn(model: str) -> Prediction: 

781 return await ( 

782 judge_fn(response_b, response_a, model) if swapped else judge_fn(response_a, response_b, model) 

783 ) 

784 

785 # _run_jury_core, not run_jury: this comparison owns ONE orq.pairwise_jury span 

786 # across both orderings, and run_jury would open a second one per 

787 # ordering. Everything below the span (fan-out, per-judge spans, 

788 # repetition collapse) is identical. 

789 # 

790 # Replacements are handled here at the pair level, not inside the core, 

791 # so a stand-in gets a fair shot in both orderings rather than being 

792 # promoted independently per ordering (and then silently dropped in 

793 # reconciliation). 

794 deliberation = await _run_jury_core( 

795 judge_fn=_fn, 

796 panel=models, 

797 repetitions=repetitions, 

798 replacement_judges=None, 

799 min_successful_judges=1, 

800 propagate_errors=propagate_errors, 

801 max_concurrency=semaphore, 

802 label_swapped=swapped, 

803 parent_context=jury_ctx, 

804 replacement=replacement, 

805 ) 

806 return {v.model: v for v in deliberation.jury.votes}, deliberation.token_usage 

807 

808 async def _both( 

809 models: Sequence[str], *, replacement: bool = False 

810 ) -> tuple[dict[str, JuryVote], dict[str, JuryVote], list[TokenUsage]]: 

811 if not swap: 

812 first_votes, first_usage = await _ordering(models, swapped=False, replacement=replacement) 

813 return first_votes, {}, [u for u in (first_usage,) if u] 

814 (first_votes, first_usage), (second_votes, second_usage) = await asyncio.gather( 

815 _ordering(models, swapped=False, replacement=replacement), 

816 _ordering(models, swapped=True, replacement=replacement), 

817 ) 

818 return first_votes, second_votes, [u for u in (first_usage, second_usage) if u] 

819 

820 async with with_span('orq.pairwise_jury') as jury_span: 

821 # Captured before any judge runs so every judge span (both orderings, 

822 # replacements included) parents to this one comparison-level span. 

823 jury_ctx = current_otel_context() 

824 first_votes, second_votes, usages = await _both(resolved_panel) 

825 

826 def _failed(model: str) -> bool: 

827 first = first_votes.get(model) 

828 if first is None or not first.success: 

829 return True 

830 if swap: 

831 second = second_votes.get(model) 

832 return second is None or not second.success 

833 return False 

834 

835 num_failed = sum(1 for model in resolved_panel if _failed(model)) 

836 seen = set(resolved_panel) 

837 stand_ins: list[str] = [] 

838 for candidate in replacement_judges or []: 

839 if candidate and candidate not in seen: 

840 stand_ins.append(candidate) 

841 seen.add(candidate) 

842 stand_ins = stand_ins[:num_failed] 

843 

844 stand_in_set = set(stand_ins) 

845 if stand_ins: 

846 rep_first, rep_second, rep_usages = await _both(stand_ins, replacement=True) 

847 first_votes.update(rep_first) 

848 second_votes.update(rep_second) 

849 usages.extend(rep_usages)

851 def _observations(model: str) -> tuple[list[RepetitionObservation], int]: 

852 """Canonicalized per-repetition votes from both orderings' JuryVotes. 

853 

854 The jury already retains raw per-repetition verdicts; this keeps 

855 them on the PairwiseVote (swapped-ordering entries un-swapped) so 

856 reliability estimation can use repeated observations instead of 

857 only the collapsed vote (RES-1251).""" 

858 out: list[RepetitionObservation] = [] 

859 failures = 0 

860 for ordering, jv, swapped in ( 

861 ('ab', first_votes.get(model), False), 

862 ('ba', second_votes.get(model), True), 

863 ): 

864 if jv is None: 

865 continue 

866 failures += jv.repetitions_failed 

867 for i, rep in enumerate(jv.repetitions): 

868 value = rep.value if not swapped else _unswap(rep.value) 

869 if value is None: 

870 verdict: Literal['A', 'B', 'tie'] | None = None # genuine abstention: a valid no-decision pass 

871 elif str(value) in _PAIRWISE_VALUES: 

872 verdict = cast("Literal['A', 'B', 'tie']", str(value)) 

873 else: 

874 # Decisive-looking but off-contract (RES-1251 review): a 

875 # malformed pass, not a silently dropped None. Count it 

876 # like an error so it lowers reliability rather than 

877 # passing as a free abstention, and name it (review item 7) 

878 # so a user who sees consistency drop can find the cause. 

879 logger.warning( 

880 'pairwise judge {} returned an off-contract repetition value {!r} on the {} ' 

881 'ordering (repetition {}); counting it as a failure, not an abstention', 

882 model, 

883 value, 

884 ordering, 

885 i, 

886 ) 

887 verdict = None 

888 failures += 1 

889 out.append( 

890 RepetitionObservation( 

891 ordering=cast("Literal['ab', 'ba']", ordering), 

892 repetition=i, 

893 verdict=verdict, 

894 explanation=rep.explanation, 

895 ) 

896 ) 

897 return out, failures 

898 

899 votes: list[PairwiseVote] = [] 

900 for model in (*resolved_panel, *stand_ins): 

901 first_vote = first_votes.get(model) 

902 second_vote = second_votes.get(model) 

903 first_value = _decisive_value(first_vote) if first_vote else None 

904 if swap: 

905 second_value = _unswap(_decisive_value(second_vote)) if second_vote else None 

906 vote, flipped = reconcile_pair(first_value, second_value) 

907 completed = first_value is not None and second_value is not None 

908 else: 

909 vote, flipped, completed = first_value, False, False 

910 observations, repetition_failures = _observations(model) 

911 votes.append( 

912 PairwiseVote( 

913 model=model, 

914 vote=_as_vote(vote), 

915 flipped=flipped, 

916 completed=completed, 

917 replacement=model in stand_in_set, 

918 explanation=_reconciled_explanation(vote, first_vote, second_vote), 

919 observations=observations, 

920 repetition_failures=repetition_failures, 

921 ) 

922 ) 

923 

924 decisive = [v.vote for v in votes if v.vote is not None] 

925 winner = 'inconclusive' if len(decisive) < max(1, min_successful_judges) else pairwise_consensus(decisive) 

926 comparison = PairwiseComparison(winner=winner, votes=votes, token_usage=_sum_usage(usages)) 

927 _record_pairwise_span( 

928 jury_span, 

929 comparison, 

930 panel_size=len(resolved_panel), 

931 min_successful_judges=min_successful_judges, 

932 swap=swap, 

933 ) 

934 return comparison

937def _as_vote(value: VerdictValue | None) -> Literal['A', 'B', 'tie'] | None: 

938 """Narrow a reconciled verdict to the pairwise vote vocabulary. 

939 

940 ``_decisive_value`` has already rejected anything outside the contract, so 

941 the cast is safe; keeping the check here too guards the type against a 

942 future caller that skips that gate. 

943 """ 

944 if value is None: 

945 return None 

946 s = str(value) 

947 if s not in _PAIRWISE_VALUES: 

948 raise ValueError(f"pairwise vote {s!r} outside the contract; expected 'A', 'B', or 'tie'") 

949 return cast("Literal['A', 'B', 'tie']", s)

952def _record_pairwise_span( 

953 span: Span | None, 

954 comparison: PairwiseComparison, 

955 *, 

956 panel_size: int, 

957 min_successful_judges: int, 

958 swap: bool, 

959) -> None: 

960 """Stamp the comparison-level attributes on the ``orq.pairwise_jury`` span. 

961 

962 Deliberately reuses the ``jury.*`` namespace so one dashboard reads both 

963 modes: ``jury.verdict`` is the winner, ``judges_succeeded`` counts judges 

964 that cast a reconciled vote. ``jury.flipped`` / ``jury.flipped_judges`` are 

965 comparative-only — judges that contradicted themselves across the two 

966 orderings, i.e. position bias. 

967 

968 ``jury.judges_failed`` counts only judges with NO reconciled vote and no 

969 flip: a flipped judge answered both times, so that is position bias, not a 

970 failure. Counting it as failed would give one attribute two different 

971 meanings across the two modes. 

972 """ 

973 flipped = [v.model for v in comparison.votes if v.flipped] 

974 decisive = [v.vote for v in comparison.votes if v.vote is not None] 

975 set_span_attrs( 

976 span, 

977 { 

978 'jury.verdict': comparison.winner, 

979 'jury.aggregator': 'pairwise_plurality', 

980 'jury.min_successful_judges': min_successful_judges, 

981 'jury.raw_agreement': _agreement_rate(decisive), 

982 'jury.judges_configured': panel_size, 

983 'jury.judges_succeeded': len(decisive), 

984 'jury.judges_failed': sum( 

985 1 for v in comparison.votes if not v.replacement and v.vote is None and not v.flipped 

986 ), 

987 'jury.replacements_used': sum(1 for v in comparison.votes if v.replacement), 

988 'jury.inconclusive': comparison.winner == 'inconclusive', 

989 'jury.flipped': len(flipped), 

990 # Which judges, not just how many: a per-judge attribute is impossible 

991 # (a judge span closes before the other ordering reconciles against it). 

992 'jury.flipped_judges': ','.join(sorted(flipped)) or None, 

993 'jury.swap': swap, 

994 }, 

995 )
