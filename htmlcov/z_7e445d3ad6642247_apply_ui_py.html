1"""Apply-recommendations UI for the dashboard (RES-1143). 

2 

3Serves both surfaces on top of the shared engine in 

4:mod:`evaluatorq.common.apply`: red team surfaces the report's 

5``focus_area_recommendations`` in the Focus areas tab, agent simulation 

6surfaces the run's ``recommendations`` in its Recommendations tab. Same bar, 

7drawer, and confirm flow; only the routes, the write-back field 

8(``applied_recommendations`` vs ``applied_suggestions``), and the breakdown 

9block differ. The red-team flow: 

10 

111. The apply panel (rendered into the Focus areas tab) shows how many 

12 recommendations are pending vs already applied, an agent picker when the 

13 run tested several agents, and a "Preview changes" button. 

142. Preview POSTs to ``/r/{rid}/redteam/apply/preview``, which runs 

15 ``apply_recommendations(apply=False)`` — the agent is only READ, an LLM 

16 folds the pending recommendations into its instructions, and the result 

17 (recommendation list + unified diff) renders into a right-hand drawer. 

183. The drawer's "Apply to agent" button POSTs the previewed instructions to 

19 ``/r/{rid}/redteam/apply/confirm``, which writes them back as a new minor 

20 agent version and records the applied recommendations on the report JSON 

21 (``applied_recommendations``), so a later preview skips them. 

22 

23Nothing is written to the platform until the user clicks Apply in the drawer. 

24The confirm step reuses the previewed instructions verbatim — no second LLM 

25call, what you saw is what is written. 

26 

27Public entry points: 

28 render_apply_panel(rid, report) — red-team panel + drawer mount

58# One process-wide CSRF token, minted into every apply form. A cross-origin page 

59# cannot read this dashboard's HTML, so it cannot supply the token - which closes 

60# the form-POST CSRF hole on the dashboard's state-changing routes (review). 

61CSRF_FIELD = 'csrf' 

62_CSRF_TOKEN = secrets.token_urlsafe(32) 

63 

64# Server-side previews keyed by a SINGLE-USE token. Confirm accepts only the 

65# token, so the write is exactly what this server previewed - "what you saw is 

66# what is written" becomes an invariant instead of a UI convention (review). 

67_PREVIEWS: dict[str, dict[str, Any]] = {} 

68_PREVIEWS_LOCK = threading.Lock() 

69_PREVIEWS_CAP = 32 # FIFO evict: a dashboard session never has this many live drawers 

70 

71 

72def _store_preview(entry: dict[str, Any]) -> str: 

73 token = secrets.token_urlsafe(16) 

74 with _PREVIEWS_LOCK: 

75 while len(_PREVIEWS) >= _PREVIEWS_CAP: 

76 _PREVIEWS.pop(next(iter(_PREVIEWS))) 

77 _PREVIEWS[token] = entry 

78 return token 

79 

80 

81def _pop_preview(token: str) -> dict[str, Any] | None: 

82 with _PREVIEWS_LOCK: 

83 return _PREVIEWS.pop(token, None)

86def _csrf_field() -> str: 

87 return f'<input type="hidden" name="{CSRF_FIELD}" value="{_CSRF_TOKEN}">'

90def _request_rejected(req: Request, form: Any) -> str | None: 

91 """CSRF/origin gate for the apply routes. Returns an error message or None. 

92 

93 Two independent checks: the form must echo the per-process token (unreadable 

94 cross-origin), and when the browser sends Sec-Fetch-Site it must be a 

95 same-origin (or direct) request. Absent headers pass - test clients and 

96 older browsers do not send them; the token check still holds then.""" 

97 sec_fetch = req.headers.get('sec-fetch-site', '') 

98 if sec_fetch and sec_fetch not in ('same-origin', 'none'): 

99 return 'Cross-origin request rejected.' 

100 if str(form.get(CSRF_FIELD) or '') != _CSRF_TOKEN: 

101 return 'Stale or missing form token; reload the page and try again.' 

102 return None

278def _diff_html(diff: str) -> str: 

279 """Colorized unified diff, line by line.""" 

280 out: list[str] = [] 

281 for line in diff.splitlines(): 

282 cls = 'ctx' 

283 if line.startswith('+') and not line.startswith('+++'): 

284 cls = 'add' 

285 elif line.startswith('-') and not line.startswith('---'): 

286 cls = 'del' 

287 elif line.startswith('@@'): 

288 cls = 'hunk' 

289 elif line.startswith(('+++', '---')): 

290 cls = 'file' 

291 out.append(f'<span class="rt-diff-line rt-diff-{cls}">{esc(line)}</span>') 

292 return f'<pre class="rt-diff">{"".join(out)}</pre>'

425# ponytail: process-global lock. The read-modify-write below is not atomic, so 

426# two concurrent confirms could each read the old list and drop one another's 

427# bookkeeping. One lock serializes all report writes in this process, which is 

428# all a single dashboard needs; move to per-path locks only if it contends. 

429_RECORD_LOCK = threading.Lock() 

430 

431 

432def record_applied_on_report(path: Path, recommendations: list[str], field: str = 'applied_recommendations') -> None: 

433 """Append newly applied recommendations to the report JSON, atomically. 

434 

435 *field* is the surface's tracking field (``applied_recommendations`` for 

436 red team, ``applied_suggestions`` for simulation). Deduplicates against 

437 what is already recorded so a double-click cannot inflate the list. Blocking 

438 file IO guarded by a process lock; call it off the event loop (``to_thread``). 

439 """ 

440 with _RECORD_LOCK: 

441 raw = json.loads(path.read_text(encoding='utf-8')) 

442 existing = raw.get(field) or [] 

443 seen = {str(r).strip() for r in existing} 

444 for rec in recommendations: 

445 if rec.strip() not in seen: 

446 existing.append(rec) 

447 seen.add(rec.strip()) 

448 raw[field] = existing 

449 # Unique temp name so concurrent writers never share (and clobber) one 

450 # scratch file; os.replace is atomic on the same filesystem. 

451 tmp = path.with_suffix(path.suffix + f'.tmp.{os.getpid()}.{threading.get_ident()}') 

452 tmp.write_text(json.dumps(raw, indent=2, default=str), encoding='utf-8') 

453 tmp.replace(path)

492async def _confirm_response( 

493 rid: str, 

494 req: Request, 

495 roots: list[Any] | None, 

496 *, 

497 surface: str, 

498 field: str, 

499 version_note: str, 

500 expected_agent: str | None, 

501 already_applied: list[str], 

502) -> Response: 

503 """Shared confirm handler: write a SERVER-STORED preview to the agent and 

504 record the applied recommendations on the report JSON under *field*. 

505 

506 The request carries only a single-use token (plus the CSRF field); the 

507 instructions and recommendations come from this server's preview store, so 

508 a hand-crafted POST cannot choose what gets written. The stored preview is 

509 validated against the reloaded report (*expected_agent*, *already_applied*) 

510 and against the agent's CURRENT instructions - if they changed since the 

511 preview, the write is refused instead of silently clobbering the change. 

512 """ 

513 from evaluatorq.common.apply import read_instructions, write_instructions 

514 from evaluatorq.dashboard import library 

515 

516 form = await req.form() 

517 rejected = _request_rejected(req, form) 

518 if rejected: 

519 return Response(render_error_drawer(rejected), media_type='text/html') 

520 

521 token = str(form.get('confirm_token') or '') 

522 entry = _pop_preview(token) if token else None 

523 if entry is None: 

524 return Response( 

525 render_error_drawer('This preview has expired or was already applied; run the preview again.'), 

526 media_type='text/html', 

527 ) 

528 if entry['rid'] != rid or entry['surface'] != surface: 

529 return Response( 

530 render_error_drawer('This preview belongs to a different report; run the preview again.'), 

531 media_type='text/html', 

532 ) 

533 agent_key = str(entry['agent_key']) 

534 new_instructions = str(entry['new_instructions']) 

535 recommendations = [str(r) for r in entry['recommendations']] 

536 

537 # Validate against the reloaded report, not the drawer: a preview built when 

538 # the report said something else must not write now. 

539 if expected_agent and agent_key != expected_agent: 

540 return Response( 

541 render_error_drawer('This preview was for a different agent; reload the page and preview again.'), 

542 media_type='text/html', 

543 ) 

544 seen = {r.strip() for r in already_applied} 

545 recommendations = [r for r in recommendations if r.strip() not in seen] 

546 if not recommendations: 

547 return Response( 

548 render_error_drawer('These recommendations were already applied; reload the page.'), 

549 media_type='text/html', 

550 ) 

551 

552 try: 

553 orq_client, _llm_client, _model = _build_clients() 

554 except ValueError as e: 

555 return Response(render_error_drawer(str(e)), media_type='text/html') 

556 

557 # Lost-update guard: the preview merged into the instructions as they were 

558 # at preview time. If the agent changed since, refuse rather than clobber. 

559 try: 

560 current = await read_instructions(orq_client, agent_key) 

561 except Exception as e: 

562 logger.opt(exception=True).warning('apply_ui: pre-write read failed for {}', agent_key) 

563 return Response(render_error_drawer(f'Could not re-read the agent before writing: {e}'), media_type='text/html') 

564 if current.strip() != str(entry['original_instructions']).strip(): 

565 return Response( 

566 render_error_drawer( 

567 'The agent instructions changed after this preview was made; run the preview again ' 

568 'so the diff reflects the current instructions.' 

569 ), 

570 media_type='text/html', 

571 ) 

572 

573 try: 

574 new_version = await write_instructions( 

575 orq_client, 

576 agent_key, 

577 new_instructions, 

578 version_description=f'Applied {len(recommendations)} {version_note}', 

579 ) 

580 except Exception as e: 

581 logger.opt(exception=True).warning('apply_ui: agent update failed for {}', agent_key) 

582 return Response(render_error_drawer(f'Agent update failed: {e}'), media_type='text/html') 

583 

584 # Record on the report so a later preview skips these. A write-back 

585 # failure must not hide that the agent WAS updated - report it as a 

586 # warning inside the success drawer instead of failing the request. The 

587 # note names the actual consequence (review): unrecorded bullets come back 

588 # as pending, and applying again folds them in a second time. 

589 unrecorded = ' They will show as pending again, and applying them again would merge them into the agent twice.' 

590 path = library.resolve(rid, roots) 

591 record_note = '' 

592 if path is not None: 

593 try: 

594 await asyncio.to_thread(record_applied_on_report, path, recommendations, field) 

595 except Exception: 

596 logger.opt(exception=True).warning('apply_ui: could not record applied recs on {}', path) 

597 record_note = f' The agent was updated, but the report file could not be written.{unrecorded}' 

598 else: 

599 logger.warning('apply_ui: could not resolve report path for {}; applied recs not recorded', rid) 

600 record_note = f' The agent was updated, but the report file could not be located.{unrecorded}' 

601 

602 html = render_applied_drawer(agent_key, len(recommendations), new_version) 

603 if record_note: 

604 html = html.replace('</p>', esc(record_note) + '</p>', 1) 

605 return Response(html, media_type='text/html')

608def _rt_narrow(form: Any, report: RedTeamReport) -> tuple[list[Any], FocusAreaRecommendation | None, str, str | None]: 

609 """Narrow the red-team preview to a single bullet when `rec` is posted. 

610 

611 Returns ``(items, area, breakdown, error)``: the focus areas to merge, the 

612 single area for the drawer breakdown (None for apply-all), an empty 

613 breakdown (red team uses *area* instead), and an error message or None. 

614 """ 

615 areas = list(report.focus_area_recommendations or []) 

616 single_rec = str(form.get('rec') or '').strip() 

617 if not single_rec: 

618 return areas, None, '', None 

619 category = str(form.get('category') or '').strip() 

620 area = next( 

621 (a for a in areas if a.category == category and single_rec in [r.strip() for r in a.recommendations]), 

622 None, 

623 ) 

624 if area is None: 

625 return [], None, '', 'That recommendation is no longer on the report; reload the page.' 

626 if single_rec in {r.strip() for r in report.applied_recommendations}: 

627 return [], None, '', 'That recommendation is already applied to the agent.' 

628 return [area.model_copy(update={'recommendations': [single_rec]})], area, '', None

631def _sim_narrow(form: Any, run: SimulationRun) -> tuple[list[Any], None, str, str | None]: 

632 """Narrow the simulation preview to a single suggestion when `rec` is posted. 

633 

634 Returns ``(items, None, breakdown, error)``: the recommendations to merge, 

635 no focus area (sim renders its own *breakdown* block), and an error or None. 

636 """ 

637 recs = list(run.recommendations) 

638 single_rec = str(form.get('rec') or '').strip() 

639 if not single_rec: 

640 return recs, None, '', None 

641 try: 

642 idx = int(str(form.get('result_index') or '')) 

643 except ValueError: 

644 idx = -1 

645 card = next( 

646 (r for r in recs if r.result_index == idx and single_rec in [s.strip() for s in r.suggestions]), 

647 None, 

648 ) 

649 if card is None: 

650 return [], None, '', 'That suggestion is no longer on the report; reload the page.' 

651 if single_rec in {s.strip() for s in run.applied_suggestions}: 

652 return [], None, '', 'That suggestion is already applied to the agent.' 

653 return [card.model_copy(update={'suggestions': [single_rec]})], None, sim_breakdown_html(card), None

656async def _preview_response( 

657 rid: str, 

658 req: Request, 

659 *, 

660 surface: str, 

661 obj: Any, 

662 not_found_html: str, 

663 enable_error: str | None, 

664 agent_default: str, 

665 narrow: Any, 

666 apply_fn: Any, 

667 already_applied: list[str], 

668 empty_msg: str, 

669) -> Response: 

670 """Shared preview handler: resolve the agent, narrow to the requested 

671 bullet(s), run ``apply(apply=False)``, and render the drawer. Only the 

672 loader, enable gate, narrowing, and apply wrapper differ between surfaces. 

673 """ 

674 if obj is None: 

675 return Response(not_found_html, status_code=404, media_type='text/html') 

676 if enable_error: 

677 return Response(render_error_drawer(enable_error), media_type='text/html') 

678 

679 form = await req.form() 

680 rejected = _request_rejected(req, form) 

681 if rejected: 

682 return Response(render_error_drawer(rejected), media_type='text/html') 

683 agent_key = str(form.get('agent_key') or '').strip() or agent_default 

684 if not agent_key: 

685 return Response( 

686 render_error_drawer('This run does not record which agent it tested, so there is nothing to apply to.'), 

687 media_type='text/html', 

688 ) 

689 

690 items, area, breakdown, narrow_error = narrow(form, obj) 

691 if narrow_error: 

692 return Response(render_error_drawer(narrow_error), media_type='text/html') 

693 

694 try: 

695 orq_client, llm_client, model = _build_clients() 

696 except ValueError as e: 

697 return Response(render_error_drawer(str(e)), media_type='text/html') 

698 

699 try: 

700 result = await apply_fn(items, agent_key, orq_client, llm_client, model, already_applied) 

701 except Exception as e: 

702 logger.opt(exception=True).warning('apply_ui: preview failed for {}', agent_key) 

703 return Response(render_error_drawer(f'Preview failed: {e}'), media_type='text/html') 

704 

705 if result.merge_failed: 

706 return Response( 

707 render_error_drawer('The merge did not produce usable instructions; please try the preview again.'), 

708 media_type='text/html', 

709 ) 

710 if not result.recommendations: 

711 return Response(render_error_drawer(empty_msg), media_type='text/html') 

712 token = _store_preview({ 

713 'rid': rid, 

714 'surface': surface, 

715 'agent_key': result.agent_key, 

716 'original_instructions': result.original_instructions, 

717 'new_instructions': result.new_instructions, 

718 'recommendations': list(result.recommendations), 

719 }) 

720 return Response( 

721 render_preview_drawer(rid, result, area, surface=surface, breakdown=breakdown, confirm_token=token), 

722 media_type='text/html', 

723 )
