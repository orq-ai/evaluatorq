41def _lc_content_to_text(content: Any) -> str: 

42 """Extract the text of a LangChain message's ``content``. 

43 

44 ``content`` is a plain string on OpenAI-backed models but a list of typed 

45 blocks on Anthropic-backed ones — even for a turn that is purely text. 

46 ``str()`` on that list yields its Python repr (``"[{'type': 'text', ...}]"``), 

47 which then flows into transcripts, judges and reports looking like the 

48 model's answer. Corrupted text is worse than missing text: a judge scores it 

49 and nothing appears broken. 

50 """ 

51 if isinstance(content, str): 

52 return content 

53 if not isinstance(content, list): 

54 return tool_result_to_text(content) 

55 parts: list[str] = [] 

56 for block in content: 

57 if isinstance(block, str): 

58 parts.append(block) 

59 elif isinstance(block, dict): 

60 kind = block.get('type') 

61 if kind == 'text': 

62 parts.append(str(block.get('text') or '')) 

63 elif kind not in _NON_TEXT_BLOCKS_HANDLED_ELSEWHERE: 

64 logger.warning('LangGraphTarget: dropping unrenderable content block type=%r', kind) 

65 return ''.join(parts)

68class _TokenUsageCollector(BaseCallbackHandler): 

69 """Sync LangChain callback handler that accumulates token usage across LLM calls. 

70 

71 Subclasses sync ``BaseCallbackHandler`` (not ``AsyncCallbackHandler``) so it 

72 is invoked correctly from both ``invoke`` and ``ainvoke`` call paths. 

73 """ 

74 

75 def __init__(self) -> None: 

76 super().__init__() 

77 self._usage: TokenUsage = TokenUsage() 

78 

79 def on_llm_end(self, response: LLMResult, **kwargs: Any) -> None: 

80 """Accumulate token usage from a completed LLM call.""" 

81 for inner in response.generations: 

82 try: 

83 if not inner: 

84 continue 

85 # Only read first candidate — higher-n sampling reuses the same 

86 # underlying API call; iterating all candidates would multiply-count. 

87 gen = inner[0] 

88 meta = getattr(getattr(gen, 'message', None), 'usage_metadata', None) 

89 if meta is None: 

90 # Provider/integration didn't surface usage metadata on this 

91 # generation. Real call still happened — log so cost dashboards 

92 # showing zeros are diagnosable. Once per generation is fine 

93 # since LangGraph chains rarely emit hundreds. 

94 logger.debug( 

95 '_TokenUsageCollector: generation has no usage_metadata; ' 

96 'tokens for this call will not be captured' 

97 ) 

98 continue 

99 # Shared extractor carries cached/reasoning from input_token_details / 

100 # output_token_details and applies the unified total fallback. 

101 delta = TokenUsage.extract(meta, calls=1) 

102 if delta is not None: 

103 self._usage = self._usage + delta 

104 except Exception as exc: 

105 logger.warning('_TokenUsageCollector.on_llm_end: failed to extract usage: %s', exc) 

106 

107 def on_llm_error(self, error: BaseException, **kwargs: Any) -> None: 

108 """Handle LLM errors — try to extract partial usage, otherwise no-op.""" 

109 try: 

110 # Some providers attach a partial LLMResult / usage on the exception. 

111 response = getattr(error, 'response', None) 

112 if response is not None: 

113 partial_result = getattr(response, 'llm_result', None) or getattr(response, 'llm_output', None) 

114 if isinstance(partial_result, LLMResult): 

115 self.on_llm_end(partial_result) 

116 except Exception as exc: 

117 logger.warning('_TokenUsageCollector.on_llm_error: failed to extract partial usage: %s', exc) 

118 

119 def to_token_usage(self) -> TokenUsage | None: 

120 """Return aggregated usage, or None if no calls were recorded.""" 

121 if self._usage.calls == 0: 

122 return None 

123 return self._usage

126class LangGraphTarget(AgentTarget): 

127 """Wraps a LangGraph CompiledStateGraph as an AgentTarget. 

128 

129 Each instance generates its own ``memory_entity_id`` used as the LangGraph 

130 ``thread_id`` — this is the checkpointer's isolation key, so parallel 

131 attacks never share thread state. The pipeline reads ``memory_entity_id`` 

132 off the target rather than injecting it. 

133 

134 Usage: 

135 

136 ```python 

137 # NOTE: langgraph < 2.0 path. create_react_agent moved to 

138 # `langchain.agents.create_agent` in langgraph V1.0 and is removed in V2.0. 

139 # LangGraphTarget wraps whatever compiled graph you pass, so no change is 

140 # needed here — only update this import when you bump to langgraph 2.x. 

141 from langgraph.prebuilt import create_react_agent 

142 from evaluatorq.integrations.langgraph_integration import LangGraphTarget 

143 

144 graph = create_react_agent(model=model, tools=[...]) 

145 target = LangGraphTarget(graph=graph) 

146 

147 # Pass to simulation or red teaming 

148 results = await simulate(target=target, ...) 

149 report = await red_team(target=target) 

150 ``` 

151 """ 

152 

153 def __init__( 

154 self, 

155 graph: CompiledStateGraph[Any, Any, Any, Any], 

156 *, 

157 config: dict[str, Any] | None = None, 

158 agent_context: AgentContext | None = None, 

159 ) -> None: 

160 """Create a LangGraph agent target. 

161 

162 Args: 

163 graph: A compiled LangGraph state graph. 

164 config: Optional extra LangGraph RunnableConfig keys 

165 (e.g. ``{"recursion_limit": 50}``). The ``thread_id`` 

166 is managed automatically — do not pass it here. 

167 agent_context: Optional `AgentContext` override. When 

168 provided, this context is returned from 

169 `get_agent_context` verbatim. When omitted, the target 

170 introspects the compiled graph (tools from a ``ToolNode``, 

171 checkpointer presence) on a best-effort basis. 

172 """ 

173 super().__init__(memory_entity_id=uuid4().hex) 

174 self._graph = graph 

175 self._extra_config = config or {} 

176 self._agent_context = agent_context 

177 # Tracks how many messages were in the LangGraph thread before this turn. 

178 # Not safe for concurrent respond() calls on the same instance — 

179 # use .new() to get independent instances for parallel use. 

180 self._prev_msg_count: int = 0 

181 graph_name: str = getattr(graph, 'name', None) or 'langgraph_target' 

182 self._key = f'{graph_name}_{uuid4().hex[:8]}' 

183 # Guard against spamming the log on every respond() call when 

184 # an unknown callbacks type is encountered (hot path). 

185 self._warned_unknown_callbacks: bool = False 

186 

187 def _build_config(self) -> RunnableConfig: 

188 """Build the RunnableConfig with the current thread_id.""" 

189 extra = {k: v for k, v in self._extra_config.items() if k != 'configurable'} 

190 return RunnableConfig( 

191 **extra, 

192 configurable={ 

193 **self._extra_config.get('configurable', {}), 

194 'thread_id': self.memory_entity_id, 

195 }, 

196 )

198 async def respond(self, messages: list[Message]) -> AgentResponse: 

199 """Send the last user message to the LangGraph agent; return its response. 

200 

201 LangGraph owns thread state (keyed by ``thread_id``/``memory_entity_id``), 

202 so ``respond`` forwards only the latest user turn. Token usage is 

203 collected via a per-call ``_TokenUsageCollector`` callback, drained in a 

204 ``finally:`` block so partial spend on error paths is preserved. 

205 """ 

206 if not messages or messages[-1].role != 'user': 

207 raise ValueError("LangGraphTarget.respond requires messages[-1].role == 'user'") 

208 prompt = content_to_text(messages[-1].content) 

209 collector = _TokenUsageCollector() 

210 

211 base_config = self._build_config() 

212 existing = base_config.get('callbacks') 

213 

214 if existing is None: 

215 new_callbacks: Any = [collector] 

216 elif isinstance(existing, list): 

217 new_callbacks = [*existing, collector] 

218 elif isinstance(existing, BaseCallbackManager) and hasattr(existing, 'copy'): 

219 # Copy before mutating — avoid accumulating stale collectors on the 

220 # original manager across repeated respond() calls and .new() clones. 

221 manager_copy = existing.copy() 

222 manager_copy.add_handler(collector, inherit=True) 

223 new_callbacks = manager_copy 

224 else: 

225 # Unknown type — wrap alongside existing without mutating. 

226 if not self._warned_unknown_callbacks: 

227 logger.warning( 

228 'LangGraphTarget: unrecognised callbacks type %s; wrapping in list. ' 

229 'Pass a list or BaseCallbackManager instead.', 

230 type(existing).__name__, 

231 ) 

232 self._warned_unknown_callbacks = True 

233 new_callbacks = [existing, collector] 

234 

235 new_config: RunnableConfig = {**base_config, 'callbacks': new_callbacks} 

236 

237 prev_count = self._prev_msg_count 

238 usage: TokenUsage | None = None 

239 try: 

240 result = await self._graph.ainvoke( 

241 {'messages': [{'role': 'user', 'content': prompt}]}, 

242 config=new_config, 

243 ) 

244 except Exception: 

245 self._prev_msg_count = prev_count # cursor is stable on failure 

246 raise 

247 finally: 

248 usage = collector.to_token_usage() 

249 

250 result_messages = result.get('messages') 

251 if result_messages is None: 

252 raise ValueError( 

253 "LangGraphTarget requires a graph whose state has a 'messages' key " 

254 '(e.g. built with MessagesState). Got state keys: ' + str(list(result.keys())) 

255 ) 

256 if not result_messages: 

257 raise ValueError( 

258 "LangGraphTarget: graph returned an empty 'messages' list. " 

259 'Ensure every execution path appends at least one AI message.' 

260 ) 

261 # Build output items directly from messages added in this turn, preserving 

262 # interleaving of text and tool calls (ReAct-style: text -> tool_call -> text). 

263 # LangGraph checkpointer returns the full accumulated thread state, so 

264 # slicing from _prev_msg_count avoids duplicating tool calls across turns. 

265 # Build ToolCallOutputItem directly (not via the .tool_calls view) so 

266 # interleaved text/tool ordering is preserved in .output. 

267 output_items: list[OutputMessage] = [] 

268 sliced = result_messages[self._prev_msg_count :] 

269 if not sliced and self._prev_msg_count > 0: 

270 # Graph may use a trimming reducer; fall back to the full result set 

271 # so the caller receives the current turn's response, not silence. 

272 logger.warning( 

273 'LangGraphTarget: _prev_msg_count (%d) exceeds result len (%d); ' 

274 'graph may be trimming messages. Falling back to full result.', 

275 self._prev_msg_count, 

276 len(result_messages), 

277 ) 

278 sliced = result_messages 

279 for msg in sliced: 

280 if isinstance(msg, dict): 

281 if msg.get('role') != 'assistant': 

282 continue 

283 msg_content = msg.get('content', '') 

284 tool_calls_iter = msg.get('tool_calls') or [] 

285 else: 

286 if not isinstance(msg, AIMessage): 

287 continue 

288 msg_content = getattr(msg, 'content', '') 

289 tool_calls_iter = getattr(msg, 'tool_calls', None) or [] 

290 

291 msg_content = _lc_content_to_text(msg_content) 

292 if msg_content: 

293 output_items.append(TextOutputItem(text=msg_content, annotations=[])) 

294 

295 for tc in tool_calls_iter: 

296 call_id = tc.get('id', '') if isinstance(tc, dict) else getattr(tc, 'id', '') 

297 name = tc.get('name', '') if isinstance(tc, dict) else getattr(tc, 'name', '') 

298 args = tc.get('args', {}) if isinstance(tc, dict) else getattr(tc, 'args', {}) 

299 args_str = json.dumps(args if isinstance(args, dict) else {}, default=str) 

300 output_items.append( 

301 ToolCallOutputItem(name=str(name), arguments=args_str, id=call_id, call_id=call_id) 

302 if call_id 

303 else ToolCallOutputItem(name=str(name), arguments=args_str) 

304 ) 

305 

306 self._prev_msg_count = len(result_messages) 

307 

308 # Fallback: if no AIMessage text was emitted (e.g. duck-typed message objects 

309 # that don't subclass AIMessage), use the last message's .content so that 

310 # AgentResponse.text remains well-defined. 

311 if not any(isinstance(item, TextOutputItem) for item in output_items): 

312 last = result_messages[-1] 

313 logger.warning( 

314 'LangGraphTarget: no AIMessage text in turn; falling back to last message content (type=%s)', 

315 type(last).__name__, 

316 ) 

317 last_content = last.get('content', '') if isinstance(last, dict) else getattr(last, 'content', '') 

318 output_items.append(TextOutputItem(text=_lc_content_to_text(last_content), annotations=[])) 

319 return AgentResponse(output=output_items, usage=usage)

330 async def get_agent_context(self) -> AgentContext: 

331 """Return agent context introspected from the compiled graph. 

332 

333 If an ``agent_context`` was passed at construction time, returns it 

334 unchanged. Otherwise performs best-effort introspection: 

335 

336 * tools: extracted from any node whose ``bound`` is a ``ToolNode`` via 

337 ``tools_by_name`` 

338 * memory_stores: single synthetic entry pointing at the checkpointer 

339 thread when a checkpointer is attached 

340 

341 Callers needing stronger guarantees (custom graph shapes, non-standard 

342 tool nodes) should pass ``agent_context`` explicitly. 

343 """ 

344 if self._agent_context is not None: 

345 return self._agent_context 

346 

347 tools = _introspect_tools(self._graph) 

348 memory_stores = _introspect_memory_stores(self._graph, self.memory_entity_id) 

349 

350 return AgentContext( 

351 key=self._key, 

352 description='LangGraph compiled state graph target', 

353 tools=tools, 

354 memory_stores=memory_stores, 

355 )

370def _introspect_tools(graph: CompiledStateGraph[Any, Any, Any, Any]) -> list[ToolInfo]: 

371 """Best-effort extraction of tool metadata from a compiled graph. 

372 

373 Looks for nodes whose ``bound`` attribute exposes ``tools_by_name`` 

374 (LangGraph's ``ToolNode``). Falls back to an empty list on anything else. 

375 """ 

376 tools: list[ToolInfo] = [] 

377 seen: set[str] = set() 

378 nodes = getattr(graph, 'nodes', None) 

379 if not nodes: 

380 return tools 

381 for node in nodes.values(): 

382 bound = getattr(node, 'bound', None) 

383 tools_by_name = getattr(bound, 'tools_by_name', None) 

384 if not isinstance(tools_by_name, dict): 

385 continue 

386 for name, tool in tools_by_name.items(): 

387 name_str = str(name) 

388 if name_str in seen: 

389 continue 

390 seen.add(name_str) 

391 tools.append( 

392 ToolInfo( 

393 name=name_str, 

394 description=getattr(tool, 'description', None), 

395 parameters=None, 

396 ) 

397 ) 

398 return tools
