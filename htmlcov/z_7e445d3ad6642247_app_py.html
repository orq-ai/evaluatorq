79def _paging(req: Request) -> tuple[int, int]: 

80 """Parse ``?page=`` / ``?per_page=`` into a (page, per_page) pair. Bad input 

81 falls back to page 1 and the default page size; ``per_page`` is allow-listed.""" 

82 try: 

83 page = max(1, int(req.query_params.get('page', 1))) 

84 except (TypeError, ValueError): 

85 page = 1 

86 try: 

87 per_page = int(req.query_params.get('per_page', RUN_PAGE_SIZES[0])) 

88 except (TypeError, ValueError): 

89 per_page = RUN_PAGE_SIZES[0] 

90 if per_page not in RUN_PAGE_SIZES: 

91 per_page = RUN_PAGE_SIZES[0] 

92 return page, per_page

95def _mask_key(value: str) -> str: 

96 """Show only a key's last 4 characters so a user can validate *which* key is 

97 loaded without exposing the secret. Hidden chars render as ``*`` (capped so a 

98 long key doesn't blow out the row). Short keys reveal nothing but length.""" 

99 if len(value) <= 4: 

100 return '*' * len(value) 

101 stars = '*' * min(len(value) - 4, 16) 

102 return f'{stars}{value[-4:]}'

108def _csv_safe(value: object) -> object: 

109 """Neutralise spreadsheet formula injection in a CSV cell. 

110 

111 Excel and Sheets evaluate a cell whose text begins with ``= + - @`` (or a 

112 tab / carriage return), so free text reaching an export can execute on the 

113 machine that opens the file. The pairwise export is the first here to carry 

114 free text — questions and judge model names are model-generated, and this 

115 tool renders adversarial red-team output — so a leading quote makes such a 

116 cell literal text. 

117 

118 Values that only look like formulas are left alone: a number keeps its sign, 

119 and a bare ``-`` is the n/a placeholder the red-team export already writes. 

120 """ 

121 if not isinstance(value, str) or not value.startswith(_CSV_FORMULA_LEAD): 

122 return value 

123 if value == '-': 

124 return value 

125 try: 

126 _ = float(value) 

127 except ValueError: 

128 return f"'{value}" 

129 return value

132def _settings_config(roots: list[Path] | None) -> list[tuple[str, str | list[str]]]: 

133 """Build the read-only runtime config shown on the Settings page: the run 

134 stores being scanned, the default sim model, and API-key presence with a 

135 masked suffix (never the full value).""" 

136 import os 

137 

138 from evaluatorq.dashboard.library import default_roots 

139 from evaluatorq.dashboard.orq_workspace import classify_host, resolve_base_url, resolve_slug 

140 from evaluatorq.simulation.types import DEFAULT_MODEL 

141 

142 scan_roots = roots if roots is not None else default_roots() 

143 store_paths = [str(p) for p in scan_roots] or ['—'] 

144 from evaluatorq.dashboard.apply_ui import APPLY_MODEL_ENV, DEFAULT_APPLY_MODEL, apply_model 

145 

146 config: list[tuple[str, str | list[str]]] = [('Run stores', store_paths)]

333 # Route: POST /r/{rid}/filter — HTMX filter round-trip 

334 # ------------------------------------------------------------------ 

335 @app.post('/r/{rid}/filter') 

336 async def report_filter(rid: str, req: Request) -> NotStr | Response: 

337 path = library.resolve(rid, roots) 

338 if path is None: 

339 return Response('404 Not Found', status_code=404, media_type='text/plain') 

340 

341 surface, _raw = library.load_surface(path) 

342 adapter = ADAPTERS.get(surface or '') 

343 filter_def = FILTERS.get(surface or '') 

344 if adapter is None or filter_def is None: 

345 return Response('404 Not Found', status_code=404, media_type='text/plain') 

346 

347 try: 

348 report_obj = adapter.load(path) 

349 except Exception as exc: 

350 logger.warning('Failed to load report for filter {}: {}', path.name, exc) 

351 return Response( 

352 f'Error loading report: {exc}', 

353 status_code=422, 

354 media_type='text/plain', 

355 ) 

356 

357 # Parse form data — build selections dict[str, list[str]] 

358 form_data = await req.form() 

359 selections: dict[str, list[str]] = {} 

360 for key, value in form_data.multi_items(): 

361 selections.setdefault(key, []).append(str(value)) 

362 

363 # Apply filters to get the shown rows, but keep the option lists fixed 

364 # to the FULL dataset. Recomputing options from the filtered rows made 

365 # a just-deselected value disappear from its own multi-select. 

366 filtered = filter_def.apply(report_obj, selections) 

367 new_opts = filter_def.options(report_obj) 

368 

369 # Render the tabbed body from the filtered results so the static tab 

370 # content (tables, charts) tracks the filter, not just the HTMX panels. 

371 if surface == 'sim': 

372 body_html = report_tabs.sim_report_tabs(rid, report_obj, filtered) 

373 elif surface == 'redteam': 

374 from evaluatorq.redteam.reports.converters import rebuild_filtered_report 

375 

376 body_html = report_tabs.redteam_report_tabs(rid, rebuild_filtered_report(report_obj, filtered)) 

377 else: 

378 body_html = adapter.body_from_results(report_obj, filtered) 

379 

380 form_html = render_filter_form( 

381 rid, surface or '', new_opts, selections, shown=len(filtered), total=len(filter_def.results(report_obj)) 

382 ) 

383 fragment_html = filter_fragment(rid, surface or '', body_html, form_html) 

384 

385 # Signal interactive panels to refetch with the new filter. Panels 

386 # that carry hx-trigger="load, orq:filter-changed from:body" and 

387 # hx-include="#filter-form" will catch this event, re-issue their 

388 # hx-get requests with the current form values, and re-render from the 

389 # filtered result set. 

390 return Response( 

391 fragment_html, 

392 media_type='text/html', 

393 headers={'HX-Trigger': 'orq:filter-changed'}, 

394 )
