40def build_persona_system_prompt(persona: Persona) -> str: 

41 """Convert a persona to a system prompt for the user simulator.""" 

42 if persona.patience < 0.3: 

43 patience_desc = 'very impatient' 

44 elif persona.patience > 0.7: 

45 patience_desc = 'patient' 

46 else: 

47 patience_desc = 'moderately patient' 

48 

49 if persona.assertiveness > 0.7: 

50 assertive_desc = 'very assertive and direct' 

51 elif persona.assertiveness < 0.3: 

52 assertive_desc = 'passive' 

53 else: 

54 assertive_desc = 'balanced' 

55 

56 if persona.politeness < 0.3: 

57 polite_desc = 'rude and curt' 

58 elif persona.politeness > 0.7: 

59 polite_desc = 'very polite' 

60 else: 

61 polite_desc = 'neutral in tone' 

62 

63 if persona.technical_level < 0.3: 

64 tech_desc = 'a complete novice' 

65 elif persona.technical_level > 0.7: 

66 tech_desc = 'a technical expert' 

67 else: 

68 tech_desc = 'somewhat technical' 

69 

70 arc_text = '' 

71 arc_key = persona.emotional_arc or EmotionalArc.stable 

72 arc_instruction = EMOTIONAL_ARC_INSTRUCTIONS.get(arc_key, '') 

73 if arc_instruction: 

74 arc_text = f'\n\nEmotional Arc: {arc_instruction}' 

75 

76 cultural_text = '' 

77 cultural_key = persona.cultural_context or CulturalContext.neutral 

78 cultural_instruction = CULTURAL_CONTEXT_INSTRUCTIONS.get(cultural_key, '') 

79 if cultural_instruction: 

80 cultural_text = f'\n\nCultural Communication Style: {cultural_instruction}' 

81 

82 return ( 

83 f'You are simulating a user with the following characteristics:\n\n' 

84 f'Name: {delimit(persona.name)}\n' 

85 f'Patience: You are {patience_desc}\n' 

86 f'Assertiveness: You are {assertive_desc}\n' 

87 f'Politeness: You are {polite_desc}\n' 

88 f'Technical Level: You are {tech_desc}\n' 

89 f'Communication Style: {delimit(persona.communication_style.value)}\n\n' 

90 f'Background: {delimit(persona.background)}\n' 

91 f'{arc_text}{cultural_text}\n' 

92 f'Stay in character throughout the conversation. Your responses should reflect these traits consistently.\n' 

93 f'Do not break character or acknowledge that you are a simulation.' 

94 )

102def build_scenario_user_context(scenario: Scenario) -> str: 

103 """Convert a scenario to context for the user simulator.""" 

104 criteria_text = '' 

105 if scenario.criteria: 

106 must_happen = [delimit(c.description) for c in scenario.criteria if c.type == 'must_happen'] 

107 must_not = [delimit(c.description) for c in scenario.criteria if c.type == 'must_not_happen'] 

108 

109 if must_happen: 

110 criteria_text += f'\n\nYou expect the agent to: {", ".join(must_happen)}' 

111 if must_not: 

112 criteria_text += f'\n\nYou would be dissatisfied if: {", ".join(must_not)}' 

113 

114 strategy_text = '' 

115 strategy_key = scenario.conversation_strategy or ConversationStrategy.cooperative 

116 strategy_instruction = STRATEGY_INSTRUCTIONS.get(strategy_key, '') 

117 if strategy_instruction: 

118 strategy_text = f'\n\nConversation Strategy: {strategy_instruction}' 

119 

120 format_text = '' 

121 format_key = scenario.input_format or InputFormat.plain_text 

122 format_instruction = INPUT_FORMAT_INSTRUCTIONS.get(format_key, '') 

123 if format_instruction: 

124 format_text = f'\n\nMessage Format: {format_instruction}' 

125 

126 emotion_text = _EMOTION_INSTRUCTIONS.get( 

127 (scenario.starting_emotion or 'neutral') 

128 if isinstance(scenario.starting_emotion, str) 

129 else (scenario.starting_emotion.value if scenario.starting_emotion else 'neutral'), 

130 '', 

131 ) 

132 

133 return ( 

134 f'Scenario: {delimit(scenario.name)}\n\n' 

135 f'Your Goal: {delimit(scenario.goal)}\n\n' 

136 f'Context: {delimit(scenario.context or "")}\n\n' 

137 f'Emotional State: {emotion_text}\n' 

138 f'{criteria_text}{strategy_text}{format_text}\n\n' 

139 f'Work towards your goal naturally. React authentically based on how the agent responds.' 

140 )

148def build_datapoint_system_prompt(persona: Persona, scenario: Scenario) -> str: 

149 """Build the combined system prompt from persona and scenario.""" 

150 persona_prompt = build_persona_system_prompt(persona) 

151 scenario_context = build_scenario_user_context(scenario) 

152 return f'{persona_prompt}\n\n---\n\n{scenario_context}'

155def generate_datapoint(persona: Persona, scenario: Scenario, first_message: str = '') -> SimulationDatapoint: 

156 """Generate a datapoint from persona and scenario.""" 

157 dp_id = f'dp_{uuid.uuid4().hex[:12]}' 

158 return SimulationDatapoint( 

159 id=dp_id, 

160 persona=persona, 

161 scenario=scenario, 

162 user_system_prompt=build_datapoint_system_prompt(persona, scenario), 

163 first_message=first_message, 

164 )
