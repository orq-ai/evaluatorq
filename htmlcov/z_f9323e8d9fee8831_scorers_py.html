27class SimulationScoringConfig(BaseModel): 

28 """Tunables for the two built-in scorers that encode a *policy*, not a fact. 

29 

30 ``goal_achieved`` and ``criteria_met`` read the judge's verdicts and have nothing to 

31 tune. The other two make judgement calls this class exposes, mirroring 

32 ``SimulationRecommendationConfig``: bounded fields, ``extra='forbid'``, and defaults 

33 that carry over the previously hardcoded cliffs verbatim. The one deliberate 

34 difference is the tail past the last cliff, which now decays from that cliff's score 

35 instead of restarting from ``1.0`` — the old curve scored 7 turns higher than 6, so a 

36 run past the last cliff scores lower than it used to. See ``CHANGELOG.md``. Pass an 

37 instance as ``simulate(scoring=...)`` / 

38 ``generate_and_simulate(scoring=...)``, or to ``get_evaluator`` / 

39 ``get_all_evaluators`` when driving the scorers directly. 

40 

41 **What `turn_efficiency` measures.** It is a *cost* proxy, not a quality one: given 

42 that the goal was reached, how many conversational turns did it take? The assumption 

43 is that a user who got what they came for in two turns had a better experience — and 

44 cost less to serve — than one who needed twelve, because the extra turns are usually 

45 the agent asking for something it could have inferred, re-asking, or wandering. A run 

46 that did **not** reach the goal scores ``0.0`` outright: efficiency at failing is not 

47 a virtue worth crediting. 

48 

49 **Where that assumption breaks.** A task that legitimately needs many turns — a long 

50 intake form, a multi-step troubleshooting tree, a negotiation — is penalised by this 

51 metric for doing its job properly. If your scenarios look like that, either move the 

52 cliffs out (``turn_efficiency_cliffs=((6, 1.0), (10, 0.9), (16, 0.7))``) so the curve 

53 matches a realistic conversation length, or leave ``turn_efficiency`` out of 

54 ``evaluator_names`` and ignore the score. It is not in 

55 ``DEFAULT_EVALUATOR_NAMES`` precisely because it is not universally meaningful. 

56 

57 Worked example — a 4-turn conversation that reached its goal and met 1 of 2 criteria, 

58 scored with the defaults: 

59 

60 ```python 

61 from evaluatorq.simulation.evaluators import SimulationScoringConfig, conversation_quality_scorer 

62 

63 # goal_achieved = 1.0 (the judge set goal_achieved) 

64 # criteria_met = 0.5 (1 of 2 criteria met) 

65 # turn_efficiency = 0.9 (4 turns: past the <=2 cliff, inside the <=4 one) 

66 # 

67 # conversation_quality = 1.0 * 0.4 + 0.5 * 0.3 + 0.9 * 0.3 

68 # = 0.4 + 0.15 + 0.27 = 0.82 

69 conversation_quality_scorer(result) # 0.82 

70 

71 # Same conversation, scored as if only the goal mattered: 

72 goal_only = SimulationScoringConfig( 

73 goal_achieved_weight=1.0, criteria_met_weight=0.0, turn_efficiency_weight=0.0 

74 ) 

75 conversation_quality_scorer(result, goal_only) # 1.0 

76 ``` 

77 """

81 turn_efficiency_cliffs: tuple[tuple[int, float], ...] = ((2, 1.0), (4, 0.9), (6, 0.7)) 

82 """``(max_turns, score)`` steps, first match wins: a conversation of ``turn_count`` 

83 turns scores the first entry whose ``max_turns`` it does not exceed. Must be ordered 

84 by strictly increasing ``max_turns`` and non-increasing ``score`` — a curve that pays 

85 *more* for a longer conversation is a config bug, and is rejected at construction 

86 rather than producing a quietly nonsensical report.""" 

87 

88 turn_efficiency_decay_per_turn: float = Field(default=0.1, ge=0.0, le=1.0) 

89 """Beyond the last cliff, each additional turn costs this much, starting from the last 

90 cliff's score. With the defaults: 7 turns -> ``0.7 - 0.1 = 0.6``, 8 -> ``0.5``.""" 

91 

92 turn_efficiency_floor: float = Field(default=0.3, ge=0.0, le=1.0) 

93 """The decay stops here, so a very long conversation that *did* reach the goal keeps a 

94 non-zero score — it is inefficient, not a failure. Must not exceed the last cliff's 

95 score, or the floor would raise long conversations above shorter ones.""" 

96 

97 goal_achieved_weight: float = Field(default=0.4, ge=0.0, le=1.0) 

98 """Weight in ``conversation_quality`` of ``goal_achieved``: 1.0 when the judge decided 

99 the persona's goal was reached, 0.0 otherwise.""" 

100 

101 criteria_met_weight: float = Field(default=0.3, ge=0.0, le=1.0) 

102 """Weight in ``conversation_quality`` of ``criteria_met``: the fraction of the 

103 scenario's ``must_happen`` / ``must_not_happen`` criteria the judge audited and found 

104 satisfied (0.0 for an errored, timed-out or unverified run).""" 

105 

106 turn_efficiency_weight: float = Field(default=0.3, ge=0.0, le=1.0) 

107 """Weight in ``conversation_quality`` of ``turn_efficiency``, the turn-count cost proxy 

108 described above.""" 

109 

110 @field_validator('turn_efficiency_cliffs') 

111 @classmethod 

112 def _validate_cliffs(cls, value: tuple[tuple[int, float], ...]) -> tuple[tuple[int, float], ...]: 

113 if not value: 

114 raise ValueError('turn_efficiency_cliffs must contain at least one (max_turns, score) step') 

115 previous_turns = 0 

116 previous_score = None 

117 for max_turns, score in value: 

118 if max_turns < 1: 

119 raise ValueError(f'turn_efficiency_cliffs: max_turns must be >= 1, got {max_turns}') 

120 if max_turns <= previous_turns: 

121 raise ValueError( 

122 f'turn_efficiency_cliffs: max_turns must strictly increase, got {max_turns} after {previous_turns}' 

123 ) 

124 if not 0.0 <= score <= 1.0: 

125 raise ValueError(f'turn_efficiency_cliffs: score must be in [0.0, 1.0], got {score}') 

126 if previous_score is not None and score > previous_score: 

127 raise ValueError( 

128 f'turn_efficiency_cliffs: score must not increase with turns, got {score} after {previous_score} ' 

129 '(that curve rewards a longer conversation)' 

130 ) 

131 previous_turns, previous_score = max_turns, score 

132 return value 

133 

134 @model_validator(mode='after') 

135 def _validate_shape(self) -> SimulationScoringConfig: 

136 last_score = self.turn_efficiency_cliffs[-1][1] 

137 if self.turn_efficiency_floor > last_score: 

138 raise ValueError( 

139 f'turn_efficiency_floor ({self.turn_efficiency_floor}) exceeds the last cliff score ({last_score}); ' 

140 'that scores a long conversation above a shorter one' 

141 ) 

142 total = self.goal_achieved_weight + self.criteria_met_weight + self.turn_efficiency_weight 

143 if abs(total - 1.0) > _WEIGHT_SUM_TOLERANCE: 

144 raise ValueError( 

145 f'conversation_quality weights must sum to 1.0, got {total} ' 

146 f'(goal_achieved={self.goal_achieved_weight}, criteria_met={self.criteria_met_weight}, ' 

147 f'turn_efficiency={self.turn_efficiency_weight}); an unnormalised sum makes the composite ' 

148 'incomparable with other runs' 

149 ) 

150 return self

157def read_criteria_meta(result: SimulationResult) -> tuple[list[CriteriaMeta], list[object]]: 

158 """Read the persisted criteria contract and record malformed entries.""" 

159 valid, invalid = parse_criteria_meta(result.metadata.get('criteria_meta')) 

160 if invalid: 

161 errors = result.metadata.setdefault('criteria_errors', []) 

162 if not isinstance(errors, list): 

163 errors = result.metadata['criteria_errors'] = [] 

164 for entry in invalid: 

165 message = f'criteria_meta entry is invalid: {entry!r}' 

166 logger.warning(message) 

167 if message not in errors: 

168 errors.append(message) 

169 return valid, invalid

182def criteria_met_scorer(result: SimulationResult) -> float: 

183 """Fraction of the scenario's criteria that were met, 0..1. 

184 

185 A run that errored or timed out is scored **0.0**: it terminated before the 

186 judge could audit anything, so its criteria outcome is unknown, and scoring an 

187 unchecked run 1.0 let a dead target inflate the run average (RES-1308). 

188 A run whose `criteria_verified` is False scores 0.0 for the same reason: the 

189 judge never returned an occurrence audit, so the verdicts came from the 

190 free-text `rules_broken` list, which cannot fail a `must_happen` criterion. 

191 A run with no criteria at all still scores 1.0 — nothing to fail. 

192 

193 An individual criterion the judge never audited is **not met** either, even on 

194 an otherwise verified run: it is passing only by its not-observed default, and 

195 that is the same flattering silence at criterion granularity. This is read from 

196 ``metadata['criteria_meta']`` (the only place ``audited`` survives — 

197 ``criteria_results`` is keyed by description and carries no provenance), so the 

198 score matches the ``N/M criteria met`` tally the reports print, which counts 

199 `CriteriaRow.state == 'pass'`. A criterion the judge settled early *is* audited 

200 (`_CriteriaTracker` records a verdict before it can become settled), so 

201 `JudgeAgent.mark_settled` never costs a run a point. 

202 

203 A malformed ``criteria_meta`` entry is an error: it is logged, recorded on 

204 the result for downstream reports, and scores 0.0 rather than being dropped. 

205 """ 

206 if result.terminated_by in UNEVALUATED_TERMINATIONS: 

207 logger.warning( 

208 'criteria_met: run terminated by {} before any criteria audit; scoring 0.0 (unknown, not met).', 

209 result.terminated_by.value, 

210 ) 

211 return 0.0 

212 

213 # `None` is a run saved before the flag existed — leave those scored as they were. 

214 if result.criteria_verified is False: 

215 logger.warning( 

216 'criteria_met: judge returned no per-criterion occurrence audit, so these verdicts cannot ' 

217 'fail a must_happen criterion; scoring 0.0 (unverified, not met).' 

218 ) 

219 return 0.0 

220 

221 raw_meta = result.metadata.get('criteria_meta') 

222 if raw_meta is not None: 

223 entries, invalid = read_criteria_meta(result) 

224 if invalid: 

225 logger.warning('criteria_met: invalid criteria_meta is an error; scoring 0.0 (unknown, not met).') 

226 return 0.0 

227 if entries: 

228 unaudited = [c for c in entries if c.audited is False and c.passed] 

229 if unaudited: 

230 logger.warning( 

231 'criteria_met: {} of {} criteria passed only by default (the judge never audited them); ' 

232 'counting them as not met.', 

233 len(unaudited), 

234 len(entries), 

235 ) 

236 met = sum(1 for c in entries if c.passed and c.audited is not False) 

237 return met / len(entries) 

238 logger.warning( 

239 'criteria_met: criteria_meta is present but empty; falling back to criteria_results, ' 

240 'which carries no audit provenance.' 

241 ) 

242 

243 criteria_results = result.criteria_results or {} 

244 if not criteria_results: 

245 return 1.0 

246 

247 met = sum(1 for v in criteria_results.values() if v) 

248 return met / len(criteria_results)

251def turn_efficiency_scorer(result: SimulationResult, config: SimulationScoringConfig | None = None) -> float: 

252 """How cheaply the goal was reached, 0..1 — fewer turns scores higher. 

253 

254 A run that did not reach its goal scores ``0.0``: this is a cost metric conditioned on 

255 success, and "failed quickly" is not efficiency. Otherwise the turn count is mapped 

256 through ``config.turn_efficiency_cliffs`` (first match wins), then decayed by 

257 ``turn_efficiency_decay_per_turn`` from the last cliff's score down to 

258 ``turn_efficiency_floor``. 

259 

260 The assumption — extra turns mean re-asking, clarifying or wandering — does not hold 

261 for tasks that legitimately need many turns. See ``SimulationScoringConfig`` for how to 

262 move the cliffs, or drop this scorer for such scenarios. 

263 """ 

264 config = config or DEFAULT_SCORING_CONFIG 

265 

266 if not result.goal_achieved: 

267 return 0.0 

268 

269 total_turns = result.turn_count 

270 for max_turns, score in config.turn_efficiency_cliffs: 

271 if total_turns <= max_turns: 

272 return score 

273 

274 last_turns, last_score = config.turn_efficiency_cliffs[-1] 

275 decayed = last_score - (total_turns - last_turns) * config.turn_efficiency_decay_per_turn 

276 # Round: repeated float subtraction turns 0.6 into 0.6000000000000001, which then 

277 # renders in a report as a 16-digit score. 

278 return round(max(config.turn_efficiency_floor, decayed), 4)

281def conversation_quality_scorer(result: SimulationResult, config: SimulationScoringConfig | None = None) -> float: 

282 """Weighted composite of the other three scorers, 0..1, rounded to 2 decimals. 

283 

284 Defaults: ``goal_achieved`` 0.4, ``criteria_met`` 0.3, ``turn_efficiency`` 0.3. The 

285 weights are validated to sum to 1.0 at config construction, so the composite stays on 

286 the same 0..1 scale as its parts and is comparable across runs. 

287 """ 

288 config = config or DEFAULT_SCORING_CONFIG 

289 goal_score = goal_achieved_scorer(result) 

290 criteria_score = criteria_met_scorer(result) 

291 efficiency_score = turn_efficiency_scorer(result, config) 

292 

293 score = ( 

294 goal_score * config.goal_achieved_weight 

295 + criteria_score * config.criteria_met_weight 

296 + efficiency_score * config.turn_efficiency_weight 

297 ) 

298 return round(score * 100) / 100

313def get_evaluator(name: str, config: SimulationScoringConfig | None = None) -> SimulationScorer: 

314 """Get a built-in simulation evaluator by name, bound to ``config`` when given. 

315 

316 ``config`` only affects ``turn_efficiency`` and ``conversation_quality``; the other two 

317 read the judge's verdicts and have nothing to tune, so they are returned unwrapped. 

318 

319 Raises: 

320 ValueError: If evaluator not found. 

321 """ 

322 evaluator = get_all_evaluators(config).get(name) 

323 if not evaluator: 

324 available = ', '.join(SIMULATION_EVALUATORS.keys()) 

325 raise ValueError(f'Unknown evaluator: {name}. Available: {available}') 

326 return evaluator

329def get_all_evaluators(config: SimulationScoringConfig | None = None) -> dict[str, SimulationScorer]: 

330 """Get all built-in simulation evaluators, bound to ``config`` when given.""" 

331 evaluators = dict(SIMULATION_EVALUATORS) 

332 if config is not None: 

333 evaluators['turn_efficiency'] = partial(turn_efficiency_scorer, config=config) 

334 evaluators['conversation_quality'] = partial(conversation_quality_scorer, config=config) 

335 return evaluators
