256class Persona(BaseModel): 

257 """A simulated user profile driving one side of a conversation. 

258 

259 Usage: 

260 

261 ```python 

262 from evaluatorq.simulation import CommunicationStyle, EmotionalArc, Persona

264 persona = Persona( 

265 name="Impatient Customer", 

266 patience=0.2, 

267 assertiveness=0.8, 

268 politeness=0.4, 

269 technical_level=0.3,

271 background="Received the wrong item and wants a refund urgently", 

272 emotional_arc=EmotionalArc.escalating, # optional: tone escalates each turn 

273 ) 

274 ``` 

275 

276 See `Scenario` for the matching goal/criteria half of a simulation datapoint. 

277 """

399 thread_id: str | None = Field( 

400 default=None, 

401 description='Orq observability thread id for this conversation (deterministic: f"{run_id}:{index}"). ' 

402 'None for runs saved before thread grouping existed.', 

403 ) 

404 response_traces: list[ResponseTrace] = Field( 

405 default_factory=list, 

406 description='Per-turn Orq trace/span handles for each successful target-agent response '

409 ) 

410 

411 @property 

412 def last_trace_id(self) -> str | None: 

413 """Trace id of the last successful target response, for the table deep-link.""" 

414 for rt in reversed(self.response_traces): 

415 if rt.trace_id: 

416 return rt.trace_id 

417 return None

470class SimulationRun(BaseModel): 

471 run_name: str 

472 created_at: datetime 

473 mode: Literal['run', 'simulate', 'generate'] 

474 target_kind: Literal['orq_agent', 'orq_deployment', 'vercel', 'openai_model', 'callback'] 

475 # The concrete target under test: agent key / deployment key / model id / 

476 # 'callback'. ``target_model`` is the model that target used *when the client 

477 # knows it* (OpenAI-model targets); for orq agents/deployments the model is 

478 # server-side config, so it stays None. Optional for back-compat with runs 

479 # saved before these fields existed. 

480 target: str | None = None 

481 target_model: str | None = None 

482 # Configured turn cap for the run. Optional for back-compat with runs saved 

483 # before this field existed (older reports simply omit the Config row). 

484 max_turns: int | None = None 

485 # Snapshot of the ORQ agent's configuration at run time (orq_agent targets 

486 # only; None for other target kinds, fetch failures, and pre-existing runs). 

487 # Deliberately excludes the system prompt / instructions. 

488 agent_info: AgentInfoSnapshot | None = None 

489 orq_base_url: str | None = None 

490 """The Orq host that served this run (``ORQ_BASE_URL`` or the prod default), 

491 recorded so a saved run remembers which deployment — prod / staging / on-prem 

492 — it ran against. None when no Orq agent/deployment was used (plain callable / 

493 OpenAI-model targets) and for runs saved before this field existed.""" 

494 evaluator_names: list[str] 

495 total_results: int 

496 scorer_averages: dict[str, float] 

497 results: list[SimulationResult] 

498 datapoints: list[SimulationDatapoint] | None = None 

499 """The exact cases this run simulated, stored so the run can be replayed 

500 verbatim (``previous_run=`` / ``--from-run``). Results alone don't carry 

501 enough — they keep persona/scenario *names*, not the objects. None for runs 

502 saved before this field existed, which therefore cannot be replayed.""" 

503 replay_version: int | None = None 

504 """Format version of the replay payload above, stamped when ``datapoints`` is 

505 written so a future format change reports itself instead of failing 

506 structurally. None for runs saved before versioning, which read as v1.""" 

507 executive_summary: str | None = None 

508 run_id: str | None = None 

509 """Client-minted run-grouping id (uuid hex, not an Orq-side run id) shared by every 

510 conversation's ``thread_id`` (``{run_id}:{index}``). Powers the dashboard's 

511 'View all run traces' deep link. None for older runs.""" 

512 experiment_url: str | None = None 

513 """Absolute URL of the Orq experiment this run was uploaded to, captured from the 

514 results upload. Powers the terminal 'View on Orq' line and the dashboard's 

515 'Open experiment' button. None when upload was skipped/failed or for older runs.""" 

516 recommendations: list[SimulationRecommendation] | None = None 

517 """LLM-generated remediation suggestions for remediable failures 

518 (see ``reports.recommendations``). None when never generated.""" 

519 applied_suggestions: list[str] = Field(default_factory=list) 

520 """Suggestion strings already applied to the agent via ``reports.apply``. 

521 Written back onto the run so the dashboard renders them differently and a 

522 later apply skips them instead of re-applying the same fix.""" 

523 

524 def manifest_summary(self) -> RunSummary: 

525 """Compact run-list summary stored on this run's ``RunManifest``. 

526 

527 Single source of truth for the shape: the runner writes it on completion 

528 and the dashboard's backfill writes it for legacy runs. `eq sim runs` 

529 reads every field here, so a second hand-rolled shape silently blanks 

530 columns. 

531 """ 

532 return { 

533 'mode': self.mode, 

534 'target_kind': self.target_kind, 

535 'total_results': self.total_results, 

536 'scorer_averages': dict(self.scorer_averages), 

537 }
