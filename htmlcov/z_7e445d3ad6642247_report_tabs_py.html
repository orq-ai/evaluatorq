47def _tabs(group: str, items: list[tuple[str, str] | tuple[str, str, str]]) -> str: 

48 """Render a CSS-only tab group. 

49 

50 ``items`` is an ordered list of ``(label, panel_html)`` or ``(label, 

51 panel_html, label_html)``. In the 2-tuple form the label is HTML-escaped; 

52 in the 3-tuple form ``label_html`` is rendered raw inside the ``<label>`` 

53 (the caller is responsible for escaping it), which lets a surface inject 

54 e.g. a count pill next to the label. Tabs whose panel is empty are dropped 

55 so a surface that lacks (say) error or comparison data simply shows fewer 

56 tabs, matching the Streamlit conditional-tab behaviour. The first 

57 surviving tab is checked. Switching is pure CSS (see ``styles.py`` 

58 ``_TAB_RULES``): the Nth radio toggles the Nth label and Nth panel. 

59 """ 

60 live = [it for it in items if it[1] and it[1].strip()] 

61 if not live: 

62 return '' 

63 radios: list[str] = [] 

64 labels: list[str] = [] 

65 panels: list[str] = [] 

66 for i, it in enumerate(live): 

67 label, html = it[0], it[1] 

68 label_html = it[2] if len(it) > 2 else esc(label) 

69 tid = f'{group}-{i}' 

70 checked = ' checked' if i == 0 else '' 

71 radios.append(f'<input class="tab-radio" type="radio" name="{esc(group)}" id="{esc(tid)}"{checked}>') 

72 labels.append(f'<label class="tab-label" for="{esc(tid)}">{label_html}</label>') 

73 panels.append(f'<section class="tab-panel">{html}</section>') 

74 return ( 

75 f'<div class="tabs">{"".join(radios)}' 

76 f'<div class="tab-bar">{"".join(labels)}</div>' 

77 f'<div class="tab-panels">{"".join(panels)}</div>' 

78 f'</div>' 

79 )

102def sim_report_tabs(rid: str, run: SimulationRun, results: list[Any] | None = None, compare_html: str = '') -> str: 

103 """Render the Agent Sim report body as Streamlit-aligned tabs. 

104 

105 Tabs: Overview · Breakdown · Transcripts · Turn quality · Config — each 

106 populated from the precomputed report sections (empty tabs drop out; Turn 

107 quality drops when a run carries no ``turn_metrics``). Config folds job-level 

108 metadata (run configuration, personas, scenarios) plus the kept token_usage 

109 table. Pass ``results`` to render a filtered subset (the filter round-trip); 

110 it defaults to the run's full result list. ``compare_html`` is an optional 

111 pre-rendered compare-with-another-run control shown in the hero actions. 

112 """ 

113 from evaluatorq.dashboard.view import sim_interactive_panels 

114 from evaluatorq.simulation.reports.export_html import _SECTION_RENDERERS 

115 from evaluatorq.simulation.reports.sections import build_report_sections 

116 

117 rows = run.results if results is None else results 

118 # The executive summary describes the complete run and always stays visible, 

119 # even on filtered views — it is whole-run context, not a subset metric. 

120 sections = build_report_sections(rows, executive_summary=run.executive_summary) 

121 by_kind: dict[str, Any] = {} 

122 for s in sections: 

123 by_kind.setdefault(s.kind, s) 

124 

125 # Filtering affects report metrics and Breakdown rows, but Config must 

126 # remain a faithful registry of every configured persona and scenario. 

127 # Build its drawer templates from the full run so a cohort that has no 

128 # matching filtered conversations remains reachable and keeps its stable 

129 # DOM ID. Only the cohort conversation lists below use filtered entries. 

130 entity_sections = sections if results is None else build_report_sections(run.results) 

131 entity_by_kind: dict[str, Any] = {} 

132 for s in entity_sections: 

133 entity_by_kind.setdefault(s.kind, s) 

134 

135 def render(*kinds: str) -> str: 

136 return _render_sections(by_kind, _SECTION_RENDERERS, kinds) 

137 

138 hero = _sim_hero(run, compare_html) 

139 

140 entries = _stable_entries(run, rows) 

141 

142 entity_context = _sim_entity_context(entity_by_kind, entries, rows, rid) 

143 _set_failure_full_run_indexes(by_kind.get('failures_first'), entries) 

144 

145 # When a filter matches no conversations, the data-driven tabs would 

146 # collapse to blank/dropped. Keep them present with an explicit "no matches" 

147 # note so the report structure stays stable (Config still shows the full-run 

148 # registry; Overview handles its own empty state internally). 

149 no_matches = results is not None and not rows 

150 

151 breakdown_body = _SIM_NO_MATCH_NOTE if no_matches else _sim_breakdown(by_kind, rid, entity_context) 

152 transcripts_body = ( 

153 _SIM_NO_MATCH_NOTE 

154 if no_matches 

155 else sim_interactive_panels(rid, entries) + render('evaluator_scores', 'judge_verdicts', 'errors') 

156 ) 

157 turn_quality_body = _SIM_NO_MATCH_NOTE if no_matches else _sim_turn_quality(by_kind) 

158 

159 # Folded 7→5 to curb tab sprawl: Evaluators + Judge & errors → Transcripts 

160 # (all per-conversation verdicts); Tokens → Config. Turn quality is its own 

161 # tab (unfolded from Breakdown) and drops out when a run has no turn_metrics. 

162 tabs = _tabs( 

163 'simtab', 

164 [ 

165 ('Overview', _sim_overview(rid, by_kind, entity_by_kind, rows, run, filtered=results is not None)), 

166 ('Breakdown', breakdown_body), 

167 ( 

168 'Transcripts', 

169 transcripts_body, 

170 f'Transcripts <span class="tab-count">{len(entries)}</span>', 

171 ), 

172 ('Turn quality', turn_quality_body), 

173 ('Config', _sim_config(by_kind, run, entity_context) + render('token_usage')), 

174 ], 

175 ) 

176 return f'<div class="report-aligned sim-report">{hero}{tabs}{_sim_entity_modal(entity_context)}</div>'

368def _sim_config_criteria_split(criteria: list[dict[str, Any]]) -> str: 

369 """`N must-happen · M must-not` label, split by criterion type. Falls back 

370 to the flat `N checks` label when no criterion carries a recognised type.""" 

371 must_happen = sum(1 for c in criteria if c.get('type') == 'must_happen') 

372 must_not = sum(1 for c in criteria if c.get('type') == 'must_not_happen') 

373 if must_happen == 0 and must_not == 0: 

374 count = len(criteria) 

375 label = f'{count} {"check" if count == 1 else "checks"}' 

376 return f'<span class="sim-config-check-count">{label}</span>' 

377 

378 parts: list[str] = [] 

379 if must_happen: 

380 parts.append(f'{must_happen} must-happen') 

381 if must_not: 

382 parts.append(f'{must_not} must-not') 

383 return f'<span class="sim-config-check-count">{" · ".join(parts)}</span>'

828 '<span class="sim-fm-filter">' 

829 '<label for="sim-fm-slider">min count</label>' 

830 f'<input id="sim-fm-slider" type="range" class="sim-fm-slider" data-fm-slider '

845def _turn_delta_callout(series: dict[str, list[float | None]]) -> str: 

846 """Templated first-to-last-turn delta callout, no confidence pill (spec 

847 §Turn.1). A clause renders only for series with >= 2 non-None points; 

848 absent/short metrics are dropped. Returns '' when nothing qualifies.""" 

849 clauses: list[str] = [] 

850 for metric in TURN_METRICS: 

851 values = series.get(metric.key, []) 

852 points = [v for v in values if v is not None] 

853 if len(points) < 2: 

854 continue 

855 delta = points[-1] - points[0] 

856 label = esc(metric.label) 

857 if abs(delta) < 0.005: 

858 clauses.append(f'{label} held steady around <strong>{points[-1]:.2f}</strong>') 

859 continue 

860 verb = ('rose' if delta > 0 else 'fell') if metric.high_is_risky else ('improved' if delta > 0 else 'declined') 

861 clauses.append(f'{label} {verb} by <strong>{abs(delta):.2f}</strong> from turn 1 to the last turn') 

862 if not clauses: 

863 return '' 

864 body = '; '.join(clauses) + '.' 

865 sentence = body[0].upper() + body[1:] 

866 return ( 

867 '<div class="exec-summary">' 

868 '<div class="es-head"><span class="es-label">Turn quality trend</span></div>' 

869 f'<p class="es-body">{sentence}</p>' 

870 '</div>' 

871 )

1105async def _resolve_agent_info( 

1106 run: SimulationRun, 

1107) -> tuple[dict[str, Any] | None, dict[str, Any] | None, str]: 

1108 """Resolve the agent card's data + provenance as ``(display, original, source)``. 

1109 

1110 ``source`` is one of: 

1111 - ``captured`` — the run's own snapshot is complete; no Orq call. 

1112 - ``augmented`` — snapshot present but missing a core field; gaps filled live. 

1113 - ``fetched`` — nothing captured; whole card loaded live from Orq. 

1114 - ``stored`` — Orq could not enrich the run; card uses saved target data. 

1115 - ``none`` — no data and nothing to fetch. 

1116 

1117 We only call Orq when the target is an Orq agent AND the captured snapshot is 

1118 absent or missing a core field. Captured (as-run) values always win 

1119 per-field, so a live fetch only fills gaps — it never overwrites what the run 

1120 actually recorded. 

1121 """ 

1122 captured = run.agent_info if isinstance(run.agent_info, dict) and run.agent_info else None 

1123 if run.target_kind != 'orq_agent': 

1124 return captured, captured, ('captured' if captured else 'none') 

1125 

1126 missing_core = captured is None or any(not captured.get(f) for f in _AGENT_CORE_FIELDS) 

1127 if not missing_core: 

1128 return captured, captured, 'captured' 

1129 

1130 agent_key = _agent_key_for(run) 

1131 if not agent_key: 

1132 return captured, captured, ('captured' if captured else 'none') 

1133 fetched = await _orq_agent_info_cached(agent_key) 

1134 if not fetched: 

1135 return captured or _stored_agent_info(run), captured, ('captured' if captured else 'stored') 

1136 if captured: 

1137 # Fill only missing attributes; captured values win per-field. 

1138 merged = {**fetched, **{k: v for k, v in captured.items() if v}} 

1139 return merged, captured, 'augmented' 

1140 return fetched, None, 'fetched'

1484 unevaluated agent reports ``asr``/``resistance`` as ``None`` rather than 

1485 a false 0% ASR / 100% resistance. 

1486 """

1489 key = r.agent.key or r.agent.display_name or r.agent.model or 'unknown' 

1490 entry: dict[str, Any] = stats.setdefault( 

1491 key, 

1492 { 

1493 'display_name': r.agent.display_name or key, 

1494 'model': r.agent.model or '', 

1495 'attacks': 0, 

1496 'evaluated': 0, 

1497 'vulns': 0, 

1498 'critical': 0, 

1499 'errors': 0, 

1500 }, 

1501 ) 

1502 entry['attacks'] += 1 

1503 if r.vulnerable is not None: 

1504 entry['evaluated'] += 1 

1505 if r.vulnerable: 

1506 entry['vulns'] += 1 

1507 if r.attack.severity == 'critical': 

1508 entry['critical'] += 1 

1509 if r.error is not None: 

1510 entry['errors'] += 1 

1511 for entry in stats.values(): 

1512 evaluated = entry['evaluated'] 

1513 vulns = entry['vulns'] 

1514 entry['asr'] = vulns / evaluated if evaluated else None 

1515 entry['resistance'] = (1.0 - entry['asr']) if entry['asr'] is not None else None 

1516 return stats
