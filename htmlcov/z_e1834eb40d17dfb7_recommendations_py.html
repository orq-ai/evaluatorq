1"""Shared shape for the ``recommendations=`` flag and the config behind it. 

2 

3Both surfaces spell the toggle the same way — ``red_team(recommendations=...)`` and the 

4simulation run entry points — so a caller learns one rule: ``True`` means "on, with 

5defaults", ``False`` means "skip the LLM call", and a config instance means "on, tuned". 

6 

7The two configs share only what genuinely means the same thing on both surfaces (the 

8suggestion cap and the analysis call's token budget), which is what 

9``RecommendationConfigBase`` holds. Everything else is surface-specific — red 

10teaming samples failed traces per risk area, simulation gates on judge metric thresholds 

11— so each surface subclasses with its own fields rather than sharing one wide model where 

12half the knobs are inert. 

13""" 

14 

15from __future__ import annotations 

16 

17from typing import TYPE_CHECKING, TypeVar 

18 

19from pydantic import BaseModel, ConfigDict, Field 

20 

21if TYPE_CHECKING: 

22 from collections.abc import Callable 

23 

24T = TypeVar('T') 

25 

26 

27class RecommendationConfigBase(BaseModel): 

28 """Recommendation-generation knobs that mean the same thing on every surface. 

29 

30 Bounds are enforced because a parseable value is not necessarily a meaningful one: 

31 a suggestion cap of 0 pays for the LLM call and then drops every result. 

32 """ 

33 

34 model_config = ConfigDict(extra='forbid') 

35 

36 max_suggestions: int = Field(default=3, ge=1) 

37 """How many suggestions the analysis LLM is asked for, and kept from, per call.""" 

38 

39 max_tokens: int = Field(default=800, ge=1) 

40 """Completion budget for one analysis call. Raise it if suggestions get truncated.""" 

41 

42 

43def resolve_recommendations(value: bool | T, default: Callable[[], T]) -> T | None: # noqa: FBT001 — the flag IS the API 

44 """The config to generate recommendations with, or ``None`` to skip generation.""" 

45 if value is True: 

46 return default() 

47 if value is False: 

48 return None 

49 return value
