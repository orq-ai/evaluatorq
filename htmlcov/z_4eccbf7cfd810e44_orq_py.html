72def _orq_retry_config( 

73 retry_count: int, 

74 retry_on_codes: list[int] | None = None, 

75 *, 

76 timeout_ms: int | None = None, 

77) -> Any: 

78 """Client-level retry for the Orq SDK: the configured ``retry_on_codes`` 

79 (default 429/500/502/503/504, matching the SDK's own per-operation default) 

80 plus connection errors, with ``Retry-After`` honored. 

81 

82 The Speakeasy-generated SDK bounds retries by *total wall clock from before 

83 the first attempt* (``max_elapsed_time``), not by attempt count — request 

84 duration counts against the window. So the window must budget for the 

85 attempts themselves, not just the backoff sleeps: ``retry_count`` failed 

86 attempts of up to ``timeout_ms`` each, plus the backoff rounds between them 

87 (0.5s initial, doubling, 10s cap). A backoff-only window would let a single 

88 slow failure (a 503 behind a slow upstream, a TCP timeout) exhaust the 

89 budget on the first attempt and retry nothing. 

90 

91 ``retry_count=0`` disables. Returns None when disabled or the SDK is 

92 unavailable. 

93 """ 

94 if retry_count <= 0 or _orq_cls is None: 

95 return None 

96 from orq_ai_sdk.utils.retries import BackoffStrategy, RetryConfig 

97 

98 backoff_window_ms = int(sum(min(500 * 2**i, 10_000) for i in range(retry_count))) 

99 attempt_budget_ms = retry_count * (timeout_ms or PIPELINE_CONFIG.target_agent_timeout_ms) 

100 return RetryConfig( 

101 strategy='backoff', 

102 backoff=BackoffStrategy( 

103 initial_interval=500, 

104 max_interval=10_000, 

105 exponent=2, 

106 max_elapsed_time=backoff_window_ms + attempt_budget_ms, 

107 ), 

108 retry_connection_errors=True, 

109 status_codes_override=[str(code) for code in retry_on_codes] if retry_on_codes else None, 

110 )

133def _orq_map_error(exc: Exception) -> tuple[str, str]: 

134 """Map an ORQ exception to a normalized (error_code, error_message) tuple.""" 

135 name = type(exc).__name__.lower() 

136 text = str(exc).lower() 

137 status_code = extract_status_code(exc) 

138 provider_code = extract_provider_error_code(exc) 

139 if status_code is not None: 

140 return f'orq.http.{status_code}', f'{type(exc).__name__}: {exc}' 

141 if provider_code: 

142 return f'orq.code.{provider_code}', f'{type(exc).__name__}: {exc}' 

143 if 'timeout' in name or 'timed out' in text: 

144 return 'orq.timeout', f'{type(exc).__name__}: {exc}' 

145 if 'auth' in name or 'unauthorized' in text or 'forbidden' in text: 

146 return 'orq.auth', f'{type(exc).__name__}: {exc}' 

147 if 'ratelimit' in name or '429' in text: 

148 return 'orq.rate_limit', f'{type(exc).__name__}: {exc}' 

149 return 'orq.unknown', f'{type(exc).__name__}: {exc}'

209 async def respond(self, messages: list[Message]) -> AgentResponse: 

210 """Send the last user message to the ORQ agents endpoint, threaded via ``task_id``. 

211 

212 The ORQ agents endpoint holds conversation state server-side via 

213 ``task_id``, so ``respond`` forwards only the last user turn. Prior turns 

214 in ``messages`` are assumed consistent with the server-side history that 

215 ``task_id`` references; the first call (``task_id is None``) seeds it. 

216 

217 Caller contract: ``messages[-1].role == "user"``. A mismatch with the 

218 server-side history is a caller bug. 

219 

220 Token usage is accumulated across pending-tool-call continuations. 

221 

222 Retry owner: ``call_target_with_retry`` in the orchestrator. Each SDK 

223 operation passes ``retries=None`` **explicitly** — omitting it means 

224 ``UNSET`` (inherit the client budget) in the Speakeasy-generated SDK, not 

225 "no retry", so the two layers would multiply. 

226 """ 

227 if not messages or messages[-1].role != 'user': 

228 raise ValueError( 

229 "ORQAgentTarget.respond requires messages[-1].role == 'user'. " 

230 'Server-side conversation state is held via task_id.' 

231 ) 

232 # The agents endpoint takes a plain string ``prompt``, not Responses 

233 # content-parts, so RES-879's "pass through" does not apply here: we 

234 # intentionally flatten to text. content_to_text raises on image/file 

235 # parts, which this stateful text endpoint cannot forward anyway. 

236 prompt = content_to_text(messages[-1].content) 

237 accumulated_usage = TokenUsage()

240 """Accumulate token usage from a response into a running total. 

241 

242 Routes through the shared extractor so cached/reasoning/cost are carried 

243 and the total fallback matches every other path. 

244 """ 

245 nonlocal accumulated_usage 

246 delta = TokenUsage.extract(getattr(resp, 'usage', None), calls=1) 

247 if delta is not None: 

248 accumulated_usage = accumulated_usage + delta

252 output = getattr(resp, 'output', None) or []

262 async def _create_traced(input_messages: list[dict[str, Any]], **kwargs: Any) -> Any: 

263 """Run one ``agents.responses.create`` inside its own LLM span. 

264 

265 The agents endpoint is stateful and loops for tool continuations, so a 

266 single ``respond()`` makes several calls. Each gets a span mirroring the 

267 chat-completions / responses shape other backends emit, carrying that 

268 call's own usage. The enclosing target span deliberately records no 

269 usage of its own: a total there would double-count against these. 

270 """ 

271 async with with_llm_span( 

272 model=self.model or self.agent_key, 

273 operation='agents.responses', 

274 provider='orq', 

275 input_messages=input_messages, 

276 attributes={'orq.redteam.llm_purpose': 'target', 'orq.agent.key': self.agent_key}, 

277 ) as llm_span: 

278 # Propagate W3C trace context so the server-side agent trace nests 

279 # under this span instead of starting a loose root trace. Captured 

280 # inside the span so `traceparent` points at it; gated by 

281 # EVALUATORQ_PROPAGATE_TRACE_CONTEXT. 

282 trace_headers = await get_trace_context_headers() 

283 if trace_headers: 

284 kwargs['http_headers'] = {**(kwargs.get('http_headers') or {}), **trace_headers} 

285 resp = await asyncio.to_thread(self.orq_client.agents.responses.create, **kwargs) 

286 record_llm_response(llm_span, resp, output_content=_extract_text(resp)) 

287 return resp

600 async def _enrich_tool(self, binding: Any) -> ToolInfo: 

601 """Resolve an agent tool binding to full ToolInfo (description + parameter schema). 

602 

603 The agent-retrieve endpoint only returns a tool *binding* (no JSON schema). 

604 When a tool_id is present we fetch the underlying tool definition for its 

605 real description and parameters; otherwise we fall back to binding fields. 

606 """ 

607 name = ( 

608 getattr(binding, 'key', None) or getattr(binding, 'display_name', None) or getattr(binding, 'id', 'unknown') 

609 ) 

610 description = getattr(binding, 'description', None) 

611 action_type = getattr(binding, 'action_type', None) 

612 parameters = None 

613 tool_id = getattr(binding, 'tool_id', None) 

614 if tool_id: 

615 try: 

616 rich = await asyncio.to_thread(self.orq_client.tools.retrieve, tool_id=tool_id) 

617 fn = getattr(rich, 'function', None) 

618 js = getattr(rich, 'json_schema', None) 

619 if fn is not None: 

620 description = getattr(fn, 'description', None) or getattr(rich, 'description', None) or description 

621 raw_params = getattr(fn, 'parameters', None) 

622 elif js is not None: 

623 description = getattr(js, 'description', None) or getattr(rich, 'description', None) or description 

624 raw_params = getattr(js, 'schema_', None) 

625 else: 

626 # http, mcp, and code tools do not expose parameter schemas in 

627 # the retrieved tool definition; we return description only. 

628 description = getattr(rich, 'description', None) or description 

629 raw_params = None 

630 # SDK returns parameters/schema as a Pydantic model; ToolInfo wants a plain dict. 

631 if raw_params is not None: 

632 parameters = raw_params.model_dump() if hasattr(raw_params, 'model_dump') else raw_params 

633 except Exception as e: # noqa: BLE001 

634 logger.warning(f'Failed to enrich tool {tool_id}: {e} — attack strategies will use limited context') 

635 return ToolInfo(name=name, description=description, parameters=parameters, action_type=action_type)

646class ORQBackend(Backend): 

647 """Backend for ORQ-hosted agents. 

648 

649 Owns the ORQ SDK client. Creates ``ORQAgentTarget`` instances per job. 

650 Performs memory cleanup via the SDK. The shared SDK client keeps the 

651 configured retry budget for context retrieval, enrichment, and cleanup; 

652 ``ORQAgentTarget`` disables that budget per target operation because the 

653 orchestrator owns target retries. Maps ORQ exceptions to a normalized error 

654 taxonomy. 

655 """ 

656 

657 def __init__( 

658 self, 

659 *, 

660 orq_client: Any = None, 

661 timeout_ms: int | None = None, 

662 retry_count: int | None = None, 

663 retry_on_codes: list[int] | None = None, 

664 max_tool_continuations: int | None = None, 

665 ) -> None: 

666 super().__init__(name='orq') 

667 timeout_ms = timeout_ms or PIPELINE_CONFIG.target_agent_timeout_ms

669 self._max_tool_continuations = max_tool_continuations 

670 warn_ignored_target_retries(
