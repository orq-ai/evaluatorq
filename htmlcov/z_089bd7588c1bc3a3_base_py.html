96def _default_reasoning_effort() -> str | None: 

97 """Reasoning effort from ``EVALUATORQ_REASONING_EFFORT``, or ``None`` when unset. 

98 

99 There is deliberately no global default. Sending an effort the user did not ask 

100 for costs a rejected request plus a retry on every model that does not support 

101 the parameter, and it overrides the model's own tuned default on every model 

102 that does. Unset means "say nothing and let the model decide". 

103 

104 Set ``LLMCallConfig.reasoning_effort`` per-agent to override the env value 

105 (including an explicit ``None`` to opt out). ``""`` / ``none`` / ``off`` in the 

106 env var also resolve to ``None``. 

107 

108 A separate knob from red team's ``LLMConfig.target_reasoning_effort``, which 

109 configures the agent *under test* rather than the simulator's own LLM calls 

110 (user simulator, judge). Both ultimately resolve into a per-call 

111 ``LLMCallConfig.reasoning_effort`` — this one via the env fallback below when 

112 a `BaseAgent`'s config leaves it unset, that one via an explicit 

113 ``reasoning_effort=`` threaded into the target's backend construction. 

114 """ 

115 raw = os.environ.get('EVALUATORQ_REASONING_EFFORT', '').strip().lower() 

116 return raw if raw not in ('', 'none', 'off') else None

174def _mirror_gaps(config_fields: Iterable[str], agent_fields: Iterable[str]) -> tuple[set[str], set[str]]: 

175 """What the hand-maintained `_MIRRORED_FIELDS` mirror is missing, in both directions. 

176 

177 Returns the `LLMCallConfig` fields nothing mirrors, and the mirrored names 

178 `AgentConfig` has no attribute for. ``model`` and ``client`` are excluded 

179 because `_config_from_agent_config` passes those outside the loop. 

180 """ 

181 return ( 

182 set(config_fields) - set(_MIRRORED_FIELDS) - {'model', 'client'}, 

183 set(_MIRRORED_FIELDS) - set(agent_fields), 

184 ) 

185 

186 

187# Checked at import time, like the vulnerability registry: a field forgotten here vanishes from the request in silence. 

188_unmirrored, _unbacked = _mirror_gaps(LLMCallConfig.model_fields, {f.name for f in fields(AgentConfig)}) 

189if _unmirrored: 

190 raise RuntimeError( 

191 f'LLMCallConfig fields missing from _MIRRORED_FIELDS: {sorted(_unmirrored)}. Add each one there, and to ' 

192 'AgentConfig, or the legacy AgentConfig path drops it from the request in silence.' 

193 ) 

194if _unbacked: 

195 raise RuntimeError(f'_MIRRORED_FIELDS names fields AgentConfig does not have: {sorted(_unbacked)}.')

198def _config_from_agent_config(agent_cfg: AgentConfig) -> tuple[LLMCallConfig, str | None]: 

199 """Convert a legacy AgentConfig into a LLMCallConfig + optional api_key. 

200 

201 Only fields the caller actually set on ``agent_cfg`` (non-``None``) are 

202 passed through to the `LLMCallConfig` constructor, so 

203 `LLMCallConfig.model_fields_set` accurately reflects caller intent for the 

204 resolvers in `BaseAgent` (``_resolved_temperature`` etc.) — a field left at 

205 its `AgentConfig` default of ``None`` must NOT shadow the per-call-site 

206 literal / env fallback with `LLMCallConfig`'s own field default. 

207 

208 ``api`` is the one deliberate exception: it always lands in 

209 ``model_fields_set`` so that the legacy path stays pinned to 

210 ``chat_completions`` against `BaseAgent.DEFAULT_API`, regardless of 

211 whether the caller touched it. ``model`` is unconditional too, since it 

212 has a non-``None`` default on `AgentConfig` and is always meaningful. 

213 """ 

214 kwargs: dict[str, Any] = {'model': agent_cfg.model, 'api': agent_cfg.api} 

215 if agent_cfg.client is not None: 

216 kwargs['client'] = agent_cfg.client 

217 for field in _MIRRORED_FIELDS: 

218 value = getattr(agent_cfg, field) 

219 if field != 'api' and value is not None: 

220 kwargs[field] = value 

221 return LLMCallConfig(**kwargs), agent_cfg.api_key

224class BaseAgent(UsageTracking, ABC): 

225 """Abstract base class for simulation agents. 

226 

227 Provides common LLM interaction functionality with exponential-backoff 

228 retry logic and cumulative token-usage tracking. 

229 

230 **Client injection**: pass an existing ``AsyncOpenAI`` client via 

231 ``config.client`` to share a single HTTP connection across multiple agents. 

232 The agent will NOT close an injected client. 

233 """ 

234 

235 # Not LLMCallConfig's own chat_completions default: the judge sends function 

236 # tools and reasoning_effort together, which chat completions answers with a 400. 

237 DEFAULT_API: ClassVar[Literal['chat_completions', 'responses']] = 'responses' 

238 

239 # Set by a subclass whose request shape only one endpoint accepts. It wins over a 

240 # caller's `api`, because the shared config is chosen for the agent under test and 

241 # the judge's 400 would surface as a failed run, not as a bad setting. 

242 REQUIRED_API: ClassVar[Literal['chat_completions', 'responses'] | None] = None 

243 

244 def __init__(self, config: LLMCallConfig | AgentConfig | None = None) -> None: 

245 # Normalise legacy AgentConfig into LLMCallConfig 

246 extra_api_key: str | None = None 

247 if isinstance(config, AgentConfig): 

248 self.config, extra_api_key = _config_from_agent_config(config) 

249 else: 

250 self.config = config or LLMCallConfig(model=DEFAULT_MODEL) 

251 if self.REQUIRED_API is not None: 

252 if 'api' in self.config.model_fields_set and self.config.api != self.REQUIRED_API: 

253 logger.warning( 

254 '%s runs on the %s endpoint; llm_config.api=%r does not apply to it.', 

255 type(self).__name__, 

256 self.REQUIRED_API, 

257 self.config.api, 

258 ) 

259 self.config = self.config.model_copy(update={'api': self.REQUIRED_API}) 

260 elif 'api' not in self.config.model_fields_set: 

261 self.config = self.config.model_copy(update={'api': self.DEFAULT_API}) 

262 

263 self._client_owned: bool 

264 self._client: AsyncOpenAI = self._build_client(extra_api_key) 

265 self._model = self.config.model 

266 self.reset_usage()

327 def _build_client(self, api_key: str | None = None) -> AsyncOpenAI: 

328 """Construct (or reuse) an ``AsyncOpenAI`` client from ``self.config``. 

329 

330 Delegates to `evaluatorq.openresponses.client.build_simulation_client`. 

331 

332 Resolution order: 

333 1. ``self.config.client`` — injected client, used as-is (not owned). 

334 2. ``api_key`` argument (extracted from legacy ``AgentConfig.api_key``), 

335 treated as an ORQ key and routed through the Orq router. 

336 3. ``ORQ_API_KEY`` env var — routes through 

337 ``ORQ_BASE_URL/v3/router`` (default: ``https://my.orq.ai/v3/router``). 

338 4. ``OPENAI_API_KEY`` env var — uses the OpenAI SDK default base URL so 

339 traffic goes to OpenAI directly, not to the Orq router. 

340 

341 Retry owner: ``with_retry`` in ``_call_chat_completions`` / ``_call_responses``, 

342 bounded at ``config.retry_count + 1`` attempts. The SDK budget is disarmed 

343 here (``max_retries=0``) so the two cannot multiply. 

344 """ 

345 client, owned = build_simulation_client( 

346 self.config.client, 

347 extra_api_key=api_key, 

348 max_retries=0, 

349 ) 

350 self._client_owned = owned 

351 return client

353 def _resolved_temperature(self, call_value: float | None) -> float | None: 

354 """Effective temperature: an explicitly set ``self.config.temperature`` 

355 beats the per-call ``call_value`` — including an explicit ``None``, which 

356 opts this agent out of the call site's value on purpose. Unresolved means 

357 the request omits the parameter, which is what reasoning-class models 

358 need: they answer 400 to ``temperature`` at any value. 

359 

360 Gated on ``model_fields_set`` rather than on the value, matching 

361 `_resolved_max_tokens` / `_resolved_timeout_s` / `_resolved_reasoning_effort`. 

362 `LLMCallConfig` now defaults ``temperature`` to ``None``, so a value check 

363 could not tell an explicit ``None`` from an untouched field. 

364 """ 

365 if 'temperature' in self.config.model_fields_set: 

366 return self.config.temperature 

367 return call_value

369 def _resolved_max_tokens(self, call_value: int | None) -> int: 

370 """Effective max-tokens budget: explicit ``self.config.max_tokens`` beats 

371 ``call_value``, which beats the call-time-resolved env fallback 

372 (`_default_max_tokens`, EVALUATORQ_LLM_MAX_TOKENS). 

373 

374 Pair with `_max_tokens_advice` when building a truncation message: it 

375 derives from the same ``model_fields_set`` check, so the two can't drift. 

376 """ 

377 if 'max_tokens' in self.config.model_fields_set: 

378 return self.config.max_tokens 

379 return call_value if call_value is not None else _default_max_tokens()

381 def _max_tokens_advice(self, call_value: int | None) -> str: 

382 """Remedy text for a truncation message, naming whichever knob 

383 `_resolved_max_tokens` actually used for this agent. 

384 

385 Pass the same ``call_value`` that was handed to `_resolved_max_tokens`, 

386 so the message walks the identical three tiers: config, then the 

387 caller's per-call ``max_tokens=``, then the env fallback. 

388 

389 A user whose `LLMCallConfig.max_tokens` is pinned and raises 

390 ``EVALUATORQ_LLM_MAX_TOKENS`` instead sees no change and no signal why — 

391 the env var is only consulted when the config leaves ``max_tokens`` 

392 unset *and* the caller passed nothing. 

393 """ 

394 if 'max_tokens' in self.config.model_fields_set: 

395 return "raise max_tokens on this agent's LLMCallConfig" 

396 if call_value is not None: 

397 return 'raise the max_tokens argument passed to this call' 

398 return 'raise the budget via EVALUATORQ_LLM_MAX_TOKENS'

400 def _resolved_timeout_s(self, call_value: float | None) -> float: 

401 """Effective per-call timeout in seconds: explicit ``self.config.timeout_ms`` 

402 (converted from ms) beats ``call_value`` (already seconds — the unit 

403 `respond_async` / `_call_llm` use), which beats the call-time-resolved 

404 env fallback (`_default_timeout_s`, EVALUATORQ_LLM_TIMEOUT_S). 

405 """

408 def _resolved_reasoning_effort(self) -> str | None: 

409 """Effective reasoning effort: an explicitly set ``self.config.reasoning_effort``

424 async def _call_llm( 

425 self, 

426 messages: list[Message], 

427 *, 

428 temperature: float | None = None, 

429 max_tokens: int | None = None, 

430 timeout: float | None = None, 

431 tools: list[dict[str, Any]] | None = None, 

432 llm_purpose: str | None = None, 

433 volatile_tail: int = 0, 

434 ) -> LLMResult: 

435 """Call the LLM with retry logic, dispatching to chat or responses API. 

436 

437 Retries on rate-limit (429) and server errors (500+). All other errors 

438 are raised immediately. ``asyncio.TimeoutError`` is never retried. Retry 

439 is owned by ``with_retry``; client retries are disabled. 

440 

441 ``volatile_tail`` is the number of trailing messages this caller rebuilds 

442 every turn instead of appending to the transcript. It keeps the cache 

443 breakpoint off them on both paths — see `common.prompt_cache`: 

444 `apply_cache_breakpoints` on the chat path, `mark_responses_input` on the 

445 Responses path (which is the judge's default). 

446 """ 

447 if self.config.api == 'responses': 

448 return await self._call_responses( 

449 messages, 

450 temperature=temperature, 

451 max_tokens=max_tokens, 

452 timeout=timeout, 

453 tools=tools, 

454 llm_purpose=llm_purpose, 

455 volatile_tail=volatile_tail, 

456 ) 

457 return await self._call_chat_completions( 

458 messages, 

459 temperature=temperature, 

460 max_tokens=max_tokens, 

461 timeout=timeout, 

462 tools=tools, 

463 llm_purpose=llm_purpose, 

464 volatile_tail=volatile_tail, 

465 )

467 async def _call_chat_completions( 

468 self, 

469 messages: list[Message], 

470 *, 

471 temperature: float | None = None, 

472 max_tokens: int | None = None, 

473 timeout: float | None = None, 

474 tools: list[dict[str, Any]] | None = None, 

475 llm_purpose: str | None = None, 

476 volatile_tail: int = 0, 

477 ) -> LLMResult: 

478 """Call the LLM via the Chat Completions API with retry logic. 

479 

480 Effective ``temperature`` / ``max_tokens`` / ``timeout`` / reasoning 

481 effort come from `_resolved_temperature` / `_resolved_max_tokens` / 

482 `_resolved_timeout_s` / `_resolved_reasoning_effort`: an explicit 

483 ``self.config`` value wins, else this call site's own argument, else the 

484 call-time env fallback. Temperature has no env tier — nothing supplies one, 

485 so an unresolved temperature is omitted from the request rather than 

486 defaulted. ``self.config.extra_kwargs`` rides along last, so it can still override 

487 any of the above (matches `LLMCallConfig.request_params`'s chat-completions contract). 

488 """ 

489 temp = self._resolved_temperature(temperature) 

490 max_tok = self._resolved_max_tokens(max_tokens) 

491 timeout_s = self._resolved_timeout_s(timeout) 

492 reasoning_effort = self._resolved_reasoning_effort() 

493 

494 full_messages: list[dict[str, Any]] = [ 

495 {'role': 'system', 'content': self.system_prompt}, 

496 *[m.to_chat_completion() for m in messages], 

497 ] 

498 # Breakpoints on the system prompt and the end of the persisted transcript: 

499 # simulation replays a growing append-only prefix, so without them Anthropic 

500 # models re-encode the whole thing every turn (see common/prompt_cache.py). 

501 if caching_applies(self._client, self._model): 

502 full_messages = apply_cache_breakpoints(full_messages, volatile_tail=volatile_tail) 

503 

504 async with with_llm_span( 

505 model=self._model, 

506 operation='chat', 

507 temperature=temp, 

508 max_tokens=max_tok, 

509 purpose=llm_purpose, 

510 ) as span: 

511 call_extra: dict[str, Any] = dict(self.config.extra_kwargs) if self.config.extra_kwargs else {} 

512 if reasoning_effort and 'reasoning_effort' not in call_extra: 

513 call_extra['reasoning_effort'] = reasoning_effort 

514 # execute_chat_completion treats None as "no extras"; an empty dict 

515 # would be splatted as a no-op but reads as "the caller set extras". 

516 reasoning_kwargs: dict[str, Any] | None = call_extra or None 

517 

518 async def _do_call() -> LLMResult: 

519 finish_reason: str | None = None 

520 # Content-level retry inside one transport attempt: costs up to 2x the budget. 

521 for attempt in range(2): 

522 response, delta = await execute_chat_completion( 

523 client=self._client, 

524 model=self._model, 

525 messages=full_messages, 

526 span=span, 

527 timeout_s=timeout_s, 

528 temperature=temp, 

529 max_tokens=max_tok, 

530 tools=tools, 

531 extra_body=self.config.extra_body or None, 

532 extra_kwargs=reasoning_kwargs, 

533 ) 

534 if delta is not None: 

535 self._accumulate(delta) 

536 

537 choice = response.choices[0] if response.choices else None 

538 if not choice: 

539 raise RuntimeError(f'{self.name}: No choices in response') 

540 message = choice.message 

541 content = message.content 

542 tool_calls = list(message.tool_calls or []) 

543 if content or tool_calls: 

544 return LLMResult(content=content or '', tool_calls=tool_calls or None) 

545 finish_reason = choice.finish_reason 

546 if attempt == 0: 

547 logger.info( 

548 '%s._call_chat_completions: empty response (finish_reason=%s), retrying once', 

549 self.name, 

550 finish_reason, 

551 ) 

552 # Truncated before the model could emit text/tool call. Common with 

553 # reasoning models whose hidden reasoning exhausts the token budget. 

554 if finish_reason == 'length': 

555 raise RuntimeError( 

556 f'{self.name}._call_chat_completions: response truncated (finish_reason=length, ' 

557 f'max_tokens={max_tok}) before any text or tool call. The model — likely a reasoning ' 

558 f'model — ran out of tokens during reasoning; {self._max_tokens_advice(max_tokens)}.' 

559 ) 

560 raise RuntimeError( 

561 f'{self.name}._call_chat_completions: LLM returned no text and no tool calls after retry ' 

562 f'(finish_reason={finish_reason}). Check model and prompt.' 

563 ) 

564 

565 return await with_retry( 

566 _do_call, 

567 label=f'{self.name}._call_chat_completions', 

568 max_attempts=self.config.retry_count + 1, 

569 )

571 async def _call_responses( 

572 self, 

573 messages: list[Message], 

574 *, 

575 temperature: float | None = None, 

576 max_tokens: int | None = None, 

577 timeout: float | None = None, 

578 tools: list[dict[str, Any]] | None = None, 

579 llm_purpose: str | None = None, 

580 volatile_tail: int = 0, 

581 ) -> LLMResult: 

582 """Call the LLM via the OpenAI Responses API with retry logic. 

583 

584 The Responses API uses ``input`` (list of message dicts) and 

585 ``instructions`` (system prompt) instead of a ``messages`` list. 

586 Text is extracted from ``response.output`` items that carry a 

587 ``content`` list of parts with a ``text`` attribute. 

588 

589 Effective ``temperature`` / ``max_tokens`` / ``timeout`` / reasoning 

590 effort follow the same config-beats-call-site-beats-fallback order as 

591 `_call_chat_completions`, temperature's missing env tier included 

592 (see `_resolved_temperature` etc.). 

593 The request itself goes through `common.llm_call.execute_response`, so 

594 this leg gets the same slot limiting, reasoning-rejection memo, pipeline 

595 metadata, trace headers and `price_usage` call as every other Responses 

596 caller. `self.config.extra_kwargs` is merged in there last (structural 

597 keys guarded by `check_reserved_keys`), so it can override any of the 

598 above — `LLMCallConfig.request_params`'s Responses contract. 

599 """ 

600 timeout_s = self._resolved_timeout_s(timeout) 

601 resolved_temp = self._resolved_temperature(temperature) 

602 max_tok = self._resolved_max_tokens(max_tokens) 

603 reasoning_effort = self._resolved_reasoning_effort() 

604 

605 # Canonical renderer: an assistant turn must arrive as output_text parts or 

606 # the Orq router silently drops it, leaving the judge blind to the agent's 

607 # replies (RES-1308). Never hand-build this list. 

608 input_messages = messages_to_responses_input(messages) 

609 # A per-item breakpoint, not the top-level switch: the latter marks the end 

610 # of the whole input, so a caller that rebuilds its trailing item (the 

611 # judge) writes every turn and reads none. `volatile_tail` counts messages 

612 # and this list counts items, which are not 1:1 — one tool-calling assistant 

613 # message renders to several items. 

614 if caching_applies(self._client, self._model): 

615 input_messages = mark_responses_input( 

616 input_messages, 

617 volatile_items=responses_volatile_items(messages, volatile_tail=volatile_tail), 

618 ) 

619 

620 async with with_llm_span( 

621 model=self._model, 

622 operation='responses', 

623 temperature=resolved_temp, 

624 max_tokens=max_tok, 

625 purpose=llm_purpose, 

626 ) as span: 

627 # Responses API sends system context via `instructions`, not as a 

628 # message in `input`. Record what is actually sent so the span 

629 # matches the real request shape (mirrors _call_chat_completions 

630 # which records full_messages including the system entry). 

631 if span is not None: 

632 span.set_attribute('gen_ai.request.instructions', self.system_prompt[:2000]) 

633 # Record from `messages`, NOT from `input_messages`: the wire payload 

634 # renders an assistant turn as `[{'type': 'output_text', ...}]`, which 

635 # `record_llm_input` would `str()` into a Python repr on the span. 

636 # Mirrors runner/simulation.py's target_call span. See CLAUDE.md's 

637 # `content_to_text` row. `record_input=False` below keeps 

638 # `execute_response` from overwriting it with the wire shape. 

639 record_llm_input(span, [{'role': m.role, 'content': span_message_text(m.content)} for m in messages]) 

640 

641 # Config body layered last: the caller's keys win per key, the call 

642 # site's thread grouping survives the ones they did not set. 

643 extra_body = thread_body_param() if client_routes_through_orq(self._client) else {} 

644 if self.config.extra_body: 

645 extra_body = {**extra_body, **self.config.extra_body} 

646 

647 async def _do_call() -> LLMResult: 

648 response, usage = await execute_response( 

649 client=self._client, 

650 model=self._model, 

651 messages=input_messages, 

652 span=span, 

653 timeout_s=timeout_s, 

654 temperature=resolved_temp, 

655 max_output_tokens=max_tok, 

656 reasoning_effort=reasoning_effort, 

657 instructions=self.system_prompt, 

658 tools=[_responses_tool_schema(tool) for tool in tools] if tools else None, 

659 record_input=False, 

660 extra_body=extra_body or None, 

661 extra_kwargs=self.config.extra_kwargs or None, 

662 ) 

663 self._accumulate(usage) 

664 

665 stop_reason = responses_stop_reason(response) 

666 if stop_reason == 'length': 

667 raise RuntimeError( 

668 f'{self.name}._call_responses: response truncated at max_output_tokens=' 

669 f'{max_tok}; {self._max_tokens_advice(max_tokens)}.' 

670 ) 

671 refusal = first_responses_refusal(response) 

672 output_items = AgentResponse.from_openresponses(response).output 

673 

674 # Separate text from tool-call items; isinstance guards prevent 

675 # ReasoningOutputItem.text leaking into response content. 

676 text_items = [i for i in output_items if isinstance(i, TextOutputItem)] 

677 tool_call_items = [i for i in output_items if isinstance(i, ToolCallOutputItem)] 

678 

679 if not text_items and not tool_call_items: 

680 # No text, no tool calls — warn but don't raise (redteam callers 

681 # may handle empty). Surface the reason so it's clear to the user: 

682 # reason=max_output_tokens means the budget was too small. 

683 incomplete = getattr(response, 'incomplete_details', None) 

684 reason = getattr(incomplete, 'reason', None) if incomplete else getattr(response, 'status', None) 

685 logger.warning( 

686 '%s._call_responses: empty response — no text or tool calls (model=%s, reason=%s). ' 

687 'If reason=max_output_tokens, %s.', 

688 self.name, 

689 self.config.model, 

690 reason, 

691 self._max_tokens_advice(max_tokens), 

692 ) 

693 

694 text = ''.join(getattr(i, 'text', '') for i in text_items) 

695 result = LLMResult(content=text, refusal=refusal) 

696 if tool_call_items: 

697 result.tool_calls = [ 

698 StrategyToolCall( 

699 id=item.call_id, 

700 function=FunctionCall(name=item.name, arguments=item.arguments), 

701 ) 

702 for item in tool_call_items 

703 ] 

704 return result 

705 

706 return await with_retry( 

707 _do_call, 

708 label=f'{self.name}._call_responses', 

709 max_attempts=self.config.retry_count + 1, 

710 )

713def _responses_tool_schema(tool: dict[str, Any]) -> dict[str, Any]: 

714 """Convert chat-completions function tools to Responses function tools.""" 

715 if tool.get('type') == 'function' and isinstance(tool.get('function'), dict): 

716 fn = tool['function'] 

717 return { 

718 'type': 'function', 

719 'name': fn.get('name'), 

720 'description': fn.get('description'), 

721 'parameters': fn.get('parameters') or {'type': 'object', 'properties': {}}, 

722 } 

723 return tool
